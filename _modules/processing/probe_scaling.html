def _require_float(params: dict[str, Any], name: str) -> float:
    """Brief: Require a named geometry dimension and coerce it to float.

    Inputs:
        params: Geometry parameter mapping (e.g. L, R) [m].
        name: Required key in ``params``.

    Outputs:
        float: Parameter value [m].
    """
    if name not in params or params[name] is None:
        raise KeyError(f"Missing geometry parameter {name!r}")
    return float(params[name])

def _resolve_geometry(
    geometry_params: dict[str, Any],
    geometry_type: str,
) -> tuple[float, float, float, float]:
    """Brief: Normalize cylinder/prism/sphere dimensions into length and cross-section radii.

    Inputs:
        geometry_params: Family-specific dimensions from ``GeometryParams.probe_scaling_params`` [m].
        geometry_type: ``cylinder``, ``prism``, or ``sphere``.

    Outputs:
        tuple[float, float, float, float]: ``(length, radius, width, height)`` analogues [m].
    """
    gtype = (geometry_type or "").strip().lower()
    if gtype == "sphere":
        radius = _require_float(geometry_params, "R_sphere")
        cap = _require_float(geometry_params, "R_cap")
        return radius, cap, cap, cap
    if gtype == "prism":
        length = _require_float(geometry_params, "L")
        width = _require_float(geometry_params, "W")
        height = _require_float(geometry_params, "H")
        return length, min(width, height), width, height
    if gtype == "cylinder":
        length = _require_float(geometry_params, "L")
        radius = _require_float(geometry_params, "R")
        return length, radius, radius, radius
    raise KeyError(f"Unknown geometry_type {geometry_type!r}")

def _l_metal_max(
    geometry_params: dict[str, Any],
    geometry_type: str,
    variant: str = "full",
) -> float:
    """Brief: Characteristic metal length used to scale exterior magnetic probe bands.

    Inputs:
        geometry_params: Geometry dimensions [m].
        geometry_type: ``cylinder``, ``prism``, or ``sphere``.
        variant: Topology variant; hollow spheres use ``R_sphere`` as the metal length.

    Outputs:
        float: ``L_metal_max`` scale length [m].
    """
    length, radius, width, height = _resolve_geometry(geometry_params, geometry_type)
    gtype = geometry_type.lower()
    if gtype == "prism":
        return max(length, 2.0 * width, 2.0 * height)
    if gtype == "sphere":
        if "hollow" in (variant or "full").lower():
            return length
        return 2.0 * length
    return max(length, 2.0 * radius)

def compute_magnetic_trusted_band(
    geometry_params: dict[str, Any],
    geometry_type: str,
    variant: str = "full",
) -> dict[str, float]:
    """Brief: Compute exterior magnetic trusted radial band and mid-band probe radius.

    Inputs:
        geometry_params: Geometry dimensions for band scaling [m].
        geometry_type: ``cylinder``, ``prism``, or ``sphere``.
        variant: Topology variant affecting ``L_metal_max`` for spheres.

    Outputs:
        dict[str, float]: ``rho_in_m``, ``rho_out_m``, ``probe_rho_m``, ``L_metal_max_m`` [m].
    """
    bext = load_processing_defaults().extraction.bext
    r_inf = bext.r_inf_trusted
    r_sup = bext.r_sup_trusted
    band_frac = bext.probe_band_fraction

    length, radius, width, height = _resolve_geometry(geometry_params, geometry_type)
    gtype = geometry_type.lower()
    lmm = _l_metal_max(geometry_params, geometry_type, variant)

    if gtype == "sphere":
        r_body = length
        d_near_max = 0.75 * r_body
        rho_in = r_body + d_near_max
        rho_out = r_body + 5.0 * r_body
        if rho_out <= rho_in:
            rho_out = rho_in * 1.5
    elif gtype == "prism":
        cross_half = min(width, height)
        d_near_max = 0.75 * cross_half
        mag_d_max = 5.0 * cross_half
        rho_in = height + d_near_max
        rho_out = min(height + mag_d_max, 5.0 * lmm)
        if rho_out <= rho_in:
            rho_out = rho_in + max(cross_half, 1.0e-6)
    else:
        d_near_max = 0.75 * radius
        rho_in = max(radius + d_near_max, r_inf * lmm)
        rho_out = r_sup * lmm
        if rho_out <= rho_in:
            rho_out = rho_in * 1.25

    probe_rho = rho_in + band_frac * (rho_out - rho_in)
    return {
        "rho_in_m": float(rho_in),
        "rho_out_m": float(rho_out),
        "probe_rho_m": float(probe_rho),
        "L_metal_max_m": float(lmm),
    }

def compute_bext_probe_positions(
    geometry_params: dict[str, Any],
    geometry_type: str,
    variant: str = "full",
    *,
    band_override: dict[str, float] | None = None,
) -> list[tuple[float, float, float]]:
    """Brief: Place exterior midplane B-field probes within the trusted magnetic band.

    Inputs:
        geometry_params: Geometry dimensions [m].
        geometry_type: ``cylinder``, ``prism``, or ``sphere``.
        variant: Topology variant for band computation.
        band_override: Optional ``rho_in_m``/``rho_out_m`` override [m].

    Outputs:
        list[tuple[float, float, float]]: Probe XYZ positions on the midplane (x≈0) [m].
    """
    bext = load_processing_defaults().extraction.bext
    if band_override is not None:
        rho_in = float(band_override["rho_in_m"])
        rho_out = float(band_override["rho_out_m"])
    else:
        band = compute_magnetic_trusted_band(geometry_params, geometry_type, variant)
        rho_in = float(band["rho_in_m"])
        rho_out = float(band["rho_out_m"])
    span = max(rho_out - rho_in, 0.0)

    points: list[tuple[float, float, float]] = []
    fractions = bext.probe_band_fractions
    for i, direction in enumerate(bext.point_directions):
        dx, dy, dz = float(direction[0]), float(direction[1]), float(direction[2])
        lat = math.hypot(dy, dz)
        if lat < 1.0e-15:
            continue
        uy, uz = dy / lat, dz / lat
        phi = math.atan2(uz, uy) % (2.0 * math.pi)
        if min(phi, abs(2.0 * math.pi - phi)) < math.radians(5.0):
            continue
        frac = float(fractions[min(i, len(fractions) - 1)])
        rho = rho_in + frac * span
        points.append((0.0, rho * uy, rho * uz))

    if not points:
        raise ValueError("extraction.bext_defaults.point_directions produced no midplane probes")
    return points

def _fmt_coord(value: float) -> str:
    """Brief: Format a coordinate for HTML probe hints with compact significant digits.

    Inputs:
        value: Coordinate or length to display [m].

    Outputs:
        str: Compact numeric text, or ``\"0\"`` for near-zero values.
    """
    if abs(value) < 1e-15:
        return "0"
    return f"{value:.4g}"

def _abs_le(axis: str, limit: float) -> str:
    """Brief: Build an absolute-value inequality fragment for interior probe hints.

    Inputs:
        axis: Axis symbol (``x``, ``y``, or ``z``).
        limit: Half-extent or radius bound [m].

    Outputs:
        str: HTML text of the form ``abs(axis) ≤ <b>limit</b>``.
    """
    return f"|{axis}| \u2264 {_b(limit)}"

def _l_metal_detail(
    geometry_params: dict[str, Any],
    geometry_type: str,
    variant: str = "full",
) -> tuple[float, str]:
    """Brief: Compute ``L_metal_max`` and a human-readable formula detail string.

    Inputs:
        geometry_params: Geometry dimensions [m].
        geometry_type: ``cylinder``, ``prism``, or ``sphere``.
        variant: Topology variant. Hollow detail is ``R_sphere``; other spheres use ``2*R_sphere``.

    Outputs:
        tuple[float, str]: ``(L_metal_max [m], formula detail text)``.
    """
    length, radius, width, height = _resolve_geometry(geometry_params, geometry_type)
    lmm = _l_metal_max(geometry_params, geometry_type, variant)
    gtype = geometry_type.lower()
    if gtype == "prism":
        detail = (
            f"max(L,2W,2H)=max({_fmt_coord(length)},{_fmt_coord(2 * width)},{_fmt_coord(2 * height)})"
        )
    elif gtype == "sphere":
        if "hollow" in (variant or "full").lower():
            detail = f"R_sphere={_fmt_coord(length)}"
        else:
            detail = f"2*R_sphere={_fmt_coord(2.0 * length)}"
    else:
        detail = f"max(L,2R)=max({_fmt_coord(length)},{_fmt_coord(2 * radius)})"
    return lmm, detail

def compute_bext_sampling_hint(
    geometry_params: dict[str, Any],
    geometry_type: str,
    variant: str = "full",
) -> str:
    """Brief: Build an HTML hint describing the trusted exterior magnetic probe band.

    Inputs:
        geometry_params: Geometry dimensions [m].
        geometry_type: ``cylinder``, ``prism``, or ``sphere``.
        variant: Topology variant for band / ``L_metal`` detail.

    Outputs:
        str: HTML sentence with midplane trusted radial band bounds [m].
    """
    band = compute_magnetic_trusted_band(geometry_params, geometry_type, variant)
    rho_in = float(band["rho_in_m"])
    rho_out = float(band["rho_out_m"])
    lmm, l_detail = _l_metal_detail(geometry_params, geometry_type, variant)
    radial_symbol = "r" if geometry_type.lower() == "sphere" else _RHO_PROBE
    return (
        f"Trusted magnetic band (midplane x\u22480): {_b(rho_in)} \u2264 "
        f"{radial_symbol} \u2264 {_b(rho_out)} m "
        f"(L_metal={l_detail}={_b(lmm)} m). "
        "Use midplane probes within this band, away from the conductor and far boundary."
    )

def compute_jint_probe_hint(
    geometry_params: dict[str, Any],
    geometry_type: str,
) -> str:
    """Brief: Build an HTML hint for allowed interior current-density probe extents.

    Inputs:
        geometry_params: Geometry dimensions [m].
        geometry_type: ``cylinder``, ``prism``, or ``sphere``.

    Outputs:
        str: HTML sentence describing the conductor interior box or sphere bounds [m].
        A sphere bound is ``r <= R_sphere``.
    """
    length, radius, width, height = _resolve_geometry(geometry_params, geometry_type)
    gtype = geometry_type.lower()
    if gtype == "prism":
        return (
            "Inside conductor: "
            f"{_abs_le('x', length / 2.0)} (L={_b(length)}), "
            f"{_abs_le('y', width)} (W={_b(width)}), "
            f"{_abs_le('z', height)} (H={_b(height)}) m."
        )
    if gtype == "sphere":
        return (
            f"Inside metal sphere (R_sphere={_b(length)}): "
            f"{_abs_le('x', length)}, {_abs_le('y', length)}, {_abs_le('z', length)} m, "
            f"r \u2264 {_b(length)} m."
        )
    return (
        "Inside conductor: "
        f"{_abs_le('x', length / 2.0)} (|x| \u2264 L/2), "
        f"{_abs_le('y', radius)}, {_abs_le('z', radius)} (\u221a(y\u00b2+z\u00b2) \u2264 R)."
    )

def compute_jint_probe_positions(
    geometry_params: dict[str, Any],
    geometry_type: str,
    *,
    variant: str = "full",
) -> list[tuple[float, float, float]]:
    """Brief: Place interior J-probe points from dimensionless multipliers and Bext scale.

    Inputs:
        geometry_params: Geometry dimensions [m].
        geometry_type: ``cylinder``, ``prism``, or ``sphere``.

    Outputs:
        list[tuple[float, float, float]]: Interior probe XYZ positions [m].
    """
    jint = load_processing_defaults().extraction.jint
    length, radius, width, height = _resolve_geometry(geometry_params, geometry_type)

    if geometry_type.lower() == "sphere":
        radius_body = length

        if variant.lower() == "hollow":
           wall_thickness = float(geometry_params["T_outer"])
           inner_radius = radius_body - wall_thickness
           probe_radius = 0.5 * (inner_radius + radius_body)

           direction_x = 5.0 / 13.0
           direction_z = 12.0 / 13.0

           return [
               (0.0, 0.0, 0.0),
               (direction_x * probe_radius, 0.0, direction_z * probe_radius),
               (-direction_x * probe_radius, 0.0, -direction_z * probe_radius),
           ]
        
        if variant.lower() == "layer_over_hollow":
            t_inner = float(geometry_params["T_inner"])
            t_outer = float(geometry_params["T_outer"])

            r_outer = radius_body
            r_interface = r_outer - t_outer
            r_cavity = r_interface - t_inner

            inner_probe_radius = 0.5 * (r_cavity + r_interface)
            outer_probe_radius = 0.5 * (r_interface + r_outer)

            direction_x = 5.0 / 13.0
            direction_z = 12.0 / 13.0

            return [
                (0.0, 0.0, 0.0),
                (
                    direction_x * inner_probe_radius,
                    0.0,
                    direction_z * inner_probe_radius,
                ),
                (
                    -direction_x * outer_probe_radius,
                    0.0,
                    -direction_z * outer_probe_radius,
                ),
           ]
           
        return [
            (0.0, 0.0, 0.0),
            (0.25 * radius_body, 0.0, 0.60 * radius_body),
            (-0.25 * radius_body, 0.0, -0.60 * radius_body),
        ]

    gtype = geometry_type.lower()
    variant_key = variant.lower()
    cross_extent = height if gtype == "prism" else radius

    if variant_key == "hollow":
        t_outer = float(geometry_params["T_outer"])
        inner_extent = cross_extent - t_outer
        probe_extent = 0.5 * (inner_extent + cross_extent)
        z_positive = probe_extent
        z_negative = -probe_extent
    elif variant_key == "layer_over_full":
        t_outer = float(geometry_params["T_outer"])
        interface_extent = cross_extent - t_outer
        z_positive = 0.5 * interface_extent
        z_negative = -0.5 * (interface_extent + cross_extent)
    elif variant_key == "layer_over_hollow":
        t_inner = float(geometry_params["T_inner"])
        t_outer = float(geometry_params["T_outer"])
        interface_extent = cross_extent - t_outer
        cavity_extent = interface_extent - t_inner
        z_positive = 0.5 * (cavity_extent + interface_extent)
        z_negative = -0.5 * (interface_extent + cross_extent)
    else:
        z_positive = 0.5 * cross_extent
        z_negative = -0.5 * cross_extent

    return [
        (0.0, 0.0, 0.0),
        (0.25 * length, 0.0, z_positive),
        (-0.25 * length, 0.0, z_negative),
    ]
