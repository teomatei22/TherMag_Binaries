@dataclass
class GeometryParams:
    """Brief: Conductor dimensions and hollow/layer variant parsed from ``geometry.geo``.

    Inputs:
        geometry_type: Family ``cylinder``, ``prism``, or ``sphere``.
        variant: Topology ``full``, ``hollow``, or ``layer_over_*``.
        is_two_layer: Whether an inner material layer is present.
        T_inner: Inner-layer wall thickness [m].
        T_outer: Outer-layer / hollow wall thickness [m].
        L: Axial length (cylinder/prism) [m].
        R: Cylinder radius [m].
        W: Prism half-width [m].
        H: Prism half-height [m].
        R_sphere: Metal sphere radius [m].
        R_cap: Sphere contact-cap radius [m].
        R_sph: Optional far-field air-sphere radius [m].

    Outputs:
        GeometryParams: Dimension container for glyph scale and probe band layout.
    """

    geometry_type: str
    variant: str = "full"
    is_two_layer: bool = False
    T_inner: float = 0.0
    T_outer: float = 0.0
    L: Optional[float] = None
    R: Optional[float] = None
    W: Optional[float] = None
    H: Optional[float] = None
    R_sphere: Optional[float] = None
    R_cap: Optional[float] = None
    R_sph: Optional[float] = None

    def cavity_thickness_m(self) -> float:
        """Brief: Total cavity wall thickness for hollow or layered-hollow variants.

        Inputs:
            N/A: Uses ``self.variant``, ``self.T_inner``, ``self.T_outer``.

        Outputs:
            float: Cavity wall thickness [m]; ``0.0`` for solid full bodies.
        """
        if self.variant == "hollow":
            return float(self.T_outer)
        if self.variant == "layer_over_hollow":
            return float(self.T_inner + self.T_outer)
        return 0.0

    def characteristic_extent(self) -> float:
        """Brief: Characteristic conductor extent used for camera and glyph span scaling.

        Inputs:
            N/A: Uses family dimensions on ``self`` (L/R, L/W/H, or R_sphere/R_cap).

        Outputs:
            float: Positive characteristic length [m] (floor ``1e-9``).
        """
        gtype = str(self.geometry_type).lower()
        if gtype == "sphere":
            span = max(2.0 * float(self.R_sphere or 0.0), 2.0 * float(self.R_cap or 0.0))
        elif gtype == "prism":
            span = max(
                float(self.L or 0.0),
                float(self.W or 0.0),
                float(self.H or 0.0),
            )
        else:
            span = max(float(self.L or 0.0), 2.0 * float(self.R or 0.0))
        return max(span, 1.0e-9)

    def probe_scaling_params(self) -> dict[str, float]:
        """Brief: Export the subset of dimensions required by probe-band / Jint scaling.

        Inputs:
            N/A: Uses family-required fields on ``self``.

        Outputs:
            dict[str, float]: Geometry keys (e.g. ``L``/``R``, ``L``/``W``/``H``, ``R_sphere``/``R_cap``) [m].
        """
        gtype = str(self.geometry_type).lower()
        if gtype == "sphere":
            if self.R_sphere is None or self.R_cap is None:
                raise ValueError("Sphere geometry requires R_sphere and R_cap")
            return {"R_sphere": float(self.R_sphere), "R_cap": float(self.R_cap)}
        if gtype == "prism":
            if self.L is None or self.W is None or self.H is None:
                raise ValueError("Prism geometry requires L, W, and H")
            return {
                "L": float(self.L),
                "W": float(self.W),
                "H": float(self.H),
            }
        if self.L is None or self.R is None:
            raise ValueError("Cylinder geometry requires L and R")
        return {"L": float(self.L), "R": float(self.R)}

    @classmethod
    def from_geo_file(
        cls, geo_path: str, pp_settings: Optional[dict[str, Any]] = None
    ) -> GeometryParams:
        """Brief: Parse conductor dimensions and variant from a Gmsh ``geometry.geo`` file.

        Inputs:
            geo_path: Path to ``geometry.geo`` in the FEM run directory.
            pp_settings: Optional post-processing settings; loaded from the run root if omitted.

        Outputs:
            GeometryParams: Populated geometry for the detected family and variant.
            A sphere stores metal radius ``R_sphere`` and air-domain radius ``R_sph``.
        """
        root = str(Path(geo_path).parent)
        pp = pp_settings if pp_settings is not None else load_post_processing_settings(root)
        detection = pp["geometry_detection"]
        geo_file = Path(geo_path)
        text = geo_file.read_text(encoding="utf-8")
        vals = parse_geo_assignments(text)
        contract = contract_from_text(text)
        family = contract.family
        if family == "unknown":
            name_lower = geo_file.name.lower()
            if any(s.lower() in name_lower for s in detection["sphere_path_substrings"]) or any(
                marker in text for marker in detection["sphere_content_markers"]
            ):
                family = "sphere"
            elif first_value(vals, ("W",)) is not None and first_value(vals, ("H",)) is not None:
                family = "prism"
            else:
                family = str(detection["default_geometry_type"])

        r_sph = domain_radius_m(vals)
        variant = contract.variant if contract.variant != "unknown" else "full"
        is_two_layer = variant in ("layer_over_full", "layer_over_hollow")
        thicknesses = {
            "T_inner": float(first_value(vals, ("T_inner",)) or 0.0),
            "T_outer": float(first_value(vals, ("T_outer",)) or 0.0),
        }

        if family == "sphere":
            radius = first_value(vals, ("R_sphere",))
            r_cap = first_value(vals, ("R_cap",))
            if radius is None or r_cap is None:
                raise ValueError(f"{geo_file} must define R_sphere and R_cap for sphere geometry")
            return cls(
                geometry_type="sphere",
                R_sphere=float(radius),
                R_cap=float(r_cap),
                R_sph=r_sph,
                is_two_layer=is_two_layer,
                variant=variant,
                **thicknesses,
            )

        if family == "prism":
            length = first_value(vals, ("L",))
            width = first_value(vals, ("W",))
            height = first_value(vals, ("H",))
            if length is None or width is None or height is None:
                raise ValueError(f"{geo_file} must define L, W, and H for prism geometry")
            return cls(
                geometry_type="prism",
                L=float(length),
                W=float(width),
                H=float(height),
                R_sph=r_sph,
                is_two_layer=is_two_layer,
                variant=variant,
                **thicknesses,
            )

        length = first_value(vals, ("L",))
        radius = first_value(vals, ("R",))
        if length is None or radius is None:
            raise ValueError(f"{geo_file} must define L and R for cylinder geometry")
        return cls(
            geometry_type="cylinder",
            L=float(length),
            R=float(radius),
            R_sph=r_sph,
            is_two_layer=is_two_layer,
            variant=variant,
            **thicknesses,
        )

@dataclass
class ModelConfig:
    """Brief: Mesh body IDs and model layout for conductive vs air thresholding.

    Inputs:
        model_type: Layout key (e.g. ``full_cylinder``, ``layer_over_hollow_prism``).
        geometry_type: Family ``cylinder``, ``prism``, or ``sphere``.
        outer_material_id: Conductive outer body GeometryIds value.
        outer_air_id: Exterior air body GeometryIds value.
        inner_air_id: Optional cavity air body id.
        inner_material_id: Optional inner conductive layer body id.

    Outputs:
        ModelConfig: Body-id configuration for ParaView thresholds and coloring.
    """

    model_type: str = "full_cylinder"
    geometry_type: str = "cylinder"
    outer_material_id: int = field(
        default_factory=lambda: int(_BUNDLED_PP_SETTINGS["model_config_defaults"]["outer_material_id"])
    )
    outer_air_id: int = field(
        default_factory=lambda: int(_BUNDLED_PP_SETTINGS["model_config_defaults"]["outer_air_id"])
    )
    inner_air_id: Optional[int] = None
    inner_material_id: Optional[int] = None

    def conductive_analysis_range(self) -> tuple[float, float]:
        """Brief: GeometryIds threshold range selecting conductive material bodies.

        Inputs:
            N/A: Uses ``self.inner_material_id``, ``self.outer_material_id``, ``self.model_type``.

        Outputs:
            tuple[float, float]: Inclusive lower/upper GeometryIds for conductive volumes.
        """
        if self.inner_material_id and self.model_type in _LAYER_OVER_MODEL_TYPES:
            return 1.0, float(self.inner_material_id)
        outer_material_id = float(self.outer_material_id)
        return outer_material_id, outer_material_id

    def validate_outer_air_sampling(self) -> None:
        """Brief: Ensure exterior B-field sampling targets outer air, not metal or cavity.

        Inputs:
            N/A: Uses ``self.outer_air_id``, ``self.outer_material_id``, ``self.inner_air_id``.

        Outputs:
            None: Raises ``RuntimeError`` when outer air id collides with metal or cavity.
        """
        outer_air_id = int(self.outer_air_id)
        outer_material_id = int(self.outer_material_id)
        if outer_air_id == outer_material_id:
            raise RuntimeError(
                f"outer_air_id ({outer_air_id}) equals outer_material_id for model "
                f"{self.model_type!r}; B-field sampling must use the exterior air body."
            )
        inner_air_id = self.inner_air_id
        if inner_air_id is not None and outer_air_id == int(inner_air_id):
            raise RuntimeError(
                f"outer_air_id ({outer_air_id}) equals inner_air_id for model "
                f"{self.model_type!r}; hollow meshes require outer_air_id "
                f"for the surrounding air body, not the cavity."
            )

    @classmethod
    def detect(
        cls,
        cwd: str,
        pp_settings: Optional[dict[str, Any]] = None,
    ) -> ModelConfig:
        """Brief: Detect model layout and body IDs from geometry, mesh names, and defaults.

        Inputs:
            cwd: FEM run working directory containing ``geometry.geo`` and mesh metadata.
            pp_settings: Optional post-processing settings; loaded from ``cwd`` if omitted.

        Outputs:
            ModelConfig: Resolved ``model_type`` and conductive/air GeometryIds.
        """
        pp = pp_settings if pp_settings is not None else load_post_processing_settings(cwd)
        mc_def = pp["model_config_defaults"]
        config = cls(
            outer_material_id=int(mc_def["outer_material_id"]),
            outer_air_id=int(mc_def["outer_air_id"]),
            inner_air_id=mc_def["inner_air_id"],
            inner_material_id=mc_def["inner_material_id"],
        )

        geo_file = os.path.join(cwd, "geometry.geo")
        geo_params = GeometryParams.from_geo_file(geo_file, pp)
        content = Path(geo_file).read_text(encoding="utf-8")
        contract = contract_from_text(content)
        config.geometry_type = (
            contract.family if contract.family != "unknown" else geo_params.geometry_type
        )

        variant = contract.variant if contract.variant != "unknown" else geo_params.variant
        is_hollow = variant in ("hollow", "layer_over_hollow")
        is_two_layer = variant in ("layer_over_full", "layer_over_hollow") or geo_params.is_two_layer

        mesh_name_ids = load_mesh_name_ids(Path(cwd))
        if "outer_material" in mesh_name_ids:
            config.outer_material_id = int(mesh_name_ids["outer_material"])
        if "outer_air" in mesh_name_ids:
            config.outer_air_id = int(mesh_name_ids["outer_air"])
        if "inner_air" in mesh_name_ids:
            config.inner_air_id = int(mesh_name_ids["inner_air"])
            is_hollow = True
        if "inner_material" in mesh_name_ids:
            config.inner_material_id = int(mesh_name_ids["inner_material"])
            is_two_layer = True

        layouts = pp["model_layouts"]

        if config.geometry_type == "prism":
            if is_two_layer or geo_params.is_two_layer:
                config.model_type = (
                    "layer_over_hollow_prism" if is_hollow else "layer_over_full_prism"
                )
            elif is_hollow:
                config.model_type = "hollow_prism"
            else:
                config.model_type = "full_prism"
        elif config.geometry_type == "sphere":
            if is_two_layer or geo_params.is_two_layer:
                config.model_type = (
                    "layer_over_hollow_sphere" if is_hollow else "layer_over_full_sphere"
                )
            elif is_hollow:
                config.model_type = "hollow_sphere"
            else:
                config.model_type = "full_sphere"
        else:
            if is_two_layer:
                config.model_type = (
                    "layer_over_hollow_cylinder" if is_hollow else "layer_over_full_cylinder"
                )
            elif is_hollow:
                config.model_type = "hollow_cylinder"
            else:
                config.model_type = "full_cylinder"

        layout = layouts.get(config.model_type)
        if layout is None:
            raise KeyError(f"Unknown post-processing model_type {config.model_type!r}")

        config.outer_material_id = int(layout["outer_material_id"])
        config.outer_air_id = int(layout["outer_air_id"])
        config.inner_air_id = layout.get("inner_air_id")
        config.inner_material_id = mesh_name_ids.get("inner_material") or layout.get("inner_material_id")
        if "outer_material" in mesh_name_ids:
            config.outer_material_id = int(mesh_name_ids["outer_material"])
        if "outer_air" in mesh_name_ids:
            config.outer_air_id = int(mesh_name_ids["outer_air"])
        if "inner_air" in mesh_name_ids:
            config.inner_air_id = int(mesh_name_ids["inner_air"])

        return config
