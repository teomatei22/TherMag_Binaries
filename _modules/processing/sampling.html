def idw_weights(distances: np.ndarray, *, epsilon: float, on_node_epsilon: float) -> tuple[np.ndarray | None, int | None, dict[str, Any]]:
    """Brief: Build IDW weights or detect an exact mesh-node hit for probe sampling.

    Inputs:
        distances: Neighbor distances from probe to mesh nodes [m].
        epsilon: Softening length added to distance before squaring [m].
        on_node_epsilon: Distance below which the probe is treated as on-node [m].

    Outputs:
        tuple: ``(weights, exact_local_index, diagnostics)``; weights is ``None`` on exact hit.
    """
    d_k = np.asarray(distances, dtype=float)
    if d_k.size == 0:
        raise RuntimeError("IDW interpolation requires at least one neighbor.")
    d_min = float(np.min(d_k))
    d_max = float(np.max(d_k))
    d_mean = float(np.mean(d_k))
    on_eps = max(float(on_node_epsilon), 1.0e-12)
    exact_hits = np.where(d_k <= on_eps)[0]
    exact_local = int(exact_hits[0]) if len(exact_hits) > 0 else None
    diagnostics = {
        "idw_neighbors_used": int(d_k.size),
        "idw_exact_node_hit": exact_local is not None,
        "idw_support_radius_m": d_max,
        "idw_support_mean_distance_m": d_mean,
        "mesh_distance": d_min,
    }
    if exact_local is not None:
        return None, exact_local, diagnostics
    eps = float(epsilon)
    weights = 1.0 / (d_k + eps) ** 2
    return weights, None, diagnostics

def select_nearest_indices(distances: np.ndarray, k: int) -> np.ndarray:
    """Brief: Select the ``k`` nearest mesh-neighbor indices for IDW support.

    Inputs:
        distances: Full set of probe-to-node distances [m].
        k: Requested neighbor count (clamped to available nodes).

    Outputs:
        np.ndarray: Local indices of the ``k`` nearest neighbors (unordered partition).
    """
    if k <= 0:
        raise RuntimeError("IDW interpolation requires at least one neighbor.")
    k = min(int(k), len(distances))
    return np.argpartition(distances, k - 1)[:k]

def idw_interpolate_vectors(
    values: np.ndarray,
    distances: np.ndarray,
    *,
    k: int,
    epsilon: float,
    on_node_epsilon: float,
) -> tuple[np.ndarray, dict[str, Any]]:
    """Brief: Inverse-distance-weight interpolate scalar or vector field values at a probe.

    Inputs:
        values: Per-node field samples (scalar or 3-vector components) [field units].
        distances: Probe-to-node distances matching ``values`` rows [m].
        k: Number of nearest neighbors for the IDW stencil.
        epsilon: Softening length for IDW weights [m].
        on_node_epsilon: Exact-hit distance threshold [m].

    Outputs:
        tuple: ``(interpolated_value, diagnostics)`` with IDW support metadata.
    """
    nn_idx = select_nearest_indices(distances, k)
    d_k = distances[nn_idx]
    weights, exact_local, diagnostics = idw_weights(
        d_k, epsilon=epsilon, on_node_epsilon=on_node_epsilon
    )
    if exact_local is not None:
        return np.asarray(values[nn_idx[exact_local]], dtype=float), diagnostics
    assert weights is not None
    w = weights / float(np.sum(weights))
    if values.ndim == 1:
        return np.asarray(np.sum(values[nn_idx] * w), dtype=float), diagnostics
    return np.asarray(np.sum(values[nn_idx] * w[:, None], axis=0), dtype=float), diagnostics
