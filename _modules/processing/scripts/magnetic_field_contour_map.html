def render_magnetic_contour_png(
    PHI: np.ndarray,
    RHO: np.ndarray,
    Bmag: np.ndarray,
    rho_in: float,
    rho_out: float,
    probes: List[Dict[str, Any]],
    out_path: Path,
    *,
    title: Optional[str] = None,
    bmax: Optional[float] = None,
    levels: Optional[np.ndarray] = None,
) -> None:
    """Brief: Render a P-R filled contour of |B| with bext probe markers on the trusted band.

    Inputs:
        PHI: Azimuth grid [rad] matching ``Bmag`` shape.
        RHO: Radial grid [m] on the midplane trusted band.
        Bmag: |B| samples on the P-R grid [T].
        rho_in: Trusted-band inner radius [m].
        rho_out: Trusted-band outer radius [m].
        probes: Probe dicts with ``phi``, ``rho``, optional ``kind``/``label``.
        out_path: Output PNG path.
        title: Optional plot title.
        bmax: Optional color-scale max [T] when ``levels`` omitted.
        levels: Optional shared contour level edges [T].

    Outputs:
        None: Writes the PNG via Matplotlib Agg; raises if no finite samples / no matplotlib.
    """
    try:
        import matplotlib

        matplotlib.use("Agg")
        import matplotlib.pyplot as plt
        from matplotlib.colors import BoundaryNorm
        from matplotlib.ticker import FuncFormatter
    except ImportError as exc:
        raise RuntimeError(
            "matplotlib is required to write magnetic_field_contour_map.png "
            "(grid CSV was still written)."
        ) from exc

    finite = Bmag[np.isfinite(Bmag)]
    if finite.size == 0:
        raise RuntimeError("No finite |B| samples on the trusted-band grid.")

    if levels is None:
        if bmax is None:
            bmax = float(np.nanpercentile(finite, 99.5))
            if not np.isfinite(bmax) or bmax <= 0.0:
                bmax = float(np.nanmax(finite))
            if not np.isfinite(bmax) or bmax <= 0.0:
                bmax = 1.0
        n_levels = 10
        levels = np.linspace(0.0, float(bmax), n_levels + 1)
    else:
        levels = np.asarray(levels, dtype=float)
        n_levels = max(int(levels.size) - 1, 1)
        bmax = float(levels[-1])

    cmap = plt.get_cmap("jet", n_levels)
    norm = BoundaryNorm(levels, ncolors=cmap.N, clip=True)

    PHI_deg = np.degrees(PHI)

    fig, ax = plt.subplots(figsize=(8.0, 5.5), constrained_layout=True)
    cf = ax.contourf(PHI_deg, RHO, Bmag, levels=levels, cmap=cmap, norm=norm)
    cs = ax.contour(PHI_deg, RHO, Bmag, levels=levels[1:-1], colors="k", linewidths=0.6)
    ax.clabel(cs, inline=True, fontsize=8, fmt=lambda v: _fmt_B_sci(v))

    cbar = fig.colorbar(cf, ax=ax, orientation="horizontal", pad=0.08, aspect=36, shrink=0.92)
    cbar.set_label("|B| (T)")
    tick_idx = list(range(0, n_levels + 1, 2))
    if tick_idx[-1] != n_levels:
        tick_idx.append(n_levels)
    tick_vals = [float(levels[i]) for i in tick_idx]
    cbar.set_ticks(tick_vals)
    cbar.set_ticklabels([_fmt_B_sci(v) for v in tick_vals])
    cbar.ax.tick_params(labelsize=8)

    try:
        from matplotlib import patheffects as pe
    except ImportError:
        pe = None

    bext_probes = [
        p
        for p in probes
        if str(p.get("kind", "bext")).lower() == "bext"
        and float(rho_in) - 1.0e-12 <= float(p["rho"]) <= float(rho_out) + 1.0e-12
    ]
    for i, p in enumerate(bext_probes):
        phi_deg = math.degrees(float(p["phi"]))
        rho = float(p["rho"])
        ax.scatter(
            [phi_deg],
            [rho],
            c="white",
            marker="o",
            s=42,
            edgecolors="k",
            linewidths=0.7,
            zorder=5,
        )
        label = str(p.get("label") or f"P{i + 1}")
        ann = ax.annotate(
            label,
            xy=(phi_deg, rho),
            xytext=(6, 6),
            textcoords="offset points",
            fontsize=9,
            fontweight="bold",
            color="white",
            zorder=6,
        )
        if pe is not None:
            ann.set_path_effects([pe.withStroke(linewidth=2.5, foreground="black")])

    ax.set_xlabel("\u03c6 (deg)")
    ax.set_ylabel(r"$\rho$ (m)")
    ax.set_xlim(0.0, 360.0)
    ax.set_yscale("linear")
    ax.xaxis.set_major_formatter(FuncFormatter(lambda value, _pos: truncate_scientific(value)))
    ax.yaxis.set_major_formatter(FuncFormatter(lambda value, _pos: truncate_scientific(value)))
    y_lo = max(float(rho_in), 1.0e-12)
    y_hi = max(float(rho_out), y_lo * 1.01)
    ax.set_ylim(y_lo, y_hi)
    ax.set_title(title or "Magnetic flux density |B|")
    fig.savefig(out_path, dpi=160)
    plt.close(fig)

def _probe_xyz(probe: Dict[str, Any]) -> Tuple[float, float, float]:
    """Brief: Convert midplane probe ``P``/``R`` to Cartesian (x=0, y, z).

    Inputs:
        probe: Dict with ``P`` [rad] and ``R`` [m].

    Outputs:
        tuple: ``(x, y, z)`` with ``x=0`` on the midplane [m].
    """
    phi = float(probe["phi"])
    rho = float(probe["rho"])
    return 0.0, rho * math.cos(phi), rho * math.sin(phi)

def analyze_probe_stability(
    timeseries: Sequence[Dict[str, Any]],
) -> Dict[str, Any]:
    """Brief: Detect B-component sign flips and abs(B) decade jumps across time for each probe label.

    Inputs:
        timeseries: Rows with ``time``, ``label``, ``x,y,z``, ``Bx,By,Bz``, ``B_mag_T``.

    Outputs:
        dict: ``label - {coords, sign_change, sign_events, magnitude_jump, magnitude_events}``.
    """
    by_label: Dict[str, List[Dict[str, Any]]] = {}
    for row in timeseries:
        label = str(row.get("label") or "P?")
        by_label.setdefault(label, []).append(row)

    reports: Dict[str, Any] = {}
    for label, rows in by_label.items():
        rows = sorted(rows, key=lambda r: float(r["time"]))
        coords = (
            float(rows[0]["x"]),
            float(rows[0]["y"]),
            float(rows[0]["z"]),
        )
        sign_events: List[Dict[str, Any]] = []
        mag_events: List[Dict[str, Any]] = []
        for i in range(len(rows) - 1):
            a, b = rows[i], rows[i + 1]
            t0, t1 = float(a["time"]), float(b["time"])
            for comp in ("Bx", "By", "Bz"):
                v0, v1 = float(a[comp]), float(b[comp])
                if not (math.isfinite(v0) and math.isfinite(v1)):
                    continue
                if abs(v0) <= _SIGN_EPS or abs(v1) <= _SIGN_EPS:
                    continue
                if v0 * v1 < 0.0:
                    sign_events.append(
                        {
                            "time": t1,
                            "time_i": t0,
                            "time_ip1": t1,
                            "component": comp,
                            "B_i": v0,
                            "B_ip1": v1,
                        }
                    )
            m0, m1 = float(a["B_mag_T"]), float(b["B_mag_T"])
            if not (math.isfinite(m0) and math.isfinite(m1)):
                continue
            a0, a1 = abs(m0), abs(m1)

            if a0 <= _SIGN_EPS or a1 <= _SIGN_EPS:
                continue

            magnitude_ratio = max(a0, a1) / min(a0, a1)

            if magnitude_ratio >= 10.0:
                mag_events.append(
                    {
                        "time": t1,
                        "time_i": t0,
                        "time_ip1": t1,
                        "Bmag_i": m0,
                        "Bmag_ip1": m1,
                    }
                )
        reports[label] = {
            "coords": coords,
            "sign_change": bool(sign_events),
            "sign_events": sign_events,
            "magnitude_jump": bool(mag_events),
            "magnitude_events": mag_events,
        }
    return reports

def write_probe_stability_txt(
    reports: Dict[str, Any],
    out_path: Path,
) -> None:
    """Brief: Write human-readable probe stability report (sign flips and |B| decade jumps).

    Inputs:
        reports: Output of ``analyze_probe_stability``.
        out_path: Destination text path (e.g. ``probe_stability.txt``).

    Outputs:
        None: Writes UTF-8 text file at ``out_path``.
    """
    lines: List[str] = [
        "Magnetic field contour probe stability (time_dependent)",
        "Magnitude jump criterion: |B| changes by >= 1 order of magnitude in one step",
        "Sign-change criterion: any of Bx, By, Bz flips sign between consecutive steps",
        "",
    ]
    for label in sorted(reports.keys(), key=lambda s: (len(s), s)):
        rep = reports[label]
        x, y, z = rep["coords"]
        lines.append(f"{label} (x, y, z) = ({x:.6g}, {y:.6g}, {z:.6g})")
        lines.append(f"  sign_change: {'Y' if rep['sign_change'] else 'N'}")
        for ev in rep["sign_events"]:
            lines.append(
                "    t={time} (steps {t0} -> {t1})  "
                "{comp}: B_i={bi:.6e}  B_i+1={bip:.6e}".format(
                    time=_fmt_time_int_s(ev["time"]),
                    t0=_fmt_time_int_s(ev["time_i"]),
                    t1=_fmt_time_int_s(ev["time_ip1"]),
                    comp=ev["component"],
                    bi=ev["B_i"],
                    bip=ev["B_ip1"],
                )
            )
        lines.append(
            f"  intense_magnitude_change: "
            f"{'Y' if rep['magnitude_jump'] else 'N'}"
        )
        for ev in rep["magnitude_events"]:
            lines.append(
                "    t={time} (steps {t0} -> {t1})  "
                "|B|_i={bi:.6e}  |B|_i+1={bip:.6e}".format(
                    time=_fmt_time_int_s(ev["time"]),
                    t0=_fmt_time_int_s(ev["time_i"]),
                    t1=_fmt_time_int_s(ev["time_ip1"]),
                    bi=ev["Bmag_i"],
                    bip=ev["Bmag_ip1"],
                )
            )
        lines.append("")
    out_path.write_text("\n".join(lines).rstrip() + "\n", encoding="utf-8")

def _resolve_contour_period_s(results_dir: Path) -> float:
    """Brief: Read driver period P from ``project.json`` for time-dependent contour snapshots.

    Inputs:
        results_dir: ``results/`` or run directory containing ``project.json`` [path].

    Outputs:
        float: Positive ``driver_period`` [s].
    """
    run_dir = results_dir.parent if results_dir.name == "results" else results_dir
    for candidate in (run_dir / "project.json", results_dir / "project.json"):
        if not candidate.is_file():
            continue
        try:
            data = json.loads(candidate.read_text(encoding="utf-8"))
            sim = data.get("simulation") or {}
            td = sim.get("time_dependent_params") or {}
            period = td.get("driver_period")
            if period is not None and float(period) > 0.0:
                return float(period)
        except (OSError, TypeError, ValueError, json.JSONDecodeError):
            continue
    raise ValueError(
        "time_dependent contour maps require simulation.time_dependent_params.driver_period "
        "in project.json"
    )

def _contour_period_frame_indices(
    times: Sequence[float], period_s: float
) -> List[Tuple[int, float, str]]:
    """Brief: Pick VTU frame indices nearest to 0, P/4, P/2, 3P/4 within simulated span.

    Inputs:
        times: Physical times per frame [s].
        period_s: Driver period P [s].

    Outputs:
        list: ``(index, time, tag)`` with tags ``0``, ``P4``, ``P2``, ``3P4``.
    """
    if not times:
        return []
    period = max(float(period_s), 1.0e-30)
    t_vals = [float(t) for t in times]
    t_max = max(t_vals)
    if len(t_vals) >= 2:
        steps = sorted(t_vals)
        dt = min(steps[i + 1] - steps[i] for i in range(len(steps) - 1) if steps[i + 1] > steps[i])
        slack = 0.5 * dt if dt > 0.0 else 0.0
    else:
        slack = 0.0

    picks: List[Tuple[int, float, str]] = []
    for frac, tag in ((0.0, "0"), (0.25, "P4"), (0.5, "P2"), (0.75, "3P4")):
        t_target = frac * period
        if t_target > t_max + slack:
            continue
        idx = min(range(len(t_vals)), key=lambda i: abs(t_vals[i] - t_target))
        picks.append((idx, t_vals[idx], tag))
    return picks

def write_contour_period_pngs(
    times: Sequence[float],
    frames_B: Sequence[np.ndarray],
    PHI: np.ndarray,
    RHO: np.ndarray,
    rho_in: float,
    rho_out: float,
    probes: List[Dict[str, Any]],
    out_dir: Path,
    *,
    period_s: Optional[float] = None,
) -> List[Path]:
    """Brief: Write shared-level |B| contour PNGs at period phase tags; prune uncovered tags.

    Inputs:
        times: Frame times [s].
        frames_B: |B| grids per time [T].
        PHI: Azimuth meshgrid [rad].
        RHO: Radius meshgrid [m].
        rho_in: Trusted-band inner radius [m].
        rho_out: Trusted-band outer radius [m].
        probes: Overlay probe dicts for markers.
        out_dir: Directory for ``magnetic_field_contour_map_t*.png``.
        period_s: Optional period override [s]; else from project.json.

    Outputs:
        list: Paths of PNGs written (empty if steady / no matplotlib / no coverage).
    """
    if len(times) < 2 or len(frames_B) != len(times):
        return []
    if not _matplotlib_available():
        print("[INFO] Contour period PNGs deferred (matplotlib unavailable).")
        return []
    period = float(period_s) if period_s is not None else _resolve_contour_period_s(out_dir)
    levels = _global_contour_levels(frames_B)
    picks = _contour_period_frame_indices(times, period)
    all_tags = ("0", "P4", "P2", "3P4")
    keep_tags = {tag for _, _, tag in picks}
    for tag in all_tags:
        if tag not in keep_tags:
            (out_dir / f"magnetic_field_contour_map_t{tag}.png").unlink(missing_ok=True)
    if not picks:
        print(
            f"[INFO] Contour period snapshots skipped: simulated span "
            f"[0, {max(float(t) for t in times):.6g}] s does not cover any of "
            f"0, P/4, P/2, 3P/4 for P={period:.6g} s"
        )
        return []
    tags = ", ".join(tag for _, _, tag in picks)
    print(
        f"[INFO] Contour period snapshots for P={period:.6g} s "
        f"(covered tags: {tags}; skipped uncovered phase PNGs)"
    )
    written: List[Path] = []
    for idx, t, tag in picks:
        out_path = out_dir / f"magnetic_field_contour_map_t{tag}.png"
        render_magnetic_contour_png(
            PHI,
            RHO,
            frames_B[idx],
            rho_in,
            rho_out,
            probes,
            out_path,
            title=f"Magnetic flux density |B|  t={_fmt_time_int_s(t)} (P={period:g} s)",
            levels=levels,
        )
        written.append(out_path)
        print(f"[OK] Wrote {out_path.name}")
    return written

def _global_contour_levels(B_stack: Sequence[np.ndarray], n_levels: int = 10) -> np.ndarray:
    """Brief: Build shared contour levels from stacked |B| frames using a 99.5th percentile max.

    Inputs:
        B_stack: Sequence of |B| grids [T].
        n_levels: Number of contour intervals.

    Outputs:
        np.ndarray: Level edges from 0 to ``bmax`` inclusive [T].
    """
    finite = np.concatenate([b[np.isfinite(b)].ravel() for b in B_stack if b.size])
    if finite.size == 0:
        return np.linspace(0.0, 1.0, n_levels + 1)
    bmax = float(np.nanpercentile(finite, 99.5))
    if not np.isfinite(bmax) or bmax <= 0.0:
        bmax = float(np.nanmax(finite))
    if not np.isfinite(bmax) or bmax <= 0.0:
        bmax = 1.0
    return np.linspace(0.0, bmax, n_levels + 1)

def _stability_rows_from_magnetic_field_csv(results: Path) -> List[Dict[str, Any]]:
    """Brief: Convert ``magnetic_field_data.csv`` rows into stability-analysis timeseries dicts.

    Inputs:
        results: Results directory path [path].

    Outputs:
        list: Rows with ``time``, ``label``, coords, ``Bx,By,Bz``, ``B_mag_T``.
    """
    path = results / "magnetic_field_data.csv"
    if not path.is_file():
        return []
    rows_out: List[Dict[str, Any]] = []
    for r in csv.DictReader(path.open(encoding="utf-8")):
        try:
            pid = int(float(r["point_id"]))
            bx = float(r["Bx"])
            by = float(r["By"])
            bz = float(r["Bz"])
            mag_raw = r.get("B_mag")
            if mag_raw not in (None, ""):
                bmag = float(mag_raw)
            else:
                bmag = math.sqrt(bx * bx + by * by + bz * bz)
            rows_out.append(
                {
                    "time": float(r["time"]),
                    "label": f"P{pid}",
                    "x": float(r.get("x", r.get("actual_x"))),
                    "y": float(r.get("y", r.get("actual_y"))),
                    "z": float(r.get("z", r.get("actual_z"))),
                    "Bx": bx,
                    "By": by,
                    "Bz": bz,
                    "B_mag_T": bmag,
                }
            )
        except (KeyError, TypeError, ValueError):
            continue
    return rows_out

def _write_stability_from_available_csvs(results: Path) -> Optional[Path]:
    """Brief: Build ``probe_stability.txt`` from contour timeseries or exterior B CSV if multi-time.

    Inputs:
        results: Results directory [path].

    Outputs:
        Path | None: Written stability path, or ``None`` if insufficient time samples.
    """
    ts_path = results / "magnetic_field_contour_map_probe_timeseries.csv"
    if ts_path.is_file():
        ts_rows = list(csv.DictReader(ts_path.open(encoding="utf-8")))
    else:
        ts_rows = _stability_rows_from_magnetic_field_csv(results)
    if not ts_rows:
        return None
    times = {row.get("time") for row in ts_rows}
    if len(times) < 2:
        return None
    reports = analyze_probe_stability(ts_rows)
    stab_path = results / "probe_stability.txt"
    write_probe_stability_txt(reports, stab_path)
    print(f"[OK] Wrote {stab_path.name}")
    return stab_path

def plot_contour_map_from_csv(results_dir: str) -> None:
    """Brief: Rebuild P-R |B| contour PNGs (and stability) from ``magnetic_field_contour_map_data.csv``.

    Inputs:
        results_dir: Directory with contour grid CSV and optional probe CSVs [path].

    Outputs:
        None: Writes period or steady contour PNGs when matplotlib is available.
    """
    results = Path(results_dir)


    _write_stability_from_available_csvs(results)

    data_path = results / "magnetic_field_contour_map_data.csv"
    if not data_path.is_file():
        print(
            f"[INFO] No {data_path.name}; skipping contour PNGs "
            "(run Magnetic Field Contour Map once to sample the midplane grid)."
        )
        return

    rows = list(csv.DictReader(data_path.open(encoding="utf-8")))
    if not rows:
        print("Skipping contour map plot: empty CSV")
        return

    has_time = "time" in rows[0]
    rho_in = float(rows[0]["rho_in_m"])
    rho_out = float(rows[0]["rho_out_m"])

    if has_time:
        times = sorted({float(r["time"]) for r in rows})
    else:
        times = [0.0]
        for r in rows:
            r["time"] = "0"

    phis = sorted({float(r["phi_rad"]) for r in rows})
    rhos = sorted({float(r["rho_m"]) for r in rows})
    PHI, RHO = np.meshgrid(np.array(phis), np.array(rhos), indexing="xy")

    frames: List[np.ndarray] = []
    by_time: Dict[float, List[Dict[str, str]]] = {t: [] for t in times}
    for r in rows:
        tr = float(r["time"])
        best = min(times, key=lambda t: abs(t - tr))
        if abs(best - tr) <= max(1.0e-12, 1.0e-9 * max(abs(best), abs(tr), 1.0)):
            by_time[best].append(r)
    for t in times:
        lookup = {
            (float(r["phi_rad"]), float(r["rho_m"])): float(r["B_mag_T"])
            for r in by_time[t]
        }
        Bmag = np.empty(PHI.shape, dtype=float)
        for j, rho in enumerate(rhos):
            for i, phi in enumerate(phis):
                Bmag[j, i] = lookup.get((phi, rho), float("nan"))
        frames.append(Bmag)

    probes = _load_bext_probes_for_plot(results, rho_in, rho_out)
    levels = _global_contour_levels(frames)
    is_time_dependent = len(times) >= 2

    if not _matplotlib_available():
        print("[INFO] Contour PNG skipped (matplotlib unavailable).")
        return

    if is_time_dependent:
        write_contour_period_pngs(
            times, frames, PHI, RHO, rho_in, rho_out, probes, results
        )
        return

    render_magnetic_contour_png(
        PHI,
        RHO,
        frames[-1],
        rho_in,
        rho_out,

def _probes_from_xyz(
    points: List[Any],
    rho_in: float,
    rho_out: float,
    *,
    require_in_band: bool = False,
) -> List[Dict[str, Any]]:
    """Brief: Convert Cartesian probe points to midplane P-R overlays, optionally band-clipped.

    Inputs:
        points: List of dicts/tuples with xyz [m].
        rho_in: Trusted-band inner radius [m].
        rho_out: Trusted-band outer radius [m].
        require_in_band: Drop probes outside [rho_in, rho_out].

    Outputs:
        list: Probe dicts with ``phi``, ``rho``, ``kind=bext``, ``label``, xyz.
    """
    probes: List[Dict[str, Any]] = []
    for idx, p in enumerate(points):
        try:
            if isinstance(p, dict):
                x, y, z = float(p["x"]), float(p["y"]), float(p["z"])
            else:
                x, y, z = float(p[0]), float(p[1]), float(p[2])
        except (KeyError, TypeError, ValueError, IndexError):
            continue
        rho = math.hypot(y, z)
        if rho < 1.0e-15:
            continue
        if abs(x) > 0.05 * max(rho, rho_out, 1.0e-6):
            continue
        if require_in_band and not (rho_in - 1.0e-12 <= rho <= rho_out + 1.0e-12):
            continue
        probes.append(
            {
                "phi": math.atan2(z, y) % (2.0 * math.pi),
                "rho": rho,
                "kind": "bext",
                "label": f"P{idx + 1}",
                "x": x,
                "y": y,
                "z": z,
            }
        )
    return probes

def _read_bext_from_processing_config(results: Path) -> List[Any]:
    """Brief: Read bext probe xyz lists from ``processing_config.json`` (global or script params).

    Inputs:
        results: Results directory containing the config [path].

    Outputs:
        list: Raw probe point entries (possibly empty).
    """
    cfg_path = results / "processing_config.json"
    if not cfg_path.is_file():
        return []
    try:
        data = json.loads(cfg_path.read_text(encoding="utf-8"))
        bext_raw = list(data.get("probe_points", {}).get("bext") or [])
        if bext_raw:
            return bext_raw
        for s in data.get("scripts", []) or []:
            if not isinstance(s, dict):
                continue
            if s.get("name") in ("magnetic_field_exterior", "magnetic_field_contour_map"):
                pts = list((s.get("parameters") or {}).get("probe_points") or [])
                if pts:
                    return pts
    except (OSError, json.JSONDecodeError, TypeError, ValueError):
        return []
    return []

def _default_bext_xyz_for_contour(
    results: Path,
    rho_in: float,
    rho_out: float,
) -> List[Any]:
    """Brief: Compute default bext probe xyz from geometry.geo and trusted-band override.

    Inputs:
        results: Results or run directory used to locate ``geometry.geo`` [path].
        rho_in: Trusted-band inner radius [m].
        rho_out: Trusted-band outer radius [m].

    Outputs:
        list: Dict probes ``{x,y,z}`` from ``compute_bext_probe_positions``.
    """
    run_dir = results.parent if results.name == "results" else results
    from probe_scaling import compute_bext_probe_positions
    from geometry_contract import contract_from_text, metal_sphere_radius_m, parse_geo_assignments

    geo_text = ""
    for candidate in (run_dir / "geometry.geo", run_dir / "geometry" / "geometry.geo"):
        if candidate.is_file():
            geo_text = candidate.read_text(encoding="utf-8")
            break
    if not geo_text:
        raise FileNotFoundError(f"No geometry.geo under {run_dir} for contour probe defaults")

    vals = parse_geo_assignments(geo_text)
    contract = contract_from_text(geo_text)
    gtype = contract.family if contract.family != "unknown" else "cylinder"
    variant = contract.variant
    if gtype == "sphere":
        radius = metal_sphere_radius_m(vals)
        if radius is None:
            raise KeyError("Sphere contour probes require R_sphere")
        params = {"R_sphere": float(radius), "R_cap": float(vals["R_cap"])}
    elif gtype == "prism":
        params = {
            "L": float(vals["L"]),
            "W": float(vals["W"]),
            "H": float(vals["H"]),
        }
    else:
        params = {"L": float(vals["L"]), "R": float(vals["R"])}

    return [
        {"x": x, "y": y, "z": z}
        for x, y, z in compute_bext_probe_positions(
            params,
            gtype,
            variant=variant,
            band_override={"rho_in_m": float(rho_in), "rho_out_m": float(rho_out)},
        )
    ]

        def _resolve_trusted_band(self) -> Tuple[float, float]:
            """Brief: Resolve (rho_in, rho_out) from geo trusted radii or ``compute_magnetic_trusted_band``.

            Inputs:
                self: Contour map instance.

            Outputs:
                tuple: ``(rho_in, rho_out)`` trusted exterior band radii [m].
            """
            geo_text = self._geo_text()
            vals = parse_geo_assignments(geo_text) if geo_text else {}
            if "rho_trusted_inf" in vals and "rho_trusted_sup" in vals:
                rho_in = float(vals["rho_trusted_inf"])
                rho_out = float(vals["rho_trusted_sup"])
                if rho_out > rho_in > 0.0:
                    return rho_in, rho_out
            if "R_sample_good" in vals:
                r_body = vals.get("R_sphere")
                d_near = vals.get("d_near_max")
                if r_body is not None and d_near is not None:
                    rho_in = float(r_body) + float(d_near)
                    rho_out = float(vals["R_sample_good"])
                    if rho_out > rho_in > 0.0:
                        return rho_in, rho_out

            params = self.geometry.probe_scaling_params()
            band = compute_magnetic_trusted_band(
                params,
                str(self.config.geometry_type),
                variant=str(getattr(self.geometry_contract, "variant", "full") or "full"),
            )
            return float(band["rho_in_m"]), float(band["rho_out_m"])

        def _prepare_air_b_cache(self, vtu_file: str) -> dict:
            """Brief: Threshold outer-air VTU cells and cache point coords + B vectors for IDW.

            Inputs:
                self: Contour map instance.
                vtu_file: VTU path for this timestep [path].

            Outputs:
                dict: Cache with ``points_np``, ``B_np``, and ParaView ``objects``.
            """
            if self._cache is not None and self._cache.get("vtu_file") == vtu_file:
                return self._cache

            reader = self.make_reader("magnetic_contour_vtu", vtu_file)
            th = Threshold(registrationName="Threshold_B_contour_air", Input=reader)
            th.Scalars = ["CELLS", "GeometryIds"]
            air_id = int(self.config.outer_air_id)
            self.set_threshold_between(th, float(air_id), float(air_id))
            merged = MergeBlocks(th)
            vtk_data = servermanager.Fetch(merged)
            points = vtk_data.GetPoints()
            points_np = (
                ns.vtk_to_numpy(points.GetData()) if points is not None else np.empty((0, 3), dtype=float)
            )
            B_array = vtk_data.GetPointData().GetArray("magnetic flux density")
            if B_array is None:
                self.cleanup(merged, th, reader)
                raise RuntimeError("Missing 'magnetic flux density' point array.")
            self._cache = {
                "vtu_file": vtu_file,
                "objects": [merged, th, reader],
                "points_np": points_np,
                "B_np": ns.vtk_to_numpy(B_array),
            }
            return self._cache

        def _idw_B_vec(self, cache: dict, x: float, y: float, z: float) -> Tuple[float, float, float]:
            """Brief: IDW-interpolate magnetic flux density vector at (x,y,z) from the air cache.

            Inputs:
                self: Contour map instance.
                cache: Prepared air B cache dict.
                x: Sample x [m].
                y: Sample y [m].
                z: Sample z [m].

            Outputs:
                tuple: ``(Bx, By, Bz)`` [T], or NaNs if no mesh points / empty stencil.
            """
            points_np = cache["points_np"]
            B_np = cache["B_np"]
            if points_np.size == 0:
                return float("nan"), float("nan"), float("nan")
            target = np.array([x, y, z], dtype=float)
            distances = np.linalg.norm(points_np - target, axis=1)
            k = min(int(self.ext_cfg["idw_neighbors"]), len(points_np))
            if k <= 0:
                return float("nan"), float("nan"), float("nan")
            nn_idx = np.argpartition(distances, k - 1)[:k]
            d_k = distances[nn_idx]
            on_eps = max(float(self.ext_cfg["on_node_epsilon"]), 1.0e-12)
            eps = float(self.ext_cfg["idw_epsilon"])
            if float(np.min(d_k)) <= on_eps:
                Bvec = B_np[int(nn_idx[int(np.argmin(d_k))])]
            else:
                w = 1.0 / (d_k + eps) ** 2
                w = w / float(np.sum(w))
                Bvec = np.sum(B_np[nn_idx] * w[:, None], axis=0)
            return float(Bvec[0]), float(Bvec[1]), float(Bvec[2])

        def _idw_B_mag(self, cache: dict, x: float, y: float, z: float) -> float:
            """Brief: IDW-interpolate |B| at (x,y,z) from the air cache.

            Inputs:
                self: Contour map instance.
                cache: Prepared air B cache dict.
                x: Sample x [m].
                y: Sample y [m].
                z: Sample z [m].

            Outputs:
                float: |B| [T], or NaN if vector components are non-finite.
            """
            bx, by, bz = self._idw_B_vec(cache, x, y, z)
            if not all(math.isfinite(v) for v in (bx, by, bz)):
                return float("nan")
            return float(math.sqrt(bx * bx + by * by + bz * bz))

        def _sample_grid(
            self, cache: dict, rho_in: float, rho_out: float
        ) -> Tuple[np.ndarray, np.ndarray, np.ndarray]:
            """Brief: IDW-sample |B| on the N_PHI x N_RHO midplane P-R grid within the trusted band.

            Inputs:
                self: Contour map instance.
                cache: Prepared air B cache.
                rho_in: Band inner radius [m].
                rho_out: Band outer radius [m].

            Outputs:
                tuple: ``(PHI, RHO, Bmag)`` meshgrids [rad], [m], [T].
            """
            phi = np.linspace(0.0, 2.0 * math.pi, self.N_PHI, endpoint=False)
            rho = np.linspace(rho_in, rho_out, self.N_RHO)
            PHI, RHO = np.meshgrid(phi, rho, indexing="xy")
            Bmag = np.empty(PHI.shape, dtype=float)
            for j in range(self.N_RHO):
                for i in range(self.N_PHI):
                    r = float(RHO[j, i])
                    p = float(PHI[j, i])
                    y = r * math.cos(p)
                    z = r * math.sin(p)
                    Bmag[j, i] = self._idw_B_mag(cache, 0.0, y, z)
            return PHI, RHO, Bmag

        def _sample_probes(
            self, cache: dict, probes: List[Dict[str, Any]], time_val: float
        ) -> List[Dict[str, Any]]:
            """Brief: IDW-sample Bx,By,Bz,|B| at overlay probes for one physical time.

            Inputs:
                self: Contour map instance.
                cache: Prepared air B cache.
                probes: Overlay probe dicts (xyz or P-R).
                time_val: Physical time for the rows [s].

            Outputs:
                list: Timeseries row dicts for stability/CSV export.
            """
            rows: List[Dict[str, Any]] = []
            for i, p in enumerate(probes):
                label = str(p.get("label") or f"P{i + 1}")
                if "x" in p and "y" in p and "z" in p:
                    x, y, z = float(p["x"]), float(p["y"]), float(p["z"])
                else:
                    x, y, z = _probe_xyz(p)
                bx, by, bz = self._idw_B_vec(cache, x, y, z)
                bmag = (
                    float(math.sqrt(bx * bx + by * by + bz * bz))
                    if all(math.isfinite(v) for v in (bx, by, bz))
                    else float("nan")
                )
                rows.append(
                    {
                        "time": time_val,
                        "label": label,
                        "x": x,
                        "y": y,
                        "z": z,
                        "Bx": bx,
                        "By": by,
                        "Bz": bz,
                        "B_mag_T": bmag,
                    }
                )
            return rows

        def run(self) -> None:
            """Brief: Sample all VTU steps, write grid/probe CSVs, stability report, and contour PNGs.

            Inputs:
                self: Contour map instance with VTU inputs configured.

            Outputs:
                None: Side-effect files under ``out_dir``; defers PNG if matplotlib missing.
            """
            rho_in, rho_out = self._resolve_trusted_band()
            if self.verbose:
                print(f"Trusted band \u03c1 \u2208 [{rho_in:.6g}, {rho_out:.6g}] m")

            probes = self._load_overlay_probes(rho_in, rho_out)
            times: List[float] = []
            frames: List[np.ndarray] = []
            probe_rows: List[Dict[str, Any]] = []
            PHI = RHO = None

            for step_idx, vtu in enumerate(self.vtu_files):
                t = float(self.physical_time_for_vtu_index(step_idx))
                cache = self._prepare_air_b_cache(vtu)
                PHI, RHO, Bmag = self._sample_grid(cache, rho_in, rho_out)
                times.append(t)
                frames.append(Bmag)
                probe_rows.extend(self._sample_probes(cache, probes, t))
                if self._cache is not None:
                    self.cleanup(*self._cache.get("objects", []))
                    self._cache = None

            assert PHI is not None and RHO is not None and frames

            out = Path(self.out_dir)
            csv_path = out / "magnetic_field_contour_map_data.csv"
            self._write_grid_csv(times, frames, PHI, RHO, rho_in, rho_out, csv_path)
            print(f"[OK] Wrote {csv_path.name}")

            if probe_rows:
                if len(times) >= 2:
                    ts_path = out / "magnetic_field_contour_map_probe_timeseries.csv"
                    self._write_probe_timeseries(probe_rows, ts_path)
                    print(f"[OK] Wrote {ts_path.name}")
                    reports = analyze_probe_stability(probe_rows)
                    stab_path = out / "probe_stability.txt"
                    write_probe_stability_txt(reports, stab_path)
                    print(f"[OK] Wrote {stab_path.name}")


            if not _matplotlib_available():
                print(
                    "[INFO] Contour PNG deferred to generate_plots.py "
                    "(matplotlib not available in this ParaView/pvpython)."
                )
                return

            if len(times) >= 2:
                write_contour_period_pngs(
                    times, frames, PHI, RHO, rho_in, rho_out, probes, out
                )
                return

            levels = _global_contour_levels(frames)
            try:
                render_magnetic_contour_png(
                    PHI,
                    RHO,
                    frames[-1],
                    rho_in,
                    rho_out,
                    probes,
                    out / "magnetic_field_contour_map.png",
                    title="Magnetic flux density |B|",
                    levels=levels,
                )
                print("[OK] Wrote magnetic_field_contour_map.png")
            except RuntimeError as exc:
                print(f"[WARN] Contour PNG not written: {exc}")
