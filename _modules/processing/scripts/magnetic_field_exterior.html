class MagneticFieldExterior(ProbeRegionMixin, BasePostProcessor):
    """Brief: Exterior B probes in outer air via IDW on VTU magnetic flux density.

    Inputs:
        working_dir: FEM run directory [path].
        verbose: Verbose logging flag.
        probe_points: Optional probes (max 3); else scaled bext defaults.

    Outputs:
        MagneticFieldExterior: Post-processor writing ``magnetic_field_data.csv``.
    """
    def __init__(
        self,
        working_dir=None,
        verbose=False,
        probe_points: list | None = None,
    ):
        """Brief: Initialize exterior B probe extractor, validate config, resolve probe points.

        Inputs:
            self: Processor instance.
            working_dir: FEM run directory [path].
            verbose: Verbose flag.
            probe_points: Optional list of xyz probes [m].

        Outputs:
            None: Sets IDW config, geometry contract, and validated probe list.
        """
        super().__init__(working_dir=working_dir, verbose=verbose)
        self.ext_cfg = self.pp_settings["extraction"]
        self.bext_cfg = self.ext_cfg["bext_defaults"]
        self._validate_required_config()
        self._cache_key: tuple | None = None
        self._cache: dict | None = None
        self.geometry_contract = load_geometry_contract(Path(self.cwd))

        self.probe_points: list[dict] = []
        if probe_points:
            for probe in probe_points:
                if isinstance(probe, (tuple, list)) and len(probe) >= 3:
                    self.probe_points.append(
                        {"x": float(probe[0]), "y": float(probe[1]), "z": float(probe[2])}
                    )
                elif isinstance(probe, dict):
                    normalized = normalize_probe_dict(probe)
                    if normalized is not None:
                        self.probe_points.append(normalized)

        if not self.probe_points:
            self.probe_points = self._default_bext_probe_points()
        if len(self.probe_points) > 3:
            self.probe_points = list(self.probe_points)[:3]
            print("[WARN] Truncated exterior probes to the first three user/automatic points")

    def __del__(self):
        """Brief: Destructor clearing the ParaView B-field cache.

        Inputs:
            self: Processor instance.

        Outputs:
            None: Best-effort ``_clear_cache``.
        """
        try:
            self._clear_cache()
        except Exception:
            pass

    def _clear_cache(self) -> None:
        """Brief: Destroy cached outer-air Threshold/MergeBlocks objects.

        Inputs:
            self: Processor instance.

        Outputs:
            None: Clears ``_cache`` and ``_cache_key``.
        """
        if not self._cache:
            return
        self.cleanup(*self._cache.get("objects", []))
        self._cache = None
        self._cache_key = None

    def _validate_required_config(self) -> None:
        """Brief: Require IDW extraction keys and bext ``point_directions`` in settings.

        Inputs:
            self: Processor with ``pp_settings["extraction"]``.

        Outputs:
            None: Raises ``KeyError`` if required keys are missing.
        """
        for key in ("idw_neighbors", "idw_epsilon", "on_node_epsilon"):
            if key not in self.ext_cfg:
                raise KeyError(f"Missing required extraction config key: {key}")
        if "point_directions" not in self.bext_cfg:
            raise KeyError("Missing extraction.bext_defaults.point_directions")

    def _default_bext_probe_points(self) -> list:
        """Brief: Compute default exterior probe xyz from geometry probe-scaling helpers.

        Inputs:
            self: Processor with geometry metadata.

        Outputs:
            list: Dict probes ``{x,y,z}`` [m].
        """
        gtype = str(self.geometry.geometry_type).lower()
        geo_params = self.geometry.probe_scaling_params()
        from probe_scaling import compute_bext_probe_positions

        return [
            {"x": x, "y": y, "z": z}
            for x, y, z in compute_bext_probe_positions(geo_params, gtype)
        ]

    def _validate_probe_point(self, x: float, y: float, z: float) -> tuple[float, float, float]:
        """Brief: Ensure probe lies in outer air (sphere: R_sphere < r < R_sph; else non-cavity air).

        Inputs:
            self: Processor with geometry contract.
            x: Probe x [m].
            y: Probe y [m].
            z: Probe z [m].

        Outputs:
            tuple: Validated ``(x, y, z)`` [m].
        """
        r3 = float(np.sqrt(x * x + y * y + z * z))
        if self.geometry_contract.family == "sphere":
            R = self.geometry_contract.body_outer_radius_m
            R_domain = self.geometry_contract.domain_radius_m
            if R is None or R_domain is None:
                raise RuntimeError("Sphere exterior validation requires R_sphere and R_sph.")
            if not (float(R) < r3 < float(R_domain)):
                raise ValueError(
                    f"Sphere exterior probe radius {r3:.9g} must satisfy "
                    f"R_sphere={float(R):.9g} < r < R_sph={float(R_domain):.9g}."
                )
            return x, y, z

        Rs = self.geometry.R_sph
        if Rs is not None and r3 >= float(Rs):
            raise ValueError(
                f"Exterior probe radius {r3:.9g} is outside the simulation domain {float(Rs):.9g}."
            )
        if not self._is_in_air_region(x, y, z) or self._is_inside_inner_cavity(x, y, z):
            raise ValueError(
                "MagneticFieldExterior probes must lie in outer_air, not conductor or inner cavity."
            )
        return x, y, z

    def _prepare_b_cache(self, vtu_file: str) -> dict:
        """Brief: Threshold outer-air cells and cache point coords + B vectors for IDW probes.

        Inputs:
            self: Processor instance.
            vtu_file: VTU path [path].

        Outputs:
            dict: Cache with ``points_np``, ``B_np``, and ParaView ``objects``.
        """
        cache_key = (vtu_file, True, (int(self.config.outer_air_id),))
        if self._cache_key == cache_key and self._cache is not None:
            return self._cache

        self._clear_cache()
        reader = self.make_reader("magnetic_ext_vtu", vtu_file)
        th = Threshold(registrationName="Threshold_B_probe", Input=reader)
        th.Scalars = ["CELLS", "GeometryIds"]
        air_id = int(self.config.outer_air_id)
        self.set_threshold_between(th, float(air_id), float(air_id))
        merged = MergeBlocks(th)
        vtk_data = servermanager.Fetch(merged)
        points = vtk_data.GetPoints()
        points_np = (
            ns.vtk_to_numpy(points.GetData()) if points is not None else np.empty((0, 3), dtype=float)
        )
        point_data = vtk_data.GetPointData()
        B_array = point_data.GetArray("magnetic flux density")
        if B_array is None:
            self.cleanup(merged, th, reader)
            raise RuntimeError("Missing 'magnetic flux density' point array.")
        self._cache = {
            "objects": [merged, th, reader],
            "points_np": points_np,
            "B_np": ns.vtk_to_numpy(B_array),
        }
        self._cache_key = cache_key
        return self._cache

    def _extract_at_point(self, x: float, y: float, z: float, vtu_file: str | None = None) -> dict:
        """Brief: IDW-interpolate Bx,By,Bz at an exterior probe and return nearest-node coords.

        Inputs:
            self: Processor instance.
            x: Probe x [m].
            y: Probe y [m].
            z: Probe z [m].
            vtu_file: Optional VTU path [path].

        Outputs:
            dict: ``Bx,By,Bz``, nearest ``x,y,z``, and IDW diagnostics.
        """
        if vtu_file is None:
            vtu_file = self.vtu_files[0]
        cache = self._prepare_b_cache(vtu_file)
        points_np = cache["points_np"]
        B_np = cache["B_np"]
        target_point = np.array([x, y, z], dtype=float)
        if points_np.size == 0:
            raise RuntimeError("No mesh points available in outer_air region for B-field probe.")
        distances = np.linalg.norm(points_np - target_point, axis=1)
        Bvec, diagnostics = idw_interpolate_vectors(
            B_np,
            distances,
            k=int(self.ext_cfg["idw_neighbors"]),
            epsilon=float(self.ext_cfg["idw_epsilon"]),
            on_node_epsilon=float(self.ext_cfg["on_node_epsilon"]),
        )
        Bx, By, Bz = float(Bvec[0]), float(Bvec[1]), float(Bvec[2])
        nearest_idx = int(np.argmin(distances))
        nearest = points_np[nearest_idx]
        return {
            "Bx": Bx,
            "By": By,
            "Bz": Bz,
            "x": float(nearest[0]),
            "y": float(nearest[1]),
            "z": float(nearest[2]),
            **diagnostics,
        }

    def run(self) -> None:
        """Brief: Sample all timesteps at exterior probes and write B CSV plus sampling log.

        Inputs:
            self: Processor with probe points and VTU inputs.

        Outputs:
            None: Writes ``magnetic_field_data.csv``, sampling log, and integrity evidence.
        """
        csv_path = os.path.join(self.out_dir, "magnetic_field_data.csv")
        sampling_log = Path(self.cwd) / "logs" / "magnetic_field_sampling.log"
        sampling_log.parent.mkdir(parents=True, exist_ok=True)
        times_written: list[float] = []

        with open(csv_path, "w", newline="", encoding="utf-8") as handle, open(
            sampling_log, "w", encoding="utf-8"
        ) as log_file:
            writer = csv.writer(handle)
            writer.writerow(
                [
                    "time",
                    "point_id",
                    "B_mag",
                    "Bx",
                    "By",
                    "Bz",
                    "x",
                    "y",
                    "z",
                ]
            )
            log_file.write(
                "time,point_id,mesh_distance_m,idw_neighbors_used,idw_exact_node_hit,"
                "idw_support_radius_m,idw_support_mean_distance_m\n"
            )

            if self.is_steady_state:
                step_files = [(0.0, None)]
            else:
                step_files = [
                    (self.physical_time_for_vtu_index(idx), path)
                    for idx, path in enumerate(self.vtu_files)
                ]

            for time_val, vtu_file in step_files:
                times_written.append(float(time_val))
                for i, pt in enumerate(self.probe_points):
                    tx, ty, tz = float(pt["x"]), float(pt["y"]), float(pt["z"])
                    vx, vy, vz = self._validate_probe_point(tx, ty, tz)
                    result = self._extract_at_point(vx, vy, vz, vtu_file=vtu_file)
                    bx, by, bz = result["Bx"], result["By"], result["Bz"]
                    writer.writerow(
                        [
                            time_val,
                            i + 1,
                            format_csv_float(np.sqrt(bx * bx + by * by + bz * bz)),
                            format_csv_float(bx),
                            format_csv_float(by),
                            format_csv_float(bz),
                            format_csv_float(result["x"]),
                            format_csv_float(result["y"]),
                            format_csv_float(result["z"]),
                        ]
                    )
                    log_file.write(
                        f"{time_val},{i + 1},{result['mesh_distance']},"
                        f"{result['idw_neighbors_used']},"
                        f"{int(bool(result['idw_exact_node_hit']))},"
                        f"{result['idw_support_radius_m']},"
                        f"{result['idw_support_mean_distance_m']}\n"
                    )

        print("[OK] Wrote magnetic-field CSV")
        append_run_integrity(self.cwd, "magnetic_field_exterior_fields", "required Bx/By/Bz fields written")
        log_time_coverage(
            self.cwd,
            "magnetic_field_exterior",
            times_written,
            expected_time_dependent=not self.is_steady_state,
            n_steps=len(self.vtu_files),
        )

def main() -> None:
    """Brief: CLI entry for exterior magnetic probes; loads optional config ``probe_points``.

    Inputs:
        None: Parses ``--config`` and ``--run-dir`` from argv.

    Outputs:
        None: Instantiates ``MagneticFieldExterior`` and runs extraction.
    """
    parser = argparse.ArgumentParser(description="Extract magnetic field exterior probe data")
    parser.add_argument("--config", default="results/processing_config.json")
    parser.add_argument("--run-dir", default=".")
    args = parser.parse_args()

    probe_points = None
    os.chdir(args.run_dir)
    config_path = Path(args.config)
    if config_path.exists():
        data = json.loads(config_path.read_text(encoding="utf-8"))
        for script in data.get("scripts", []) or []:
            name = script.get("name", "") if isinstance(script, dict) else str(script)
            if name == "magnetic_field_exterior" and isinstance(script, dict):
                probe_points = (script.get("parameters") or {}).get("probe_points")
                break

    processor = MagneticFieldExterior(verbose=False, probe_points=probe_points)
    processor.run()




if __name__ == "__main__":
    main()
