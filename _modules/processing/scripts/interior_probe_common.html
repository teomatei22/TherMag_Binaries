def load_jint_probe_points_from_config(
    config_path: Path,
    script_names: Sequence[str],
) -> List[Any] | None:
    """Brief: Load J_int ``probe_points`` from processing config for named script entries.

    Inputs:
        config_path: Path to ``processing_config.json`` [path].
        script_names: Script ``name`` values to search for probe parameters.

    Outputs:
        list | None: Probe point entries, or ``None`` if config missing / unset.
    """
    if not config_path.exists():
        return None
    with open(config_path, "r", encoding="utf-8") as handle:
        data = json.load(handle)
    for want in script_names:
        for script in data.get("scripts", []) or []:
            name = script.get("name", "") if isinstance(script, dict) else str(script)
            if name != want or not isinstance(script, dict):
                continue
            params = script.get("parameters") or {}
            probe_points = params.get("probe_points")
            if probe_points:
                return probe_points
    return data.get("probe_points", {}).get("jint")

def normalize_probe_dict(probe: dict) -> Optional[Dict[str, Any]]:
    """Brief: Validate and coerce a probe dict to float ``x,y,z`` keys.

    Inputs:
        probe: Mapping expected to contain ``x``, ``y``, ``z`` [m].

    Outputs:
        dict | None: ``{"x","y","z"}`` floats, or ``None`` if invalid.
    """
    try:
        x = float(probe["x"])
        y = float(probe["y"])
        z = float(probe["z"])
    except (KeyError, TypeError, ValueError):
        return None
    return {"x": x, "y": y, "z": z}

    def _effective_geometry_type(self) -> str:
        """Brief: Resolve geometry family from contract, falling back to ``geometry.geometry_type``.

        Inputs:
            self: Mixin host.

        Outputs:
            str: Lowercased family (``sphere``, ``prism``, ``cylinder``, …).

            return contract.family
        return str(getattr(self.geometry, "geometry_type", "")).lower()

    def _sphere_radii(self) -> Tuple[float, Optional[float], Optional[float], Optional[float]]:
        """Brief: Return sphere body/layer/cavity/domain radii required for material-body sampling.

        Inputs:
            self: Mixin host whose sphere contract has metal radius ``R_sphere`` and domain ``R_sph``.

        Outputs:
            tuple: ``(r_outer, r_layer, r_cavity, r_domain)`` in metres (``None`` where absent).
        """
        contract = self._geometry_contract()
        if not contract.is_sphere or contract.body_outer_radius_m is None:
            raise RuntimeError(
                "Sphere material-body sampling requires R_sphere; "
                "R_sph is the simulation-domain radius."
            )
        return (
            float(contract.body_outer_radius_m),
            None if contract.layer_interface_radius_m is None else float(contract.layer_interface_radius_m),
            None if contract.cavity_radius_m is None else float(contract.cavity_radius_m),
            None if contract.domain_radius_m is None else float(contract.domain_radius_m),
        )

    def _is_inside_inner_cavity(self, x: float, y: float, z: float) -> bool:
        """Brief: Test whether (x,y,z) lies in the hollow inner cavity (air) region.

        Inputs:
            self: Mixin host.
            x: Probe x [m].
            y: Probe y [m].
            z: Probe z [m].

        Outputs:
            bool: ``True`` if inside cavity for hollow sphere/prism/cylinder topologies.
        """
        contract = self._geometry_contract()
        if not contract.is_hollow or self.config.inner_air_id is None:
            return False

        geo_type = self._effective_geometry_type()
        if geo_type == "sphere":
            _, _, r_cavity, _ = self._sphere_radii()
            if r_cavity is None:
                return False
            return float(np.sqrt(x * x + y * y + z * z)) <= r_cavity

        L = float(self.geometry.L)
        T = self.geometry.cavity_thickness_m()
        if geo_type == "prism":
            W = float(self.geometry.W)
            H = float(self.geometry.H)
            if T <= 0:
                return False
            W_inn = max(0.0, W - T)
            H_inn = max(0.0, H - T)
            return abs(x) <= L / 2.0 and abs(y) <= W_inn and abs(z) <= H_inn

        R = float(self.geometry.R)
        if T > 0 and abs(x) <= L / 2.0:
            rho = float(np.sqrt(y * y + z * z))
            R_inn = max(0.0, R - T)
            return rho <= R_inn
        return False

    def _is_in_air_region(self, x: float, y: float, z: float) -> bool:
        """Brief: Test whether (x,y,z) is in outer or inner air (not solid conductor).

        Inputs:
            self: Mixin host.
            x: Probe x [m].
            y: Probe y [m].
            z: Probe z [m].

        Outputs:
            bool: ``True`` for exterior air, domain exterior, or hollow cavity air.
        """
        geo_type = self._effective_geometry_type()
        if geo_type == "sphere":
            contract = self._geometry_contract()
            r_outer, _, r_cavity, r_domain = self._sphere_radii()
            r3 = float(np.sqrt(x * x + y * y + z * z))
            if r_domain is not None and r3 > r_domain:
                return True
            if r3 > r_outer:
                return True
            return bool(contract.is_hollow and r_cavity is not None and r3 <= r_cavity)

        L = float(self.geometry.L)
        T = self.geometry.cavity_thickness_m()
        if geo_type == "prism":
            W = float(self.geometry.W)
            H = float(self.geometry.H)
            in_axial = abs(x) <= L / 2.0
            in_y = abs(y) <= W
            in_z = abs(z) <= H
            if T > 0:
                W_inn, H_inn = W - T, H - T
                if in_axial and abs(y) <= W_inn and abs(z) <= H_inn:
                    return True
            return not (in_axial and in_y and in_z)

        R = float(self.geometry.R)
        rho = float(np.sqrt(y * y + z * z))
        if abs(x) > L / 2.0 or rho > R:
            return True
        return T > 0 and self._is_inside_inner_cavity(x, y, z)

class InteriorProbeSampler(ProbeRegionMixin, BasePostProcessor):
    """Brief: ParaView IDW sampler for interior J, T, and φ at configured probe points.

    Inputs:
        working_dir: FEM run directory [path].
        verbose: Enable verbose logging.
        probe_points: Optional explicit probes; else scaled J_int defaults.

    Outputs:
        InteriorProbeSampler: Post-processor with ``sample_at_point`` IDW API.
    """
    def __init__(
        self,
        working_dir=None,
        verbose=False,
        probe_points: list | None = None,
    ):
        """Brief: Initialize sampler, validate IDW config, and resolve probe point list.

        Inputs:
            self: Sampler instance.
            working_dir: FEM run directory [path].
            verbose: Verbose flag.
            probe_points: Optional list/tuples/dicts of probe xyz [m].

        Outputs:
            None: Sets ``probe_points``, IDW settings cache, and extraction config.
        """
        super().__init__(working_dir=working_dir, verbose=verbose)
        self.ext_cfg = self.pp_settings["extraction"]
        self.jint_cfg = self.ext_cfg["jint_defaults"]
        self._validate_required_config()
        self._cache_key: Optional[Tuple[str, bool, Tuple[int, ...]]] = None
        self._cache: Optional[Dict[str, Any]] = None

        self.probe_points: List[Dict[str, Any]] = []
        if probe_points:
            for probe in probe_points:
                if isinstance(probe, (tuple, list)) and len(probe) >= 3:
                    self.probe_points.append(
                        {"x": float(probe[0]), "y": float(probe[1]), "z": float(probe[2])}
                    )
                elif isinstance(probe, dict):
                    normalized = normalize_probe_dict(probe)
                    if normalized is not None:
                        self.probe_points.append(normalized)

        if not self.probe_points:
            gtype = str(self.geometry.geometry_type).lower()
            geo_params = self.geometry.probe_scaling_params()
            variant = str(getattr(self.geometry, "variant", "full") or "full")
            
            if gtype == "sphere":
                if variant.lower() == "hollow":
                    geo_params["T_outer"] = float(self.geometry.T_outer)

                elif variant.lower() == "layer_over_hollow":
                    geo_params["T_inner"] = float(self.geometry.T_inner)
                    geo_params["T_outer"] = float(self.geometry.T_outer)
            
            from probe_scaling import compute_jint_probe_positions

            self.probe_points = [
                {"x": x, "y": y, "z": z}
                for x, y, z in compute_jint_probe_positions(
                    geo_params,
                    gtype,
                    variant=variant,
                )
            ]
            if verbose:
                print(f"Using default Jint probe points: {len(self.probe_points)} points")

    def __del__(self):
        """Brief: Destructor that clears ParaView pipeline cache objects.

        Inputs:
            self: Sampler instance.

        Outputs:
            None: Best-effort ``_clear_cache``.
        """
        try:
            self._clear_cache()
        except Exception:
            pass

    def _validate_required_config(self) -> None:
        """Brief: Ensure extraction IDW keys and J_int point multipliers exist in settings.

        Inputs:
            self: Sampler with ``pp_settings["extraction"]``.

    def _material_names(self) -> Dict[str, str]:
        cached = getattr(self, "_cached_material_names", None)
        if cached is not None:
            return cached

        names: Dict[str, str] = {}
        project_path = Path(self.cwd) / "project.json"
        if project_path.is_file():
            try:
                data = json.loads(project_path.read_text(encoding="utf-8"))
                materials = ((data.get("materials") or {}).get("materials") or {})
                simulation = data.get("simulation") or {}
                for region, sim_key in (
                    ("outer_material", "outer_material_name"),
                    ("inner_material", "inner_material_name"),
                ):
                    snapshot = materials.get("outer" if region == "outer_material" else "inner")
                    name = snapshot.get("name") if isinstance(snapshot, dict) else None
                    if not name:
                        name = simulation.get(sim_key)
                    if name:
                        names[region] = str(name)
            except Exception:
                pass

        self._cached_material_names = names
        return names

    def _conductive_region(self, x: float, y: float, z: float) -> str:
        if self.config.inner_material_id is None or not str(self.config.model_type).startswith("layer_over_"):
            return "outer_material"

        geo_type = self._effective_geometry_type()
        if geo_type == "sphere":
            _, r_layer, _, _ = self._sphere_radii()
            if r_layer is None:
                return "outer_material"
            radius = float(np.sqrt(x * x + y * y + z * z))
            return "inner_material" if radius <= r_layer else "outer_material"

        if geo_type == "prism":
            w_mid = float(self.geometry.W) - float(self.geometry.T_outer)
            h_mid = float(self.geometry.H) - float(self.geometry.T_outer)
            if abs(y) <= w_mid and abs(z) <= h_mid:
                return "inner_material"
            return "outer_material"

        r_mid = float(self.geometry.R) - float(self.geometry.T_outer)
        rho = float(np.sqrt(y * y + z * z))
        return "inner_material" if rho <= r_mid else "outer_material"

    def _probe_expected_region(self, x: float, y: float, z: float) -> str:
        """Brief: Classify probe xyz into ``inner_air``, ``outer_air``, or a material region.

        Inputs:
            self: Sampler instance.
            x: Probe x [m].
            y: Probe y [m].
            z: Probe z [m].

        Outputs:
            str: Expected sampling region name.
        """
        if self._is_in_air_region(x, y, z):
            return "inner_air" if self._is_inside_inner_cavity(x, y, z) else "outer_air"
        return self._conductive_region(x, y, z)

    def _sampling_region_ids(
        self, expected_region: str
    ) -> Tuple[bool, Tuple[int, ...], str]:
        """Brief: Map expected region name to (in_air, GeometryIds, region label) for Threshold.

        Inputs:
            self: Sampler with ``config`` air/conductor IDs.
            expected_region: ``inner_air`` / ``outer_air`` / material region.

        Outputs:
            tuple: ``(in_air, air_ids_tuple, region_label)``.
        """
        expected = str(expected_region).strip().lower()
        if expected == "inner_air":
            contract = self._geometry_contract()
            if not contract.is_hollow:
                raise RuntimeError("Probe requested inner_air sampling for a non-hollow geometry.")
            if self.config.inner_air_id is None:
                raise RuntimeError(
                    "Probe requested inner_air sampling, but model_config has no inner_air_id."
                )
            return True, (int(self.config.inner_air_id),), "inner_air"
        if expected == "outer_air":
            return True, (int(self.config.outer_air_id),), "outer_air"
        if expected in ("outer_material", "inner_material"):
            material_name = self._material_names().get(expected)
            label = f"{expected}:{material_name}" if material_name else expected
            return False, (), label
        return False, (), "conductor"

    def _prepare_cache(
        self, vtu_file: str, in_air: bool, air_ids: Tuple[int, ...] = ()
    ) -> Dict[str, Any]:
        """Brief: Build or reuse VTU Threshold+MergeBlocks cache of J/T/P point arrays.

        Inputs:
            self: Sampler instance.
            vtu_file: VTU path for this timestep [path].
            in_air: Threshold air GeometryIds when true; else conductive range.
            air_ids: Outer/inner air GeometryId tuple when ``in_air``.

        Outputs:
            dict: Cache with ``points_np``, J/T/P arrays, and ParaView ``objects``.
        """
        key = (vtu_file, in_air, air_ids)
        if self._cache_key == key and self._cache is not None:
            return self._cache

        self._clear_cache()
        reader = self.make_reader("current_int_vtu", vtu_file)
        reader.PointArrayStatus = [
            "potential flux",
            "temperature flux",
            "sourcecurrentx",
            "sourcecurrenty",
            "sourcecurrentz",
            "temperature",
            "potential",
        ]
        reader.UpdatePipeline()
        objects = [reader]

        if in_air:
            if not air_ids:
                air_ids = (int(self.config.outer_air_id),)
            low_id, high_id = min(air_ids), max(air_ids)
            th = Threshold(registrationName="Threshold_air_J_int", Input=reader)
        else:
            th = Threshold(registrationName="Threshold_conductor_J_int", Input=reader)
            low_id, high_id = self.config.conductive_analysis_range()
        th.Scalars = ["CELLS", "GeometryIds"]
        self.set_threshold_between(th, float(low_id), float(high_id))
        merged = MergeBlocks(th)
        objects.extend([th, merged])

        vtk_data = servermanager.Fetch(merged)
        points = vtk_data.GetPoints()
        points_np = (
            ns.vtk_to_numpy(points.GetData()) if points is not None else np.empty((0, 3), dtype=float)
        )
        point_data = vtk_data.GetPointData()

        self._cache = {
            "objects": objects,
            "points_np": points_np,
            "JOhm_array": point_data.GetArray("potential flux"),
            "JSourceX_array": point_data.GetArray("sourcecurrentx"),
            "JSourceY_array": point_data.GetArray("sourcecurrenty"),
            "JSourceZ_array": point_data.GetArray("sourcecurrentz"),
            "T_array": point_data.GetArray("temperature"),
            "P_array": point_data.GetArray("potential"),
            "sampling_ids": (
                tuple(int(v) for v in air_ids)
                if air_ids
                else tuple(self.config.conductive_analysis_range())
            ),
        }
        self._cache_key = key
        return self._cache

    def sample_at_point(self, x: float, y: float, z: float, vtu_file: str | None = None) -> Dict[str, Any]:
        """Brief: IDW-sample J, T, and P at (x,y,z) in the probe's expected mesh region.

        Inputs:
            self: Sampler instance.
            x: Probe x [m].
            y: Probe y [m].
            z: Probe z [m].
            vtu_file: Optional VTU path; defaults to first timestep [path].

        Outputs:
            dict: ``Jx,Jy,Jz,T,P``, nearest point, sample_region, and IDW diagnostics.
        """
        if vtu_file is None:
            vtu_file = self.vtu_files[0]

        expected_region = self._probe_expected_region(x, y, z)
        in_air, air_ids, region = self._sampling_region_ids(expected_region)

        cache = self._prepare_cache(vtu_file, in_air, air_ids=air_ids)
        points_np = cache["points_np"]
        if points_np.size == 0:
            raise RuntimeError(
                f"No mesh points available for IDW interpolation in region {region!r}."
            )

        target = np.array([x, y, z])
        distances = np.linalg.norm(points_np - target, axis=1)
        k = min(int(self.ext_cfg["idw_neighbors"]), len(points_np))
        nn_idx = select_nearest_indices(distances, k)
        d_k = distances[nn_idx]
        weights, exact_local, diagnostics = idw_weights(
            d_k,
            epsilon=float(self.ext_cfg["idw_epsilon"]),
            on_node_epsilon=float(self.ext_cfg["on_node_epsilon"]),
        )
        exact_idx = int(nn_idx[exact_local]) if exact_local is not None else None

        def _idw_scalar(arr, name: str) -> float:
            """Brief: IDW (or exact-node) interpolate a scalar VTK point array at the probe.

            Inputs:
                arr: VTK data array (or ``None``).
                name: Array name for error messages.

            Outputs:
                float: Interpolated scalar [array units].
            """
            if arr is None:
                raise RuntimeError(f"Missing required point array '{name}'.")
            np_arr = ns.vtk_to_numpy(arr)
            if np_arr.size == 0:
                raise RuntimeError(f"Point array '{name}' is empty.")
            if exact_idx is not None:
                return float(np_arr[exact_idx])
            assert weights is not None
            return float(np.sum(np_arr[nn_idx] * weights) / np.sum(weights))

        def _idw_vector(arr, name: str) -> Tuple[float, float, float]:
            """Brief: IDW (or exact-node) interpolate a 3-vector VTK point array at the probe.

            Inputs:
                arr: VTK vector array (or ``None``).
                name: Array name for error messages.

            Outputs:
                tuple: ``(vx, vy, vz)`` interpolated components [array units].
            """
            if arr is None:
                raise RuntimeError(f"Missing required point array '{name}'.")
            np_arr = ns.vtk_to_numpy(arr)
            if np_arr.size == 0:
                raise RuntimeError(f"Point array '{name}' is empty.")
            if exact_idx is not None:
                vec = np_arr[exact_idx]
            else:
                assert weights is not None
                vec = np.sum(np_arr[nn_idx] * weights[:, None], axis=0) / np.sum(weights)
            return float(vec[0]), float(vec[1]), float(vec[2])

        JOhmx, JOhmy, JOhmz = _idw_vector(cache["JOhm_array"], "potential flux")
        JSourcex = _idw_scalar(cache["JSourceX_array"], "sourcecurrentx")
        JSourcey = _idw_scalar(cache["JSourceY_array"], "sourcecurrenty")
        JSourcez = _idw_scalar(cache["JSourceZ_array"], "sourcecurrentz")

        JSeebeckx = JSourcex - JOhmx
        JSeebecky = JSourcey - JOhmy
        JSeebeckz = JSourcez - JOhmz

        T_val = _idw_scalar(cache["T_array"], "temperature")
        P_val = _idw_scalar(cache["P_array"], "potential")
        nearest_idx = int(np.argmin(distances))
        nearest_point = points_np[nearest_idx]
        return {
            "JOhmx": JOhmx,
            "JOhmy": JOhmy,
            "JOhmz": JOhmz,
            "JSeebeckx": JSeebeckx,
            "JSeebecky": JSeebecky,
            "JSeebeckz": JSeebeckz,
            "JSourcex": JSourcex,
            "JSourcey": JSourcey,
            "JSourcez": JSourcez,
            "T": T_val,
            "P": P_val,
            "mesh_distance": float(diagnostics["mesh_distance"]),
            "nearest_point": (
                float(nearest_point[0]),
                float(nearest_point[1]),
                float(nearest_point[2]),
            ),
            "sample_region": region,
            "idw_neighbors_used": diagnostics["idw_neighbors_used"],
            "idw_exact_node_hit": diagnostics["idw_exact_node_hit"],
            "idw_support_radius_m": diagnostics["idw_support_radius_m"],
            "idw_support_mean_distance_m": diagnostics["idw_support_mean_distance_m"],
        }
