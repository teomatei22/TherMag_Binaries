def _format_probe_coord(value: float, decimals: int = 3) -> str:
    """Brief: Format a probe coordinate for plot titles, clamping tiny values to zero.

    Inputs:
        value: Coordinate component [m].
        decimals: Digits after the decimal point.

    Outputs:
        str: Fixed-point coordinate text suitable for titles.
    """
    number = float(value)
    if abs(number) < 0.5 * 10 ** (-decimals):
        return f"{0.0:.{decimals}f}"
    text = f"{number:.{decimals}f}"
    if text.startswith("-0.") and float(text) == 0.0:
        return f"{0.0:.{decimals}f}"
    return text

def _probe_coords_label(px: float, py: float, pz: float) -> str:
    """Brief: Build a parenthesized (x, y, z) label for probe plot titles.

    Inputs:
        px: Probe x [m].
        py: Probe y [m].
        pz: Probe z [m].

    Outputs:
        str: Formatted ``(x, y, z)`` label.
    """
    return f"({_format_probe_coord(px)}, {_format_probe_coord(py)}, {_format_probe_coord(pz)})"

def _vector_probe_layout(n_cols: int) -> dict[str, Optional[int]]:
    """Brief: Map CSV column indices for vector probes (compact vs magnitude-inclusive layouts).

    Inputs:
        n_cols: Number of CSV columns.

    Outputs:
        dict: Keys ``mag``, ``vx``, ``vy``, ``vz``, ``x``, ``y``, ``z`` - column index or ``None``.
    """
    if n_cols >= 9:
        return {"mag": 2, "vx": 3, "vy": 4, "vz": 5, "x": 6, "y": 7, "z": 8}
    return {"mag": None, "vx": 2, "vy": 3, "vz": 4, "x": 5, "y": 6, "z": 7}

def parse_point_ids(text: Optional[str]) -> PointSelection:
    """Brief: Parse CLI point-ID selection text into ``None`` (auto), ``'all'``, or an explicit ID set.

    Inputs:
        text: Comma-separated IDs, or ``auto``/``all``/empty [str].

    Outputs:
        PointSelection: ``None``, ``'all'``, or ``set[int]`` of requested point IDs.
    """
    if text is None:
        return None

    s = str(text).strip().lower()
    if s == "" or s == "auto":
        return None
    if s == "all":
        return "all"

    point_ids: Set[int] = set()
    for token in s.split(","):
        token = token.strip()
        if not token:
            continue
        try:
            point_ids.add(int(token))
        except ValueError as exc:
            raise ValueError(
                f"Invalid point-id token {token!r}. "
                "Use a comma-separated list such as 1,2,3, or use 'auto'/'all'."
            ) from exc

    return point_ids

def _unique_sorted_point_ids(point_id_column: np.ndarray) -> List[int]:
    """Brief: Collect unique probe point IDs from a CSV column, sorted ascending.

    Inputs:
        point_id_column: Array of point_id values.

    Outputs:
        list: Sorted unique integer point IDs.
    """
    return sorted(int(pid) for pid in np.unique(point_id_column))

def _evenly_spaced_ids(ids: Sequence[int], count: int) -> List[int]:
    """Brief: Pick up to ``count`` evenly spaced IDs from a sorted ID sequence.

    Inputs:
        ids: Sorted available point IDs.
        count: Maximum IDs to return.

    Outputs:
        list: Selected point IDs (order of selection, de-duplicated).
    """
    if count <= 0:
        return []
    if len(ids) <= count:
        return list(ids)


    positions = np.linspace(0, len(ids) - 1, count)
    chosen: List[int] = []
    for pos in positions:
        pid = ids[int(round(pos))]
        if pid not in chosen:
            chosen.append(pid)
    return chosen

def _top_ids_by_value(raw_data: np.ndarray, ids: Sequence[int], value_cols: Sequence[int], count: int) -> List[int]:
    """Brief: Rank point IDs by peak absolute field value across given CSV columns.

    Inputs:
        raw_data: Full CSV numeric array (time, point_id, values) [mixed].
        ids: Candidate point IDs.
        value_cols: Column indices used for scoring [field units].
        count: Maximum top IDs to keep.

    Outputs:
        list: Point IDs with largest ``nanmax(|values|)``, highest first.
    """
    if count <= 0:
        return []

    scores = []
    ncols = raw_data.shape[1]
    valid_cols = [c for c in value_cols if 0 <= c < ncols]
    if not valid_cols:
        return []

    for pid in ids:
        rows = raw_data[raw_data[:, 1].astype(int) == pid]
        if rows.size == 0:
            continue
        vals = rows[:, valid_cols]
        score = np.nanmax(np.abs(vals))
        scores.append((float(score), pid))

    scores.sort(reverse=True)
    return [pid for _, pid in scores[:count]]

def filter_point_ids(
    raw_data: np.ndarray,
    selected_ids: PointSelection,
    *,
    auto_value_cols: Optional[Sequence[int]] = None,
    max_auto_points: int = 6,
    label: str = "field",
) -> List[int]:
    """Brief: Resolve which probe point IDs to plot (explicit, all, or auto top+spread).

    Inputs:
        raw_data: CSV numeric array with point_id in column 1 [mixed].
        selected_ids: ``None`` (auto), ``'all'``, or explicit ``set[int]``.
        auto_value_cols: Columns for auto top-value ranking [field units].
        max_auto_points: Cap on auto-selected IDs.
        label: Log label for the field being filtered.

    Outputs:
        list: Final sorted (or explicit-order) point IDs to plot.
    """
    available = _unique_sorted_point_ids(raw_data[:, 1])

    if selected_ids == "all":
        print(f"[INFO] {label}: plotting all {len(available)} point IDs")
        return available

    if isinstance(selected_ids, set):
        chosen = [pid for pid in available if pid in selected_ids]
        missing = sorted(selected_ids.difference(available))
        if missing:
            print(f"[WARN] {label}: requested point IDs not present in CSV: {missing}")
        print(f"[INFO] {label}: plotting explicit point IDs: {chosen}")
        return chosen


    if len(available) <= max_auto_points:
        print(f"[INFO] {label}: auto mode; plotting all available point IDs: {available}")
        return available

    n_top = max(1, max_auto_points // 2)
    n_spread = max_auto_points - n_top

    chosen: List[int] = []

    if auto_value_cols is not None:
        for pid in _top_ids_by_value(raw_data, available, auto_value_cols, n_top):
            if pid not in chosen:
                chosen.append(pid)

    for pid in _evenly_spaced_ids(available, n_spread):
        if pid not in chosen:
            chosen.append(pid)


    if len(chosen) < max_auto_points:
        for pid in _evenly_spaced_ids(available, max_auto_points):
            if pid not in chosen:
                chosen.append(pid)
            if len(chosen) >= max_auto_points:
                break

    if len(chosen) < max_auto_points:
        for pid in available:
            if pid not in chosen:
                chosen.append(pid)
            if len(chosen) >= max_auto_points:
                break

    chosen = sorted(chosen)
    print(
        f"[INFO] {label}: auto mode; plotting {len(chosen)} of "
        f"{len(available)} point IDs: {chosen}"
    )
    return chosen

def _is_steady(times: np.ndarray) -> bool:
    """Brief: Detect stationary/steady probe series that should skip component PNGs.

    Inputs:
        times: Sample times for one probe [s].

    Outputs:
        bool: ``True`` when a single sample or trivial t=0 steady set.
    """
    return len(times) == 1 or (np.max(times) == 0 and len(times) < 5)

def _save_component_plot(
    results_dir: str,
    times: np.ndarray,
    values: np.ndarray,
    *,
    ylabel: str,
    title: str,
    filename: str,
    color: str,
) -> None:
    """Brief: Write one Matplotlib line+scatter component PNG under ``results_dir``.

    Inputs:
        results_dir: Output directory for the PNG [path].
        times: Abscissa time samples [s].
        values: Ordinate field samples [field units].
        ylabel: Y-axis label with units.
        title: Plot title.
        filename: Output basename (``.png``).
        color: Matplotlib color for the series.

    Outputs:
        None: Saves ``results_dir/filename`` and closes the figure.
    """
    _fig, ax = plt.subplots(1, 1, figsize=(10, 6))
    ax.plot(times, values, "-", color=color, linewidth=2)
    ax.scatter(times, values, s=30, c=color, alpha=0.5)
    ax.set_xlabel("Time [s]", fontsize=14)
    ax.set_ylabel(ylabel, fontsize=14)
    ax.set_title(title, fontsize=16)
    ax.grid(True, alpha=0.3)
    ax.yaxis.set_major_formatter(
        FuncFormatter(lambda value, _pos: truncate_scientific(value))
    )
    plt.tight_layout()
    plt.savefig(os.path.join(results_dir, filename), dpi=150)
    plt.close()

def generate_bext_plot(
    results_dir: str,
    *,
    selected_point_ids: PointSelection = None,
    include_magnitude: bool = False,
    max_auto_points: int = 6,
) -> None:
    """Brief: Plot exterior B_x/B_y/B_z (and optional |B|) time series from ``magnetic_field_data.csv``.

    Inputs:
        results_dir: Directory containing magnetic field CSV [path].
        selected_point_ids: Point-ID filter (auto/all/explicit).
        include_magnitude: Also write |B| diagnostic PNGs.
        max_auto_points: Auto-mode ID cap.

    Outputs:
        None: Writes ``magnetic_field_B*_p*.png`` (and optional abs) under ``results_dir``.
    """
    csv_path = os.path.join(results_dir, "magnetic_field_data.csv")
    if not os.path.exists(csv_path):
        print(f"Skipping magnetic_field plot: {csv_path} not found")
        return

    try:
        raw_data = np.genfromtxt(csv_path, delimiter=",", skip_header=1)
        if raw_data.ndim == 1:
            raw_data = raw_data.reshape(1, -1)

        if raw_data.shape[1] < 8:
            print(
                f"[WARN] magnetic_field_data.csv has {raw_data.shape[1]} columns; "
                "expected at least 8 for compact Bx/By/Bz output"
            )
            return

        point_ids = filter_point_ids(
            raw_data,
            selected_point_ids,
            auto_value_cols=(2, 3, 4),
            max_auto_points=max_auto_points,
            label="B_ext",
        )

        for ipid in point_ids:
            data = raw_data[raw_data[:, 1].astype(int) == ipid]
            if data.size == 0:
                continue

            times = data[:, 0]
            if _is_steady(times):
                print(
                    f"[INFO] B_ext probe {ipid}: stationary dataset - CSV kept, skipping component plots"
                )
                continue

            cols = _vector_probe_layout(data.shape[1])
            Bx, By, Bz = data[:, cols["vx"]], data[:, cols["vy"]], data[:, cols["vz"]]
            px, py, pz = data[0, cols["x"]], data[0, cols["y"]], data[0, cols["z"]]
            coords = _probe_coords_label(px, py, pz)

            if include_magnitude:
                mag_col = cols["mag"]
                B_mag = data[:, mag_col] if mag_col is not None else np.sqrt(Bx**2 + By**2 + Bz**2)
                _save_component_plot(
                    results_dir,
                    times,
                    B_mag,
                    ylabel=r"$|\vec{B}|$ [T]",
                    title=rf"$|\vec{{B}}|$ at P{ipid} {coords}",
                    color="red",
                    filename=f"magnetic_field_abs_p{ipid}.png",
                )
                print(f"[OK] Wrote magnetic field exterior |B| diagnostic plot (probe {ipid})")

            for comp, values, color in (
                ("x", Bx, "steelblue"),
                ("y", By, "seagreen"),
                ("z", Bz, "darkorange"),
            ):
                _save_component_plot(
                    results_dir,
                    times,
                    values,
                    ylabel=rf"$B_{comp}$ [T]",
                    title=rf"$B_{{{comp}}}$ at P{ipid} {coords}",
                    color=color,
                    filename=f"magnetic_field_B{comp}_p{ipid}.png",
                )
                print(f"[OK] Wrote magnetic field exterior B{comp} plot (probe {ipid})")

    except Exception as e:
        print(f"Error generating magnetic_field plot: {e}")

def generate_jint_plot(
    results_dir: str,
    *,
    selected_point_ids: PointSelection = None,
    include_magnitude: bool = False,
    max_auto_points: int = 6,
) -> None:
    """Brief: Plot interior J_x/J_y/J_z (and optional |J|) time series from current-density CSV.

    Inputs:
        results_dir: Directory containing ``current_density_interior_data.csv`` [path].
        selected_point_ids: Point-ID filter (auto/all/explicit).
        include_magnitude: Also write |J| diagnostic PNGs.
        max_auto_points: Auto-mode ID cap.

    Outputs:
        None: Writes ``current_density_J*_p*.png`` under ``results_dir``.
    """
    csv_path = os.path.join(results_dir, "current_density_interior_data.csv")
    if not os.path.exists(csv_path):
        print(f"Skipping current_density plot: {csv_path} not found")
        return

    try:
        data = np.genfromtxt(
            csv_path,
            delimiter=",",
            names=True,
            dtype=None,
            encoding="utf-8",
        )
        data = np.atleast_1d(data)
        names = tuple(data.dtype.names or ())
        required = (
            "time",
            "point_id",
            "JSource_mag",
            "JSource_x",
            "JSource_y",
            "JSource_z",
            "x",
            "y",
            "z",
        )
        missing = [name for name in required if name not in names]
        if missing:
            print(
                "[WARN] current_density_interior_data.csv is missing required "
                f"JSource columns: {missing}"
            )
            return

        point_ids_column = np.asarray(data["point_id"], dtype=float)
        selection_data = np.column_stack(
            [
                np.asarray(data["time"], dtype=float),
                point_ids_column,
                np.asarray(data["JSource_x"], dtype=float),
                np.asarray(data["JSource_y"], dtype=float),
                np.asarray(data["JSource_z"], dtype=float),
            ]
        )

        point_ids = filter_point_ids(
            selection_data,
            selected_point_ids,
            auto_value_cols=(2, 3, 4),
            max_auto_points=max_auto_points,
            label="J_int",
        )

        for ipid in point_ids:
            mask = point_ids_column.astype(int) == ipid
            probe = data[mask]
            if probe.size == 0:
                continue

            times = np.asarray(probe["time"], dtype=float)
            if _is_steady(times):
                print(
                    f"[INFO] J_int probe {ipid}: stationary dataset - CSV kept, skipping component plots"
                )
                continue

            Jx = np.asarray(probe["JSource_x"], dtype=float)
            Jy = np.asarray(probe["JSource_y"], dtype=float)
            Jz = np.asarray(probe["JSource_z"], dtype=float)
            px = float(np.asarray(probe["x"], dtype=float)[0])
            py = float(np.asarray(probe["y"], dtype=float)[0])
            pz = float(np.asarray(probe["z"], dtype=float)[0])
            coords = _probe_coords_label(px, py, pz)

            for comp, values, color in (
                ("x", Jx, "steelblue"),
                ("y", Jy, "seagreen"),
                ("z", Jz, "darkorange"),
            ):
                _save_component_plot(
                    results_dir,
                    times,
                    values,
                    ylabel=rf"$J_{comp}$ [A/m$^2$]",
                    title=rf"$J_{{{comp}}}$ at P{ipid} {coords}",
                    color=color,
                    filename=f"current_density_J{comp}_p{ipid}.png",
                )
                print(f"[OK] Wrote current density interior J{comp} plot (probe {ipid})")

            if include_magnitude:
                J_mag = np.asarray(probe["JSource_mag"], dtype=float)
                _save_component_plot(
                    results_dir,
                    times,
                    J_mag,
                    ylabel=r"$|\vec{J}|$ [A/m$^2$]",
                    title=rf"$|\vec{{J}}|$ at P{ipid} {coords}",
                    color="blue",
                    filename=f"current_density_abs_p{ipid}.png",
                )
                print(f"[OK] Wrote current density interior |J| diagnostic plot (probe {ipid})")

    except Exception as e:
        print(f"Error generating current_density plot: {e}")

def generate_scalar_interior_plot(
    results_dir: str,
    *,
    csv_name: str,
    value_label: str,
    ylabel: str,
    title_prefix: str,
    color: str,
    filename_prefix: str,
    selected_point_ids: PointSelection = None,
    max_auto_points: int = 6,
) -> None:
    """Brief: Generic interior scalar probe time-series plots (temperature or potential).

    Inputs:
        results_dir: Results directory holding the scalar CSV [path].
        csv_name: CSV basename under ``results_dir``.
        value_label: Short field tag for logs (e.g. ``T_int``).
        ylabel: Y-axis label with units.
        title_prefix: Title stem before probe id/coords.
        color: Matplotlib series color.
        filename_prefix: PNG basename prefix before ``_p{id}.png``.
        selected_point_ids: Point-ID filter (auto/all/explicit).
        max_auto_points: Auto-mode ID cap.

    Outputs:
        None: Writes ``{filename_prefix}_p*.png`` under ``results_dir``.
    """
    csv_path = os.path.join(results_dir, csv_name)
    if not os.path.exists(csv_path):
        print(f"Skipping {value_label} plot: {csv_path} not found")
        return

    try:
        raw_data = np.genfromtxt(csv_path, delimiter=",", skip_header=1)
        if raw_data.ndim == 1:
            raw_data = raw_data.reshape(1, -1)
        if raw_data.shape[1] == 0:
            return
        n_cols = int(raw_data.shape[1])
        if n_cols < 6:
            print(
                f"[WARN] {csv_name} has {n_cols} columns; "
                "expected at least 6: time, point_id, value, x, y, z"
            )
            return
        x_col, y_col, z_col = (6, 7, 8) if n_cols >= 9 else (3, 4, 5)

        point_ids = filter_point_ids(
            raw_data,
            selected_point_ids,
            auto_value_cols=(2,),
            max_auto_points=max_auto_points,
            label=value_label,
        )

        for ipid in point_ids:
            data = raw_data[raw_data[:, 1].astype(int) == ipid]
            if data.size == 0:
                continue

            times = data[:, 0]
            if _is_steady(times):
                print(
                    f"[INFO] {value_label} probe {ipid}: stationary dataset - "
                    "CSV kept, skipping component plots"
                )
                continue
            values = data[:, 2]
            px, py, pz = data[0, x_col], data[0, y_col], data[0, z_col]
            coords = _probe_coords_label(px, py, pz)

            _save_component_plot(
                results_dir,
                times,
                values,
                ylabel=ylabel,
                title=rf"{title_prefix} at P{ipid} {coords}",
                color=color,
                filename=f"{filename_prefix}_p{ipid}.png",
            )
            print(f"[OK] Wrote {value_label} plot image (probe {ipid})")

    except Exception as e:
        print(f"Error generating {value_label} plots: {e}")

def main():
    """Brief: CLI entry: generate B/J/T/P probe PNGs and optional magnetic contour map from CSVs.

    Inputs:
        None: Parses ``working_dir`` and point-ID / magnitude flags from argv.

    Outputs:
        None: Writes plots under ``working_dir/results``; exits non-zero on missing results.
    """
    parser = argparse.ArgumentParser(description="Generate matplotlib plots from CSV data")
    parser.add_argument(
        "working_dir",
        nargs="?",
        default=os.getcwd(),
        help="Working directory containing results folder",
    )

    parser.add_argument(
        "--b-point-ids",
        default="auto",
        help=(
            "Magnetic exterior point IDs to plot: comma-separated list such as "
            "'1,2,3', or 'auto', or 'all'. Default: auto."
        ),
    )
    parser.add_argument(
        "--j-point-ids",
        default="auto",
        help=(
            "Current-density interior point IDs to plot: comma-separated list such as "
            "'5,9,14,15,23,24', or 'auto', or 'all'. Default: auto."
        ),
    )
    parser.add_argument(
        "--tp-point-ids",
        default="auto",
        help=(
            "Temperature/potential point IDs to plot: comma-separated list such as "
            "'1,2,3', or 'auto', or 'all'. Default: auto."
        ),
    )
    parser.add_argument(
        "--plot-all",
        action="store_true",
        help="Plot all point IDs for B, J, and temperature/potential.",
    )
    parser.add_argument(
        "--max-auto-points",
        type=int,
        default=6,
        help="Maximum number of point IDs plotted in auto mode for each field. Default: 6.",
    )
    parser.add_argument(
        "--include-magnitude-diagnostics",
        action="store_true",
        help="Also write |B| and |J| magnitude diagnostic plots for time_dependent data.",
    )

    args = parser.parse_args()

    working_dir = args.working_dir
    results_dir = os.path.join(working_dir, "results")

    if not os.path.exists(results_dir):
        print(f"Error: Results directory not found: {results_dir}")
        sys.exit(1)

    if args.plot_all:
        b_point_ids: PointSelection = "all"
        j_point_ids: PointSelection = "all"
        tp_point_ids: PointSelection = "all"
    else:
        try:
            b_point_ids = parse_point_ids(args.b_point_ids)
            j_point_ids = parse_point_ids(args.j_point_ids)
            tp_point_ids = parse_point_ids(args.tp_point_ids)
        except ValueError as exc:
            print(f"Error: {exc}")
            sys.exit(2)

    print(f"Generating plots from: {results_dir}")
    print("-" * 50)
    print("[INFO] Component plots are written for time_dependent datasets only.")
    if args.include_magnitude_diagnostics:
        print("[INFO] Magnitude diagnostics are enabled.")
    else:
        print("[INFO] Magnitude diagnostics are disabled.")

    generate_bext_plot(
        results_dir,
        selected_point_ids=b_point_ids,
        include_magnitude=args.include_magnitude_diagnostics,
        max_auto_points=args.max_auto_points,
    )
    generate_jint_plot(
        results_dir,
        selected_point_ids=j_point_ids,
        include_magnitude=args.include_magnitude_diagnostics,
        max_auto_points=args.max_auto_points,
    )
    generate_temperature_int_plot(
        results_dir,
        selected_point_ids=tp_point_ids,
        max_auto_points=args.max_auto_points,
    )
    generate_potential_int_plot(
        results_dir,
        selected_point_ids=tp_point_ids,
        max_auto_points=args.max_auto_points,
    )

    try:
        from magnetic_field_contour_map import plot_contour_map_from_csv

        plot_contour_map_from_csv(results_dir)
    except Exception as exc:
        print(f"Skipping magnetic contour map plot: {exc}")

    print("-" * 50)
    print("Plot generation complete")



if __name__ == "__main__":
    main()
