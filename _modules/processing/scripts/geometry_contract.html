def strip_geo_comment(line: str) -> str:
    """Brief: Strip a Gmsh ``//`` line comment and surrounding whitespace.

    Inputs:
        line: Raw ``.geo`` source line.

    Outputs:
        str: Code portion before ``//``, stripped.
    """
    return line.split("//", 1)[0].strip()

def geo_code_text(text: str) -> str:
    """Brief: Join non-empty, comment-stripped lines of a ``.geo`` text into clean code.

    Inputs:
        text: Full ``geometry.geo`` source.

    Outputs:
        str: Comment-free code with blank lines removed.
    """
    return "\n".join(
        clean for clean in (strip_geo_comment(line) for line in text.splitlines()) if clean
    )

def parse_geo_assignments(text: str, passes: int = 10) -> Dict[str, float]:
    """Brief: Evaluate successive ``name = expr;`` assignments from Gmsh geo text.

    Inputs:
        text: ``.geo`` source with numeric assignments.
        passes: Max multi-pass resolution iterations for dependency order.

    Outputs:
        dict: Mapping of symbol name - finite float value.
    """
    values: Dict[str, float] = {}
    safe: Dict[str, object] = {
        "Pi": math.pi,
        "pi": math.pi,
        "Sin": math.sin,
        "Cos": math.cos,
        "Tan": math.tan,
        "Sqrt": math.sqrt,
        "Min": min,
        "Max": max,
        "Abs": abs,
    }
    pat = re.compile(r"^\s*([A-Za-z_]\w*)\s*=\s*([^;]+)\s*;")
    lines = text.splitlines()
    for _ in range(passes):
        changed = False
        for raw in lines:
            line = strip_geo_comment(raw)
            m = pat.match(line)
            if not m:
                continue
            key, expr = m.group(1), m.group(2).strip()
            if any(token in expr for token in ("{", "}", '"', "'")):
                continue
            try:
                value = eval(expr, {"__builtins__": {}}, {**safe, **values})
            except Exception:
                continue
            if isinstance(value, (int, float)) and math.isfinite(float(value)):
                value_f = float(value)
                if values.get(key) != value_f:
                    values[key] = value_f
                    changed = True
        if not changed:
            break
    return values

def physical_names(text: str, dimension: str = "Volume") -> set[str]:
    """Brief: Collect Physical Volume/Surface/Line names declared in geo text (lowercased).

    Inputs:
        text: ``.geo`` source.
        dimension: Gmsh physical entity kind (default ``Volume``).

    Outputs:
        set: Lowercased physical region names.
    """
    code = geo_code_text(text)
    pat = re.compile(
        rf"Physical\s+{re.escape(dimension)}\s*\(\s*[\"']([^\"']+)[\"']",
        flags=re.IGNORECASE,
    )
    return {m.group(1).strip().lower() for m in pat.finditer(code)}

def first_value(values: Dict[str, float], names: Iterable[str]) -> Optional[float]:
    """Brief: Return the first finite float among named keys in a geo assignment map.

    Inputs:
        values: Symbol float map from ``parse_geo_assignments``.
        names: Preferred key names in priority order.

    Outputs:
        float | None: First finite value found, else ``None``.
    """
    for name in names:
        if name in values:
            try:
                value = float(values[name])
            except Exception:
                continue
            if math.isfinite(value):
                return value
    return None

def domain_radius_m(values: Dict[str, float]) -> Optional[float]:
    """Brief: Air-domain radius from ``R_sph``.

    Inputs:
        values: Symbol map from ``parse_geo_assignments``.

    Outputs:
        float | None: Positive outer air radius [m], else ``None``.
    """
    air = first_value(values, ("R_sph",))
    if air is not None and air > 0.0:
        return air
    return None

def domain_radius_from_geo_text(text: str) -> float:
    """Brief: Require a positive air-domain radius ``R_sph`` from geo text.

    Inputs:
        text: ``.geo`` source defining the simulation domain.

    Outputs:
        float: Domain radius [m].
    """
    values = parse_geo_assignments(text)
    radius = domain_radius_m(values)
    if radius is None or radius <= 0.0:
        raise ValueError("geometry.geo must define a positive R_sph domain radius")
    return float(radius)

@dataclass(frozen=True)
class GeometryContract:
    """Brief: Frozen dataclass of geometry family, variant, and characteristic radii [m].

    Inputs:
        family: Geometry family string (``sphere``, ``prism``, ``cylinder``, …).
        variant: Topology variant (``full``, ``hollow``, ``layer_over_*``).
        domain_radius_m: Outer simulation-domain radius [m].
        body_outer_radius_m: Conductor outer radius for spheres [m].
        layer_interface_radius_m: Optional layer interface radius [m].
        cavity_radius_m: Inner cavity radius for hollow variants [m].

    Outputs:
        GeometryContract: Immutable geometry metadata for probe/map scripts.
    """
    family: str
    variant: str
    domain_radius_m: Optional[float]
    body_outer_radius_m: Optional[float]
    layer_interface_radius_m: Optional[float]
    cavity_radius_m: Optional[float]

    @property
    def is_sphere(self) -> bool:
        """Brief: Whether this contract describes a sphere-family geometry.

        Inputs:
            self: GeometryContract instance.

        Outputs:
            bool: ``True`` when ``family == "sphere"``.
        """
        return self.family == "sphere"

    @property
    def is_hollow(self) -> bool:
        """Brief: Whether the variant includes an inner cavity (hollow / layer_over_hollow).

        Inputs:
            self: GeometryContract instance.

        Outputs:
            bool: ``True`` for hollow-bearing variants.
        """
        return self.variant in ("hollow", "layer_over_hollow")

def _infer_contract(text: str) -> GeometryContract:
    """Brief: Infer family/variant and radii from geo assignments and Physical names.

    Inputs:
        text: Full ``geometry.geo`` source.

    Outputs:
        GeometryContract: Parsed contract.
    """
    vals = parse_geo_assignments(text)
    code = geo_code_text(text).lower()
    volumes = physical_names(text, "Volume")

    has_prism = (
        first_value(vals, ("W",)) is not None
        and first_value(vals, ("H",)) is not None
    )
    sphere_flag = vals.get("geometry_family_sphere", 0.0) > 0.5
    prism_flag = vals.get("geometry_family_prism", 0.0) > 0.5
    cylinder_flag = vals.get("geometry_family_cylinder", 0.0) > 0.5
    has_canonical_sphere_body = first_value(vals, ("R_sphere",)) is not None

    if sphere_flag or has_canonical_sphere_body:
        family = "sphere"
    elif prism_flag or has_prism:
        family = "prism"
    elif cylinder_flag or "cylinder(" in code or "R" in vals:
        family = "cylinder"
    else:
        family = "unknown"

    if family == "sphere":
        domain_radius = domain_radius_m(vals)
        body_outer = metal_sphere_radius_m(vals)
        layer_interface = first_value(vals, ("R_layer_interface",))
        cavity = first_value(vals, ("R_cavity",))
    else:
        domain_radius = first_value(vals, ("R_sph",))
        body_outer = None
        layer_interface = None
        cavity = None

    if vals.get("geometry_variant_layer_over_hollow", 0.0) > 0.5:
        variant = "layer_over_hollow"
    elif vals.get("geometry_variant_layer_over_full", 0.0) > 0.5:
        variant = "layer_over_full"
    elif vals.get("geometry_variant_hollow", 0.0) > 0.5:
        variant = "hollow"
    elif family == "sphere":
        has_inner_air = "inner_air" in volumes
        has_inner_conductor = "inner_material" in volumes
        if has_inner_air and has_inner_conductor and all(
            v is not None for v in (body_outer, layer_interface, cavity)
        ):
            variant = "layer_over_hollow"
        elif has_inner_air and cavity is not None:
            variant = "hollow"
        elif has_inner_conductor and layer_interface is not None:
            variant = "layer_over_full"
        else:
            variant = "full"
    else:
        has_inner_air = "inner_air" in volumes
        has_inner_conductor = "inner_material" in volumes
        if has_inner_air and has_inner_conductor:
            variant = "layer_over_hollow"
        elif has_inner_air:
            variant = "hollow"
        elif has_inner_conductor:
            variant = "layer_over_full"
        else:
            variant = "full"

    if family == "sphere" and variant not in ("hollow", "layer_over_hollow"):
        cavity = None

    return GeometryContract(
        family=family,
        variant=variant,
        domain_radius_m=domain_radius,
        body_outer_radius_m=body_outer,
        layer_interface_radius_m=layer_interface,
        cavity_radius_m=cavity,
    )

def load_geometry_contract(run_dir: Path | str = ".") -> GeometryContract:
    """Brief: Load ``GeometryContract`` from ``geometry.geo`` under a FEM run directory.

    Inputs:
        run_dir: Run root containing ``geometry.geo`` (case-insensitive) [path].

    Outputs:
        GeometryContract: Inferred contract, or empty-text fallback if geo missing.
    """
    root = Path(run_dir)
    for name in ("geometry.geo", "Geometry.geo", "geometry.GEO"):
        path = root / name
        if path.exists():
            return _infer_contract(path.read_text(encoding="utf-8", errors="replace"))
    return _infer_contract("")

def load_mesh_name_ids(run_dir: Path | str = ".") -> Dict[str, int]:
    """Brief: Parse Elmer/Gmsh ``mesh.names`` ``$name = id`` entries into a name-id map.

    Inputs:
        run_dir: Run root; looks under ``geometry/mesh.names`` or ``mesh.names`` [path].

    Outputs:
        dict: Lowercased region name integer GeometryId; empty if file missing.
    """
    root = Path(run_dir)
    candidates = (root / "geometry" / "mesh.names", root / "mesh.names")
    for path in candidates:
        if not path.exists():
            continue
        out: Dict[str, int] = {}
        for raw in path.read_text(encoding="utf-8", errors="replace").splitlines():
            line = raw.strip()
            if not line.startswith("$") or "=" not in line:
                continue
            left, right = line[1:].split("=", 1)
            try:
                out[left.strip().lower()] = int(right.strip())
            except ValueError:
                continue
        return out
    return {}
