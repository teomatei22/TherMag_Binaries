class CurrentDensityVectorMap(BasePostProcessor):

    """Brief: ParaView glyph map of total/source current density inside conductive GeometryIds.

    Inputs:
        working_dir: FEM run directory [path].
        verbose: Verbose logging flag.
        arrow_fraction: Optional glyph arrow size fraction override.

    Outputs:
        CurrentDensityVectorMap: Writes current-density vector-map PNG or MP4.
    """
    def __init__(self, working_dir=None, verbose=False, arrow_fraction=None):
        """Brief: Initialize current-density glyph post-processor with optional arrow fraction.

        Inputs:
            self: Map instance.
            working_dir: FEM run directory [path].
            verbose: Verbose flag.
            arrow_fraction: Optional glyph scale fraction.

        Outputs:
            None: Stores ``arrow_fraction`` on the instance.
        """
        super().__init__(working_dir=working_dir, verbose=verbose)
        self.arrow_fraction = arrow_fraction

    def run(self) -> None:
        """Brief: Threshold conductive cells, glyph ``JSource``, color by abs(J), save PNG/MP4.

        Inputs:
            self: Map instance with VTU inputs.

        Outputs:
            None: Writes ``current_density_vector_map.png`` or ``.mp4``; exits 1 on failure.
        """
        try:
            render_view = CreateView("RenderView")
            self.setup_layout(render_view)
            files = self.vtu_files
            if len(files) < 2:
                print(f"Note: Only {len(files)} file(s). Creating static image.")

            reader = self.make_reader("case_current_vector_map", files)
            reader.PointArrayStatus = ["sourcecurrentx", "sourcecurrenty", "sourcecurrentz"]
            reader.CellArrayStatus = ["GeometryIds"]
            reader.UpdatePipeline()
            is_animation, frame_last = self.animation_frame_last(reader, files)

            threshold = Threshold(registrationName="Threshold_Conductive_J", Input=reader)
            threshold.Scalars = ["CELLS", "GeometryIds"]
            lower_val, upper_val = self.config.conductive_analysis_range()
            self.set_threshold_between(threshold, lower_val, upper_val)
            threshold.UpdatePipeline()
            try:
                metal_cells = int(threshold.GetDataInformation().GetNumberOfCells())
            except Exception:
                metal_cells = -1
            if metal_cells == 0:
                raise RuntimeError(
                    f"Conductive threshold produced no cells "
                )

            calculator = Calculator(registrationName="JSource_Vector", Input=threshold)
            calculator.ResultArrayName = "JSource"
            calculator.Function = (
                "sourcecurrentx*iHat + sourcecurrenty*jHat + sourcecurrentz*kHat"
            )
            calculator.UpdatePipeline()

            surface = ExtractSurface(registrationName="Surface_Conductive_J", Input=threshold)
            surface.UpdatePipeline()

            th_display = Show(surface, render_view, "GeometryRepresentation")
            self.style_conductive_surface(th_display, render_view)

            j_lut = GetColorTransferFunction("JSource")
            scalar_bar = GetScalarBar(j_lut, render_view)
            if scalar_bar:
                scalar_bar.Title = "Current Density [A/m\u00b2]"
                scalar_bar.ComponentTitle = ""
                scalar_bar.LabelFormat = "%.3e"

            glyph_scale = self.vector_glyph_scale_factor(
                "current_density",
                extent_source=surface,
                scale_by_magnitude=False,
                arrow_fraction=self.arrow_fraction,
            )
            sample_cap = self.glyph_max_sample_points("current_density")
            glyph = Glyph(registrationName="Glyph_J", Input=calculator, GlyphType="Arrow")
            self.configure_vector_glyph(
                glyph,
                array_name="JSource",
                scale_factor=glyph_scale,
                scale_by_magnitude=False,
                max_sample_points=sample_cap if sample_cap > 0 else None,
            )

            glyph_display = Show(glyph, render_view, "GeometryRepresentation")
            glyph_display.Representation = "Surface"
            ColorBy(glyph_display, ("POINTS", "JSource", "Magnitude"))
            glyph_display.LookupTable = j_lut
            glyph_display.SetScalarBarVisibility(render_view, True)
            glyph_display.DataAxesGrid.GridAxesVisibility = 0
            hi = self.global_vector_magnitude_hi(
                calculator, "JSource", frame_last
            )
            if hi is not None:
                self.lock_lut_to_range(j_lut, float(hi))
                if scalar_bar:
                    scalar_bar.UseCustomLabels = 1
                    scalar_bar.CustomLabels = self.scientific_tick_values(0.0, float(hi))

            render_view.Update()
            self.setup_camera(render_view)
            if is_animation:
                self.save_animation(
                    "current_density_vector_map.mp4", render_view, [0, frame_last]
                )
            else:
                self.save_image("current_density_vector_map.png", render_view)
            self.cleanup(glyph, surface, calculator, threshold, reader, render_view)

        except Exception as e:
            err_msg = f"ERROR: current_density_vector_map failed: {e}"
            print(err_msg)
            sys.stderr.write(err_msg + "\n")
            sys.exit(1)

def main():
    """Brief: CLI entry that runs ``CurrentDensityVectorMap``.

    Inputs:
        None: Uses process cwd / default VTU discovery.

    Outputs:
        None: Runs the vector-map pipeline.
    """
    CurrentDensityVectorMap(verbose=False).run()
