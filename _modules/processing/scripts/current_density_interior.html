    def run(self, *, write_temperature: bool = True, write_potential: bool = True) -> None:
        """Brief: Sample interior probes over time; write J/T/P CSVs plus log.

        Inputs:
            self: Processor with probe points and VTU inputs.
            write_temperature: Also write ``temperature_interior_data.csv``.
            write_potential: Also write ``potential_interior_data.csv``.

        Outputs:
            None: Writes CSVs under ``out_dir`` and integrity/time-coverage evidence.
        """
        if len(self.probe_points) > 3:
            self.probe_points = list(self.probe_points)[:3]
            print("[WARN] Truncated interior probes to the first three user/automatic points")

        jint_path = os.path.join(self.out_dir, "current_density_interior_data.csv")
        temperature_path = os.path.join(self.out_dir, "temperature_interior_data.csv")
        potential_path = os.path.join(self.out_dir, "potential_interior_data.csv")
        sampling_log = Path(self.cwd) / "logs" / "current_density_sampling.log"
        sampling_log.parent.mkdir(parents=True, exist_ok=True)

        times_written: list[float] = []
        temperature_file = None
        potential_file = None
        temperature_writer = None
        potential_writer = None

        try:
            with open(jint_path, "w", newline="", encoding="utf-8") as jint_file, open(
                sampling_log, "w", encoding="utf-8"
            ) as log_file:
                jint_writer = csv.writer(jint_file)
                jint_writer.writerow(
                    [
                        "time",
                        "point_id",
                        "JOhm_mag",
                        "JOhm_x",
                        "JOhm_y",
                        "JOhm_z",
                        "JSeebeck_mag",
                        "JSeebeck_x",
                        "JSeebeck_y",
                        "JSeebeck_z",
                        "JSource_mag",
                        "JSource_x",
                        "JSource_y",
                        "JSource_z",
                        "x",
                        "y",
                        "z",
                    ]
                )
                log_file.write(
                    "time,point_id,mesh_distance_m,idw_neighbors_used,"
                    "idw_exact_node_hit,idw_support_radius_m,idw_support_mean_distance_m,"
                    "sample_region\n"
                )

                if write_temperature:
                    temperature_file = open(temperature_path, "w", newline="", encoding="utf-8")
                    temperature_writer = csv.writer(temperature_file)
                    temperature_writer.writerow(
                        [
                            "time",
                            "point_id",
                            "Temperature",
                            "x",
                            "y",
                            "z",
                        ]
                    )

                if write_potential:
                    potential_file = open(potential_path, "w", newline="", encoding="utf-8")
                    potential_writer = csv.writer(potential_file)
                    potential_writer.writerow(
                        [
                            "time",
                            "point_id",
                            "Potential",
                            "x",
                            "y",
                            "z",
                        ]
                    )

                if self.is_steady_state:
                    step_files = [(0.0, None)]
                else:
                    step_files = [
                        (self.physical_time_for_vtu_index(idx), path)
                        for idx, path in enumerate(self.vtu_files)
                    ]

                for time_val, vtu_file in step_files:
                    times_written.append(float(time_val))
                    for i, pt in enumerate(self.probe_points):
                        tx, ty, tz = float(pt["x"]), float(pt["y"]), float(pt["z"])
                        result = self.sample_at_point(tx, ty, tz, vtu_file=vtu_file)
                        actual = result["nearest_point"]
                        ax, ay, az = float(actual[0]), float(actual[1]), float(actual[2])
                        johmx, johmy, johmz = (
                            result["JOhmx"],
                            result["JOhmy"],
                            result["JOhmz"],
                        )
                        jseebeckx, jseebecky, jseebeckz = (
                            result["JSeebeckx"],
                            result["JSeebecky"],
                            result["JSeebeckz"],
                        )
                        jsourcex, jsourcey, jsourcez = (
                            result["JSourcex"],
                            result["JSourcey"],
                            result["JSourcez"],
                        )
                        jint_writer.writerow(
                            [
                                time_val,
                                i + 1,
                                format_csv_float((johmx * johmx + johmy * johmy + johmz * johmz) ** 0.5),
                                format_csv_float(johmx),
                                format_csv_float(johmy),
                                format_csv_float(johmz),
                                format_csv_float((jseebeckx * jseebeckx + jseebecky * jseebecky + jseebeckz * jseebeckz) ** 0.5),
                                format_csv_float(jseebeckx),
                                format_csv_float(jseebecky),
                                format_csv_float(jseebeckz),
                                format_csv_float((jsourcex * jsourcex + jsourcey * jsourcey + jsourcez * jsourcez) ** 0.5),
                                format_csv_float(jsourcex),
                                format_csv_float(jsourcey),
                                format_csv_float(jsourcez),
                                format_csv_float(ax),
                                format_csv_float(ay),
                                format_csv_float(az),
                            ]
                        )
                        log_file.write(
                            f"{time_val},{i + 1},{result.get('mesh_distance', '')},"
                            f"{result.get('idw_neighbors_used', '')},"
                            f"{int(bool(result.get('idw_exact_node_hit', False)))},"
                            f"{result.get('idw_support_radius_m', '')},"
                            f"{result.get('idw_support_mean_distance_m', '')},"
                            f"{result.get('sample_region', '')}\n"
                        )
                        if temperature_writer is not None:
                            temperature_writer.writerow(
                                [
                                    time_val,
                                    i + 1,
                                    format_csv_float(result["T"]),
                                    format_csv_float(ax),
                                    format_csv_float(ay),
                                    format_csv_float(az),
                                ]
                            )
                        if potential_writer is not None:
                            potential_writer.writerow(
                                [
                                    time_val,
                                    i + 1,
                                    format_csv_float(result["P"]),
                                    format_csv_float(ax),
                                    format_csv_float(ay),
                                    format_csv_float(az),
                                ]
                            )
        finally:
            if temperature_file is not None:
                temperature_file.close()
            if potential_file is not None:
                potential_file.close()

        print("[OK] Wrote current_density_interior_data.csv")
        if write_temperature:
            print("[OK] Wrote temperature_interior_data.csv")
        if write_potential:
            print("[OK] Wrote potential_interior_data.csv")
        append_run_integrity(
            self.cwd,
            "current_density_interior_fields",
            "required Potential Flux and SourceCurrentX/Y/Z probe fields written; JSeebeck derived as JSource - JOhm (and T/P written when selected)",
        )
        log_time_coverage(
            self.cwd,
            "current_density_interior",
            times_written,
            expected_time_dependent=not self.is_steady_state,
            n_steps=len(self.vtu_files),
        )

def main() -> None:
    """Brief: CLI entry for interior J/T/P probes; enables CSV writes based on selected scripts.

    Inputs:
        None: Parses ``--config`` and ``--run-dir`` from argv.

    Outputs:
        None: Instantiates ``CurrentDensityInterior`` and runs extraction.
    """
    parser = argparse.ArgumentParser(description="Extract interior current density probes")
    parser.add_argument("--config", default="results/processing_config.json")
    parser.add_argument("--run-dir", default=".")
    args = parser.parse_args()

    os.chdir(args.run_dir)
    cfg = Path(args.config)
    probe_points = load_jint_probe_points_from_config(cfg, (SCRIPT_NAME,))

    selected_names: set[str] = set()
    if cfg.exists():
        data = json.loads(cfg.read_text(encoding="utf-8"))
        for script in data.get("scripts", []) or []:
            name = script.get("name", "") if isinstance(script, dict) else str(script)
            if name:
                selected_names.add(str(name).strip())

    processor = CurrentDensityInterior(
        verbose=False,
        probe_points=probe_points,
    )
    processor.run(
        write_temperature="temperature_interior" in selected_names,
        write_potential="potential_interior" in selected_names,
    )




if __name__ == "__main__":
    main()
