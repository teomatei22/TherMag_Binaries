def _format_probe_coord(value: float, decimals: int = 3) -> str:
    """Brief: Format a probe coordinate for sweep plot titles, clamping tiny values to zero.

    Inputs:
        value: Coordinate component [m].
        decimals: Digits after the decimal point.

    Outputs:
        str: Fixed-point coordinate text.
    """
    number = float(value)
    if abs(number) < 0.5 * 10 ** (-decimals):
        return f"{0.0:.{decimals}f}"
    text = f"{number:.{decimals}f}"
    if text.startswith("-0.") and float(text) == 0.0:
        return f"{0.0:.{decimals}f}"
    return text

def _probe_coords_label(px: float, py: float, pz: float) -> str:
    """Brief: Build a (x, y, z) label for sweep plot titles.

        px: Probe x [m].
        py: Probe y [m].
        pz: Probe z [m].

        str: Formatted ``(x, y, z)`` label.
    """
    return f"({_format_probe_coord(px)}, {_format_probe_coord(py)}, {_format_probe_coord(pz)})"

def _row_magnitude(
    row: dict,
    x_col: str,
    y_col: str,
    z_col: str,
    mag_col: Optional[str] = None,
) -> float:
    """Brief: Read |v| from a mag column or compute sqrt(x**2+y**2+z**2) from a CSV row.

    Inputs:
        row: DictReader row of component (and optional mag) fields.
        x_col: X-component column name [field units].
        y_col: Y-component column name [field units].
        z_col: Z-component column name [field units].
        mag_col: Optional magnitude column name [field units].

    Outputs:
        float: Absolute magnitude for ranking/peak tracking [field units].
    """
    if mag_col:
        raw = row.get(mag_col)
        if raw not in (None, ""):
            return abs(float(raw))
    x = float(row[x_col])
    y = float(row[y_col])
    z = float(row[z_col])
    return (x * x + y * y + z * z) ** 0.5

def _csv_has_multiple_times(csv_path: Path) -> bool:
    """Brief: Detect whether a probe CSV contains two or more distinct time values.

    Inputs:
        csv_path: Path to a probe CSV with a ``time`` column [path].

    Outputs:
        bool: ``True`` if at least two distinct times appear.
    """
    if not csv_path.exists():
        return False
    times: set[float] = set()
    with open(csv_path, newline="", encoding="utf-8") as handle:
        reader = csv.DictReader(handle)
        if "time" not in (reader.fieldnames or []):
            return False
        for row in reader:
            try:
                times.add(float(row["time"]))
            except (KeyError, TypeError, ValueError):
                continue
            if len(times) >= 2:
                return True
    return False

def _read_probe_series(
    csv_path: Path,
    x_col: str,
    y_col: str,
    z_col: str,
    mag_col: Optional[str] = None,
) -> Dict[int, Dict[str, float]]:
    """Brief: Collapse a probe CSV to per-point peak |v| and signed extrema of components.

    Inputs:
        csv_path: Probe CSV path [path].
        x_col: X-component header [field units].
        y_col: Y-component header [field units].
        z_col: Z-component header [field units].
        mag_col: Optional magnitude header [field units].

    Outputs:
        dict: ``point_id - {mag,x,y,z,abs_*,px,py,pz}`` peak/extrema summary.
    """
    if not csv_path.exists():
        return {}

    state: Dict[int, Dict[str, Any]] = {}
    with open(csv_path, newline="", encoding="utf-8") as handle:
        reader = csv.DictReader(handle)
        for row in reader:
            try:
                point_id = int(float(row["point_id"]))
                vx = float(row[x_col])
                vy = float(row[y_col])
                vz = float(row[z_col])
                mag = _row_magnitude(row, x_col, y_col, z_col, mag_col=mag_col)
                px = float(row["x"])
                py = float(row["y"])
                pz = float(row["z"])
            except (KeyError, TypeError, ValueError):
                continue

            curr = state.setdefault(
                point_id,
                {
                    "mag": 0.0,
                    "x": 0.0,
                    "y": 0.0,
                    "z": 0.0,
                    "abs_x": 0.0,
                    "abs_y": 0.0,
                    "abs_z": 0.0,
                    "px": px,
                    "py": py,
                    "pz": pz,
                },
            )
            curr["px"], curr["py"], curr["pz"] = px, py, pz
            curr["mag"] = max(float(curr["mag"]), mag)
            if abs(vx) >= float(curr["abs_x"]):
                curr["abs_x"] = abs(vx)
                curr["x"] = vx
            if abs(vy) >= float(curr["abs_y"]):
                curr["abs_y"] = abs(vy)
                curr["y"] = vy
            if abs(vz) >= float(curr["abs_z"]):
                curr["abs_z"] = abs(vz)
                curr["z"] = vz

    return {
        point_id: {
            "mag": float(vals["mag"]),
            "x": float(vals["x"]),
            "y": float(vals["y"]),
            "z": float(vals["z"]),
            "abs_x": float(vals["abs_x"]),
            "abs_y": float(vals["abs_y"]),
            "abs_z": float(vals["abs_z"]),
            "px": float(vals["px"]),
            "py": float(vals["py"]),
            "pz": float(vals["pz"]),
        }
        for point_id, vals in state.items()
    }

def _append_probe_series(
    store: Dict[int, Dict[str, List[Optional[float]]]],
    probe_vals: Dict[int, Dict[str, float]],
    idx: int,
) -> None:
    """Brief: Append one sweep-index of probe peaks into the series.

    Inputs:
        store: Mutable ``point_id - list-valued series``.
        probe_vals: This run's per-point peak dict from ``_read_probe_series``.
        idx: Zero-based sweep run index.

    Outputs:
        None: Mutates ``store`` in place, aligning list lengths to ``idx+1``.
    """
    for point_id, vals in probe_vals.items():
        point_data = store.setdefault(
            point_id,
            {
                "mag": [None] * idx,
                "x": [None] * idx,
                "y": [None] * idx,
                "z": [None] * idx,
                "abs_x": [None] * idx,
                "abs_y": [None] * idx,
                "abs_z": [None] * idx,
                "px": [None] * idx,
                "py": [None] * idx,
                "pz": [None] * idx,
            },
        )
        for key in ("mag", "x", "y", "z", "abs_x", "abs_y", "abs_z", "px", "py", "pz"):
            point_data[key].append(vals[key])
    for point_data in store.values():
        for key in ("mag", "x", "y", "z", "abs_x", "abs_y", "abs_z", "px", "py", "pz"):
            if len(point_data[key]) < idx + 1:
                point_data[key].append(None)

def _probe_location(pdata: Dict[str, List[Optional[float]]]) -> Tuple[float, float, float]:
    """Brief: Recover the first non-None (x,y,z) probe location from a sweep series store entry.

    Inputs:
        pdata: Per-probe lists including ``px``, ``py``, ``pz`` [m].

    Outputs:
        tuple: ``(px, py, pz)`` in metres, or ``(0,0,0)`` if unavailable.
    """
    for i in range(len(pdata["px"])):
        px, py, pz = pdata["px"][i], pdata["py"][i], pdata["pz"][i]
        if px is not None and py is not None and pz is not None:
            return float(px), float(py), float(pz)
    return 0.0, 0.0, 0.0

def _csv_column_names(point_id: int, field: str, *, time_dependent: bool) -> Tuple[str, str, str, str]:
    """Brief: Build (mag, |x|, |y|, |z|) CSV headers for one probe, with max- prefix if time-dependent.

    Inputs:
        point_id: Probe index.
        field: Field symbol (``B`` or ``J``).
        time_dependent: Whether headers use ``max`` peak naming.

    Outputs:
        tuple: Four column header strings for mag and |components|.
    """
    if time_dependent:
        return (
            f"probe{point_id}_max{field}",
            f"probe{point_id}_max|{field}x|",
            f"probe{point_id}_max|{field}y|",
            f"probe{point_id}_max|{field}z|",
        )
    return (
        f"probe{point_id}_{field}",
        f"probe{point_id}_|{field}x|",
        f"probe{point_id}_|{field}y|",
        f"probe{point_id}_|{field}z|",
    )

def main() -> None:
    """Brief: CLI: read sweep manifest, plot B/J vs swept parameter, write ``sweep_data.csv``.

    Inputs:
        None: Parses ``sweep_dir`` from argv; loads ``sweep_manifest.json``.

    Outputs:
        None: Writes PNGs and CSV under ``sweep_dir/sweep_results``.
    """
    parser = argparse.ArgumentParser()
    parser.add_argument("sweep_dir")
    args = parser.parse_args()

    sweep_dir = Path(args.sweep_dir)
    manifest_path = sweep_dir / "sweep_manifest.json"
    if not manifest_path.exists():
        print(f"[ERROR] sweep_manifest.json not found in {sweep_dir}", file=sys.stderr)
        sys.exit(1)

    with open(manifest_path, encoding="utf-8") as f:
        manifest = json.load(f)

    swept_param = manifest["swept_param"]
    observables = manifest.get("observables", [])
    runs = manifest["runs"]
    manifest_dir = manifest_path.parent

    out_dir = sweep_dir / "sweep_results"
    out_dir.mkdir(exist_ok=True)

    param_vals: List[float] = []
    b_data: Dict[int, Dict[str, List[Optional[float]]]] = {}
    j_data: Dict[int, Dict[str, List[Optional[float]]]] = {}
    time_dependent = False

    for idx, entry in enumerate(runs):
        run_entry = (
            entry if isinstance(entry, SweepRunEntry) else SweepRunEntry.from_dict(entry)
        )
        run_dir = run_entry.resolve_run_dir(manifest_dir)
        param_vals.append(
            float(entry["param_val"] if isinstance(entry, dict) else entry.param_val)
        )

        if "B" in observables:
            csv_b = run_dir / "results" / "magnetic_field_data.csv"
            if _csv_has_multiple_times(csv_b):
                time_dependent = True
            probe_vals = _read_probe_series(csv_b, "Bx", "By", "Bz", mag_col="B_mag")
            _append_probe_series(b_data, probe_vals, idx)

        if "J" in observables:
            csv_j = run_dir / "results" / "current_density_interior_data.csv"
            if _csv_has_multiple_times(csv_j):
                time_dependent = True
            probe_vals = _read_probe_series(csv_j, "JSource_x", "JSource_y", "JSource_z", mag_col="JSource_mag")
            _append_probe_series(j_data, probe_vals, idx)

    def _plot(
        x_data: List[float],
        y_data: List[Optional[float]],
        ylabel: str,
        title: str,
        filename: str,
    ) -> None:
        """Brief: Save one Matplotlib sweep curve (parameter vs observable) if any finite points exist.

        Inputs:
            x_data: Swept parameter samples [m].
            y_data: Observable samples (``None`` gaps allowed) [field units].
            ylabel: Y-axis label with units.
            title: Plot title.
            filename: Output PNG basename under ``sweep_results/``.

        Outputs:
            None: Writes PNG or skips/warns if matplotlib unavailable / empty series.
        """
        pairs = [(x, y) for x, y in zip(x_data, y_data) if y is not None]
        if not pairs:
            return
        try:
            import matplotlib

            matplotlib.use("Agg")
            import matplotlib.pyplot as plt
            from matplotlib.ticker import FuncFormatter, MaxNLocator
        except Exception as exc:
            print(f"[WARN] Could not generate {filename}: {exc}")
            return
        xs, ys = zip(*pairs)
        fig, ax = plt.subplots(figsize=(9.5, 5.0))
        ax.plot(xs, ys, marker="o")
        ax.set_xlabel(f"{swept_param} [m]")
        ax.set_ylabel(ylabel)
        ax.set_title(title)
        ax.grid(True)
        ax.xaxis.set_major_locator(MaxNLocator(nbins=6, prune=None))
        formatter = FuncFormatter(lambda value, _pos: truncate_scientific(value))
        ax.xaxis.set_major_formatter(formatter)
        ax.yaxis.set_major_formatter(formatter)
        fig.tight_layout()
        out = out_dir / filename
        fig.savefig(out, dpi=150, bbox_inches="tight")
        plt.close(fig)
        print(f"[OK] Wrote sweep plot image: {out.name}")

    def _labels(
        symbol: str,
        unit: str,
        *,
        magnitude: bool,
        component: Optional[str] = None,
        point_id: int,
        coords: str,
    ) -> Tuple[str, str]:
        """Brief: Build ylabel/title for magnitude or component sweep plots at a probe location.

        Inputs:
            symbol: Field symbol (``B`` or ``J``).
            unit: SI unit string (e.g. ``T``, ``A/m²``).
            magnitude: If true, label |symbol|; else a component.
            component: Component letter when not magnitude (``x``/``y``/``z``).
            point_id: Probe index for the title.
            coords: Preformatted ``(x,y,z)`` string [m].

        Outputs:
            tuple: ``(ylabel, title)`` strings including max-|symbol|(t) wording when time-dependent.
        """
        if magnitude:
            core = f"|{symbol}|" if not time_dependent else f"max |{symbol}(t)|"
        else:
            assert component is not None
            core = f"{symbol}{component}" if not time_dependent else f"max {symbol}{component}(t)"
        ylabel = f"{core} [{unit}]"
        title = f"{ylabel} vs {swept_param} at P{point_id} {coords}"
        return ylabel, title

    if "B" in observables:
        for point_id in sorted(b_data):
            pdata = b_data[point_id]
            coords = _probe_coords_label(*_probe_location(pdata))
            ylabel, title = _labels("B", "T", magnitude=True, point_id=point_id, coords=coords)
            _plot(
                param_vals,
                pdata["mag"],
                ylabel,
                title,
                f"sweep_B_probe_{point_id}_vs_{swept_param}.png",
            )
            for comp, key in (("x", "x"), ("y", "y"), ("z", "z")):
                ylabel, title = _labels(
                    "B",
                    "T",
                    magnitude=False,
                    component=comp,
                    point_id=point_id,
                    coords=coords,
                )
                _plot(
                    param_vals,
                    pdata[key],
                    ylabel,
                    title,
                    f"sweep_B{comp}_probe_{point_id}_vs_{swept_param}.png",
                )

    if "J" in observables:
        for point_id in sorted(j_data):
            pdata = j_data[point_id]
            coords = _probe_coords_label(*_probe_location(pdata))
            ylabel, title = _labels(
                "J", "A/m\u00b2", magnitude=True, point_id=point_id, coords=coords
            )
            _plot(
                param_vals,
                pdata["mag"],
                ylabel,
                title,
                f"sweep_J_probe_{point_id}_vs_{swept_param}.png",
            )
            for comp, key in (("x", "x"), ("y", "y"), ("z", "z")):
                ylabel, title = _labels(
                    "J",
                    "A/m\u00b2",
                    magnitude=False,
                    component=comp,
                    point_id=point_id,
                    coords=coords,
                )
                _plot(
                    param_vals,
                    pdata[key],
                    ylabel,
                    title,
                    f"sweep_J{comp}_probe_{point_id}_vs_{swept_param}.png",
                )

    csv_out = out_dir / "sweep_data.csv"
    with open(csv_out, "w", newline="", encoding="utf-8") as handle:
        cols = [swept_param]
        if "B" in observables:
            for point_id in sorted(b_data):
                cols += list(_csv_column_names(point_id, "B", time_dependent=time_dependent))
        if "J" in observables:
            for point_id in sorted(j_data):
                cols += list(_csv_column_names(point_id, "J", time_dependent=time_dependent))
        writer = csv.DictWriter(handle, fieldnames=cols)
        writer.writeheader()
        for i, pv in enumerate(param_vals):
            row: dict = {swept_param: format_csv_float(pv)}
            if "B" in observables:
                for point_id, pdata in b_data.items():
                    mag_c, x_c, y_c, z_c = _csv_column_names(
                        point_id, "B", time_dependent=time_dependent
                    )
                    row[mag_c] = (
                        format_csv_float(pdata["mag"][i]) if pdata["mag"][i] is not None else ""
                    )
                    row[x_c] = (
                        format_csv_float(pdata["abs_x"][i])
                        if pdata["abs_x"][i] is not None
                        else ""
                    )
                    row[y_c] = (
                        format_csv_float(pdata["abs_y"][i])
                        if pdata["abs_y"][i] is not None
                        else ""
                    )
                    row[z_c] = (
                        format_csv_float(pdata["abs_z"][i])
                        if pdata["abs_z"][i] is not None
                        else ""
                    )
            if "J" in observables:
                for point_id, pdata in j_data.items():
                    mag_c, x_c, y_c, z_c = _csv_column_names(
                        point_id, "J", time_dependent=time_dependent
                    )
                    row[mag_c] = (
                        format_csv_float(pdata["mag"][i]) if pdata["mag"][i] is not None else ""
                    )
                    row[x_c] = (
                        format_csv_float(pdata["abs_x"][i])
                        if pdata["abs_x"][i] is not None
                        else ""
                    )
                    row[y_c] = (
                        format_csv_float(pdata["abs_y"][i])
                        if pdata["abs_y"][i] is not None
                        else ""
                    )
                    row[z_c] = (
                        format_csv_float(pdata["abs_z"][i])
                        if pdata["abs_z"][i] is not None
                        else ""
                    )
            writer.writerow(row)
    print(f"[OK] Wrote sweep summary CSV: {csv_out.name}")

    print("-" * 50)
    print("Plot generation complete")




if __name__ == "__main__":
    main()
