    @staticmethod
    def _parse_commanded_temperature_range(case_text: str) -> tuple[float, float] | None:
        """Brief: Parse cold/hot commanded temperatures from Elmer ``case.sif`` MATC Time Variable block.

        Inputs:
            case_text: Full ``case.sif`` source text.

        Outputs:
            tuple | None: ``(T_cold, T_hot)`` [K] sorted, or ``None`` if not found.
        """
        match = re.search(
            r'(?is)Temperature\s*=\s*Variable\s+Time.*?'
            r'MATC\s*"([\d.eE+-]+)\s*\+\s*\(([\d.eE+-]+)\s*-\s*([\d.eE+-]+)\)',
            case_text,
        )
        if not match:
            return None
        try:
            t_cold = float(match.group(1))
            t_hot = float(match.group(2))
            if t_hot < t_cold:
                t_cold, t_hot = t_hot, t_cold
            return t_cold, t_hot
        except ValueError:
            return None

    def _commanded_temperature_range(self) -> tuple[float, float] | None:
        """Brief: Load commanded temperature range from ``case.sif`` in the run directory.

        Inputs:
            self: Map instance with ``cwd``.

        Outputs:
            tuple | None: ``(T_cold, T_hot)`` [K], or ``None`` if file missing/unparseable.
        """
        case_path = Path(self.cwd) / "case.sif"
        if not case_path.is_file():
            return None
        try:
            return self._parse_commanded_temperature_range(
                case_path.read_text(encoding="utf-8")
            )
        except OSError:
            return None

    def run(self) -> None:
        """Brief: Threshold conductive cells, color by temperature with locked LUT, save PNG/MP4.

        Inputs:
            self: Map instance with VTU inputs.

        Outputs:
            None: Writes ``temperature_map.png`` or ``.mp4``; exits 1 on failure.
        """
        try:
            render_view = CreateView('RenderView')
            self.setup_layout(render_view)
            if len(self.vtu_files) < 2:
                print(f"Note: Only {len(self.vtu_files)} file(s). Creating static image.")
                vtu_files_to_use = [self.vtu_files[0]]
                is_animation = False
            else:
                vtu_files_to_use = self.vtu_files
                is_animation = True

            reader = self.make_reader('case_temperature_map', vtu_files_to_use)
            reader.TimeArray = 'None'
            reader.PointArrayStatus = ['temperature']
            reader.CellArrayStatus = ['GeometryIds']
            reader.UpdatePipeline()

            frame_last = 0
            if is_animation:
                anim = GetAnimationScene()
                anim.UpdateAnimationUsingDataTimeSteps()
                reader.TimeArray = 'None'
                try:
                    tk = GetTimeKeeper()
                    ts = list(getattr(tk, 'TimestepValues', []) or [])
                    if ts:
                        frame_last = len(ts) - 1
                    else:
                        anim.PlayMode = 'Sequence'
                        anim.NumberOfFrames = max(1, len(vtu_files_to_use))
                        frame_last = anim.NumberOfFrames - 1
                except Exception:
                    anim.PlayMode = 'Sequence'
                    anim.NumberOfFrames = max(1, len(vtu_files_to_use))
                    frame_last = anim.NumberOfFrames - 1

            threshold = Threshold(registrationName='Threshold_Conductive_T', Input=reader)
            threshold.Scalars = ['CELLS', 'GeometryIds']

            lower_val, upper_val = self.config.conductive_analysis_range()

            self.set_threshold_between(threshold, lower_val, upper_val)

            threshold.UpdatePipeline()

            th_display = Show(threshold, render_view, 'UnstructuredGridRepresentation')
            th_display.Representation = 'Surface'
            th_display.DataAxesGrid.GridAxesVisibility = 0

            render_view.ResetCamera()
            render_view.Update()

            ColorBy(th_display, ('POINTS', 'temperature'))

            lut = GetColorTransferFunction('temperature')
            lut.ApplyPreset('Cool to Warm (Extended)', True)

            if is_animation and len(vtu_files_to_use) > 1:
                anim = GetAnimationScene()
                global_min, global_max = None, None
                try:
                    anim.GoToFirst()
                    for _ in range(frame_last + 1):
                        reader.UpdatePipeline()
                        threshold.UpdatePipeline()
                        render_view.Update()
                        data_info = threshold.GetDataInformation()
                        point_data = data_info.GetPointDataInformation()
                        temp_array = point_data.GetArrayInformation('temperature')
                        if temp_array:
                            lo, hi = temp_array.GetComponentRange(0)
                            if global_min is None or lo < global_min:
                                global_min = lo
                            if global_max is None or hi > global_max:
                                global_max = hi
                        anim.GoToNext()

                    commanded = self._commanded_temperature_range()
                    if commanded is not None:
                        t_cold, t_hot = commanded
                        if global_min is None or global_max is None:
                            global_min, global_max = t_cold, t_hot
                        else:
                            global_min = min(global_min, t_cold)
                            global_max = max(global_max, t_hot)

                    if global_min is not None and global_max is not None and global_min < global_max:
                        print(
                            f"  Using temperature map range: [{global_min:.2f}, {global_max:.2f}] K"
                        )
                        lut.RescaleTransferFunction(global_min, global_max)

                    anim.GoToFirst()
                except Exception:
                    pass

            th_display.SetScalarBarVisibility(render_view, True)

            scalar_bar = GetScalarBar(lut, render_view)
            if scalar_bar:
                scalar_bar.Title = 'Temperature [K]'
                scalar_bar.ComponentTitle = ''

            render_view.Update()
            self.setup_camera(render_view)

            if not is_animation:
                self.save_image("temperature_map.png", render_view)
            else:
                self.save_animation(
                    "temperature_map.mp4", render_view, [0, frame_last], fps=1
                )

            self.cleanup(threshold, reader, render_view)

        except Exception as e:
            err_msg = f"ERROR: temperature_map failed: {e}"
            print(err_msg)
            sys.stderr.write(err_msg + "\n")
            sys.exit(1)
