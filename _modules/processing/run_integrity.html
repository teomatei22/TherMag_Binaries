def _log_path(cwd: str | Path) -> Path:
    """Brief: Resolve and ensure the run-integrity log file path under the run directory.

    Inputs:
        cwd: FEM run working directory containing ``logs/`` [path].

    Outputs:
        Path: Absolute path to ``logs/run_integrity.log`` (parent created if missing).
    """
    path = Path(cwd) / "logs" / "run_integrity.log"
    path.parent.mkdir(parents=True, exist_ok=True)
    return path

def append_run_integrity(cwd: str | Path, check_name: str, detail: str) -> None:
    """Brief: Append one timestamped integrity check line to the run log.

    Inputs:
        cwd: FEM run working directory [path].
        check_name: Short check identifier (e.g. ``solver_completion``).
        detail: Free-text evidence for the check.

    Outputs:
        None: Appends a UTC-stamped TSV line to ``run_integrity.log``.
    """
    stamp = datetime.now(timezone.utc).strftime("%Y-%m-%dT%H:%M:%SZ")
    line = f"{stamp}\t{check_name}\t{detail}\n"
    with _log_path(cwd).open("a", encoding="utf-8") as handle:
        handle.write(line)

def log_time_coverage(
    cwd: str | Path,
    source: str,
    times: Iterable[float],
    *,
    expected_time_dependent: bool,
    n_steps: int,
) -> None:
    """Brief: Record and validate probe/time-series coverage against VTU steps.

    Inputs:
        cwd: FEM run working directory [path].
        source: Probe or script name used as the check key prefix.
        times: Sampled physical times written by the probe [s].
        expected_time_dependent: ``True`` when the run has multiple VTU steps.
        n_steps: Number of VTU timesteps expected for unique-time coverage.

    Outputs:
        None: Logs coverage detail; raises ``RuntimeError`` if coverage is insufficient.
    """
    values = [float(t) for t in times]
    if expected_time_dependent:
        detail = (
            f"{source}: wrote {len(values)} time sample(s) "
            f"across {n_steps} VTU step(s); "
            f"range=[{min(values):.6g}, {max(values):.6g}]"
            if values
            else f"{source}: expected time_dependent coverage but wrote no times"
        )
        if not values:
            raise RuntimeError(detail)
        if n_steps > 0 and len(values) < n_steps:
            # one time per step is enough; multi-probe writes repeat the same times
            unique = sorted(set(round(v, 12) for v in values))
            if len(unique) < n_steps:
                raise RuntimeError(
                    f"{source}: expected {n_steps} unique times, got {len(unique)}"
                )
    else:
        detail = f"{source}: steady-state coverage with {len(values)} sample row(s)"
    append_run_integrity(cwd, f"{source}_time_coverage", detail)

def log_solver_completion(cwd: str | Path, *, return_code: int | None, all_done: bool) -> None:
    """Brief: Record Elmer/solver exit evidence and fail on unclean completion.

    Inputs:
        cwd: FEM run working directory [path].
        return_code: Process return code, or ``None`` if unknown.
        all_done: Whether the solver emitted an ALL DONE / success marker.

    Outputs:
        None: Logs completion; raises ``RuntimeError`` when exit is unclean.
    """
    detail = f"return_code={return_code!s}; ALL_DONE={'yes' if all_done else 'no'}"
    append_run_integrity(cwd, "solver_completion", detail)
    if return_code not in (None, 0) and not all_done:
        raise RuntimeError(f"Solver did not complete cleanly ({detail})")

def log_geometry_interpretation(cwd: str | Path, family: str, variant: str) -> None:
    """Brief: Record interpreted geometry family and hollow/layer variant for the run.

    Inputs:
        cwd: FEM run working directory [path].
        family: Geometry family (``cylinder``, ``prism``, ``sphere``).
        variant: Topology variant (``full``, ``hollow``, ``layer_over_*``).

    Outputs:
        None: Appends a ``geometry_interpretation`` integrity line.
    """
    append_run_integrity(
        cwd,
        "geometry_interpretation",
        f"family={family}; variant={variant}",
    )
