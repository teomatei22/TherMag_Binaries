    def __init__(self, working_dir: Optional[str] = None, verbose: bool = False):
        """Brief: Discover VTU/PVTU results, geometry, and model layout for a FEM run.

        Inputs:
            working_dir: FEM run directory with ``geometry/`` and ``geometry.geo``.
            verbose: When ``True``, log discovered file type and model.

        Outputs:
            None: Sets ``self.geometry``, ``self.config``, ``self.vtu_files``, and I/O dirs.
        """
        self.cwd = working_dir or os.getcwd()
        self.in_dir = os.path.join(self.cwd, "geometry")
        self.out_dir = os.path.join(self.cwd, "results")
        self.verbose = verbose
        self.pp_settings = load_post_processing_settings(self.cwd)

        os.makedirs(self.out_dir, exist_ok=True)

        geo_file = os.path.join(self.cwd, "geometry.geo")
        self.geometry = GeometryParams.from_geo_file(geo_file, self.pp_settings)

        try:
            from run_integrity import log_geometry_interpretation
        except ImportError:
            log_geometry_interpretation = None
        if log_geometry_interpretation is not None:
            log_geometry_interpretation(
                self.cwd,
                self.geometry.geometry_type,
                getattr(self.geometry, "variant", "full"),
            )

        self.vtu_files = self._discover_files()
        self.is_steady_state = len(self.vtu_files) == 1
        self.is_time_dependent = len(self.vtu_files) > 1

        self.config = ModelConfig.detect(self.cwd, self.pp_settings)
        self._timestep_size: Optional[float] = None
        self._output_intervals: Optional[int] = None

        self._log_setup()

    def physical_time_for_vtu_index(self, vtu_index: int) -> float:
        """Brief: Map a VTU file index to physical simulation time from ``case.sif``.

        Inputs:
            vtu_index: Zero-based index among discovered VTU/PVTU files.

        Outputs:
            float: Physical time ``(index + 1) * output_intervals * timestep_size`` [s].
        """
        if self.is_steady_state:
            return 0.0
        if self._timestep_size is None or self._output_intervals is None:
            timestep_size = 1.0
            output_intervals = 1
            case_path = Path(self.cwd) / "case.sif"
            if case_path.is_file():
                try:
                    sif_text = case_path.read_text(encoding="utf-8")
                    size_match = re.search(
                        r"(?mi)^\s*Timestep Sizes\s*=\s*([^\s!]+)",
                        sif_text,
                    )
                    if size_match:
                        timestep_size = float(size_match.group(1))

                    exec_intervals = self._parse_result_output_exec_interval(sif_text)
                    if exec_intervals is not None:
                        output_intervals = exec_intervals
                    else:
                        output_match = re.search(
                            r"(?mi)^\s*Output Intervals\s*=\s*([^\s!]+)",
                            sif_text,
                        )
                        if output_match:
                            output_intervals = int(float(output_match.group(1)))
                except (OSError, ValueError):
                    pass
            self._timestep_size = timestep_size
            self._output_intervals = output_intervals
        return float(vtu_index + 1) * self._output_intervals * self._timestep_size

    @staticmethod
    def _parse_result_output_exec_interval(sif_text: str) -> Optional[int]:
        """Brief: Read ``Exec Intervals`` from the ResultOutput / VTU solver block in a SIF.

        Inputs:
            sif_text: Full Elmer ``case.sif`` text.

        Outputs:
            int | None: Exec interval for VTU writes, or ``None`` if not found.
        """
        for block_match in re.finditer(
            r"(?mis)^\s*Solver\s+\d+\s*$(.*?)^\s*End\s*$",
            sif_text,
        ):
            block = block_match.group(1)
            is_result_output = bool(
                re.search(r"resultoutput", block, flags=re.IGNORECASE)
                or re.search(
                    r"(?mi)^\s*Output\s+Format\s*=\s*[\"']?vtu",
                    block,
                )
            )
            if not is_result_output:
                continue
            exec_match = re.search(
                r"(?mi)^\s*Exec\s+Intervals\s*=\s*([0-9]+)",
                block,
            )
            if exec_match:
                return int(exec_match.group(1))
        return None

    def _discover_files(self) -> List[str]:
        """Brief: Discover sorted VTU or PVTU result files under ``geometry/``.

        Inputs:
            N/A: Scans ``self.in_dir``.

        Outputs:
            list[str]: Sorted ``.pvtu`` paths if present, else ``.vtu`` paths.
        """
        pvtu_files = sorted(glob.glob(os.path.join(self.in_dir, "*.pvtu")))
        vtu_files = sorted(glob.glob(os.path.join(self.in_dir, "*.vtu")))

        if pvtu_files:
            return pvtu_files
        elif vtu_files:
            return vtu_files
        else:
            raise RuntimeError(f"No .vtu or .pvtu files found in: {self.in_dir}")

    def make_reader(self, name: str, files: Any) -> Any:
        """Brief: Open VTU/PVTU data with configured point/cell arrays for FEM fields.

        Inputs:
            name: ParaView source display name.
            files: Single path or list of VTU/PVTU paths.

        Outputs:
            Any: ParaView reader proxy with pipeline updated.
        """
        src = OpenDataFile(files)
        try:
            RenameSource(name, src)
        except Exception:
            pass

        if hasattr(src, 'TimeArray'):
            src.TimeArray = 'None'
        if hasattr(src, 'PointArrayStatus'):
            rd = self.pp_settings["reader"]
            wanted = rd["wanted_point_arrays"]
            src.PointArrayStatus = [a for a in wanted if a in getattr(src, "PointArrayInfo", wanted)]
        if hasattr(src, "CellArrayStatus"):
            rd = self.pp_settings["reader"]
            cells = rd["cell_arrays"]
            cell_info = getattr(src, "CellArrayInfo", None)
            if cell_info:
                src.CellArrayStatus = [a for a in cells if a in cell_info]
            else:
                src.CellArrayStatus = cells

        src.UpdatePipeline()
        return src

    def get_geometry_glyph_fraction(self, field: str, override: Optional[float] = None) -> float:
        """Brief: Resolve glyph arrow length as a fraction of conductor extent for a field.

        Inputs:
            field: Field key (``current_density`` or ``magnetic``).
            override: Optional explicit fraction replacing the geometry default.

        if override is not None:
            return float(override)
        g = self.pp_settings["glyph"]
        geom = str(self.geometry.geometry_type).lower()
        key = f"{field}_arrow_fraction_of_extent_{geom}"
        if key not in g:
            raise KeyError(f"Missing glyph setting '{key}' in post_processing_defaults.json")
        return float(g[key])

    def glyph_extent_span(self, extent_source: Any = None) -> float:
        """Brief: Characteristic spatial span for glyph scaling from data bounds or geometry.

        Inputs:
            extent_source: Optional ParaView source whose bounds define the span [m].

        Outputs:
            float: Max axis span of bounds, else ``geometry.characteristic_extent()`` [m].
        """
        if extent_source is not None:
            try:
                extent_source.UpdatePipeline()
                bounds = extent_source.GetDataInformation().GetBounds()
                span = max(
                    bounds[1] - bounds[0],
                    bounds[3] - bounds[2],
                    bounds[5] - bounds[4],
                )
                if span > 0.0 and np.isfinite(span):
                    return float(span)
            except Exception:
                pass
        return float(self.geometry.characteristic_extent())

    def glyph_max_sample_points(self, field: str) -> int:
        """Brief: Maximum glyph sample points from ``post_processing_defaults.json``.

        Inputs:
            field: Field key used to look up ``glyph.{field}_max_sample_points``.

        Outputs:
            int: Non-negative max sample points (``0`` means all points).

        Raises:
            KeyError: If the glyph sample-points key is missing from defaults JSON.
            ValueError: If the configured value is not a valid integer.
        """
        key = f"{field}_max_sample_points"
        glyph = self.pp_settings.get("glyph")
        if not isinstance(glyph, dict) or key not in glyph:
            raise KeyError(f"Missing glyph setting '{key}' in post_processing_defaults.json")
        try:
            return max(0, int(glyph[key]))
        except (TypeError, ValueError) as exc:
            raise ValueError(
                f"Invalid glyph setting '{key}' in post_processing_defaults.json: {glyph[key]!r}"
            ) from exc

    def vector_glyph_scale_factor(
        self,
        field: str,
        *,
        extent_source: Any = None,
        magnitude_source: Any = None,
        array_name: Optional[str] = None,
        magnitude_ref: Optional[float] = None,
        scale_by_magnitude: bool = False,
        arrow_fraction: Optional[float] = None,
    ) -> float:
        """Brief: Compute ParaView glyph ScaleFactor from extent fraction and optional |V| ref.

        Inputs:
            field: Field key for glyph fraction lookup (``current_density`` / ``magnetic``).
            extent_source: Optional source for spatial span [m].
            magnitude_source: Optional source for 99th-percentile magnitude when scaling by |V|.
            array_name: Point-data vector array name when deriving magnitude ref.
            magnitude_ref: Optional explicit magnitude reference [field units].
            scale_by_magnitude: When ``True``, divide span x fraction by magnitude ref.
            arrow_fraction: Optional override of geometry glyph fraction.

        Outputs:
            float: Glyph ``ScaleFactor`` so peak arrows match fraction of extent [m / field].
        """
        frac = self.get_geometry_glyph_fraction(field, arrow_fraction)
        span = self.glyph_extent_span(extent_source if extent_source is not None else magnitude_source)
        if span <= 1.0e-30:
            span = float(self.geometry.characteristic_extent())
        if not scale_by_magnitude:
            return float(frac) * float(span)

        ref = None
        if magnitude_ref is not None and np.isfinite(magnitude_ref) and float(magnitude_ref) > 0.0:
            ref = float(magnitude_ref)
        elif magnitude_source is not None and array_name and servermanager is not None and ns is not None:
            magnitude_source.UpdatePipeline()
            arr = servermanager.Fetch(magnitude_source).GetPointData().GetArray(array_name)
            if arr is None:
                raise RuntimeError(f"Point array '{array_name}' not found for glyph scaling")
            vec = ns.vtk_to_numpy(arr)
            if not vec.size:
                raise RuntimeError(f"Point array '{array_name}' is empty for glyph scaling")
            mags = np.linalg.norm(vec, axis=1) if vec.ndim == 2 else np.abs(vec)
            ref = float(np.percentile(mags, 99))
            if not np.isfinite(ref) or ref <= 0.0:
                ref = float(np.max(mags))
        if ref is None or not np.isfinite(ref) or ref <= 0.0:
            ref = 1.0
        return float(frac) * float(span) / ref

    def configure_vector_glyph(
        self,
        glyph: Any,
        *,
        array_name: str,
        scale_factor: float,
        scale_by_magnitude: bool,
        max_sample_points: Optional[int] = None,
    ) -> None:
        """Brief: Configure orientation, scale mode, and spatial sampling on a Glyph filter.

        Inputs:
            glyph: ParaView Glyph filter proxy.
            array_name: Point-data vector array for orientation (and optional magnitude).
            scale_factor: Glyph ``ScaleFactor`` from ``vector_glyph_scale_factor`` [m / field].
            scale_by_magnitude: When ``True``, scale arrows by vector magnitude.
            max_sample_points: Cap for uniform sampling; ``<=0`` uses all points.

        Outputs:
            None: Updates ``glyph`` pipeline settings.
        """
        glyph.OrientationArray = ["POINTS", array_name]
        if scale_by_magnitude:
            glyph.ScaleArray = ["POINTS", array_name]
            glyph.VectorScaleMode = "Scale by Magnitude"
        else:
            try:
                glyph.ScaleArray = ["POINTS", ""]
            except Exception:
                glyph.ScaleArray = "No scale array"
        glyph.ScaleFactor = float(scale_factor)

        target = int(max_sample_points or 0)
        if target <= 0:
            glyph.GlyphMode = "All Points"
            glyph.UpdatePipeline()
            return

        for mode in (
            "Uniform Spatial Distribution (Volume Sampling)",
            "Uniform Spatial Distribution (Bounds Based)",
            "Every Nth Point",
        ):
            try:
                glyph.GlyphMode = mode
                break
            except Exception:
                continue
        else:
            glyph.GlyphMode = "All Points"
            glyph.UpdatePipeline()
            return
        try:
            glyph.MaximumNumberOfSamplePoints = target
        except Exception:
            pass
        glyph.UpdatePipeline()

    def animation_frame_last(self, reader: Any, files: List[str]) -> tuple[bool, int]:
        """Brief: Configure animation for multi-step VTU series and return the last frame index.

        Inputs:
            reader: ParaView reader proxy for the time series.
            files: Ordered VTU/PVTU paths constituting the series.

        Outputs:
            tuple[bool, int]: ``(is_animated, last_frame_index)``; steady runs return ``(False, 0)``.
        """
        if len(files) < 2:
            try:
                reader.TimeArray = "None"
            except Exception:
                pass
            return False, 0
        anim = GetAnimationScene()
        anim.UpdateAnimationUsingDataTimeSteps()
        reader.TimeArray = "None"
        try:
            ts = list(getattr(GetTimeKeeper(), "TimestepValues", []) or [])
            if ts:
                return True, len(ts) - 1
        except Exception:
            pass
        anim.PlayMode = "Sequence"
        anim.NumberOfFrames = max(1, len(files))
        return True, anim.NumberOfFrames - 1

    def global_vector_magnitude_hi(
        self,
        source: Any,
        array_name: str,
        frame_last: int,
    ) -> Optional[float]:
        """Brief: Scan all animation frames for the global peak vector magnitude.

        Inputs:
            source: ParaView source exposing the vector point array.
            array_name: Point-data vector array name (e.g. current density or B).
            frame_last: Inclusive last animation frame index.

        Outputs:
            float | None: Global max magnitude [field units], or ``None`` if unavailable.
        """
        if servermanager is None or ns is None:
            return None
        anim = GetAnimationScene()
        global_hi: Optional[float] = None
        times: list[float] = []
        try:
            times = [float(t) for t in (getattr(GetTimeKeeper(), "TimestepValues", []) or [])]
        except Exception:
            pass
        try:
            anim.GoToFirst()
            for i in range(int(frame_last) + 1):
                try:
                    if times and i < len(times):
                        source.UpdatePipeline(time=times[i])
                    else:
                        source.UpdatePipeline()
                    arr = servermanager.Fetch(source).GetPointData().GetArray(array_name)
                    if arr is not None:
                        vec = ns.vtk_to_numpy(arr)
                        if vec.size:
                            mags = np.linalg.norm(vec, axis=1) if vec.ndim == 2 else np.abs(vec)
                            peak = float(np.max(mags))
                            if np.isfinite(peak):
                                global_hi = peak if global_hi is None else max(global_hi, peak)
                except Exception:
                    pass
                anim.GoToNext()
            anim.GoToFirst()
        except Exception:
            try:
                anim.GoToFirst()
            except Exception:
                pass
        return global_hi

    def style_conductive_surface(self, display: Any, render_view: Any) -> None:
        """Brief: Color conductive bodies by GeometryIds categories (outer/inner material).

        Inputs:
            display: ParaView display representation for the conductive surface.
            render_view: Active render view (scalar bar hidden).

        Outputs:
            None: Applies categorical GeometryIds coloring for materials.

        display.Opacity = 1.0
        display.DataAxesGrid.GridAxesVisibility = 0

        ColorBy(display, ("CELLS", "GeometryIds"))
        geometry_lut = GetColorTransferFunction("GeometryIds")

        indexed_colors = [0.18, 0.55, 0.90]

            indexed_colors.extend([0.93, 0.44, 0.20])

        geometry_lut.Annotations = annotations
        geometry_lut.IndexedColors = indexed_colors
        display.LookupTable = geometry_lut
        display.SetScalarBarVisibility(render_view, False)
        try:
            HideScalarBarIfNotNeeded(geometry_lut, render_view)
        except Exception:
            pass

    def setup_camera(self, render_view: Any) -> None:
        """Brief: Place an isometric parallel camera scaled to the conductor extent.

        Inputs:
            render_view: Active ParaView render view.

        Outputs:
            None: Sets camera position, focal point, view-up, and parallel scale.
        """
        geom_scale = max(self.geometry.characteristic_extent(), 1.0e-9)
        isometric = 0.57735026919
        cam_distance = 3.0 * geom_scale
        render_view.CameraPosition = [
            cam_distance * isometric,
            cam_distance * isometric,
            cam_distance * isometric,
        ]
        render_view.CameraFocalPoint = [0.0, 0.0, 0.0]
        render_view.CameraViewUp = [0.0, 0.0, 1.0]
        try:
            render_view.CameraParallelProjection = 1
        except Exception:
            pass
        render_view.CameraParallelScale = 1.15 * geom_scale

    def save_image(self, filename: str, view_or_layout: Any) -> None:
        """Brief: Save a still screenshot of the current view into ``results/``.

        Inputs:
            filename: Output image name under ``self.out_dir`` (e.g. ``temperature_map.png``).
            view_or_layout: ParaView view or layout to capture.

        Outputs:
            None: Writes the screenshot file; prints OK/error status.
        """
        out_path = os.path.join(self.out_dir, filename)
        label = self._describe_output(filename)
        if self.verbose:
            print(f"Saving {label} screenshot")
        vr = [1560, 788]
        try:
            SaveScreenshot(filename=out_path, viewOrLayout=view_or_layout, ImageResolution=list(vr))
            print(f"[OK] Wrote {label} screenshot")
        except Exception as e:
            print(f"Error saving screenshot {out_path}: {e}")

            try:
                SaveScreenshot(filename=out_path, viewOrLayout=view_or_layout)
                print(f"[OK] Wrote {label} screenshot (fallback)")
            except Exception as e2:
                print(f"Fallback failed: {e2}")

        """Brief: Export a time-series animation (AVI/MP4) of the FEM field visualization.

        Inputs:
            filename: Desired animation name under ``self.out_dir`` (``.mp4`` or ``.avi``).
            view_or_layout: ParaView view or layout to capture.
            frame_window: Inclusive ``[first, last]`` animation frame indices.
            fps: Optional frames per second; defaults to 1.

        Outputs:
            None: Writes a valid animation file when possible; warns on failure.
        """
        label = self._describe_output(filename)
        actual_fps = int(fps) if fps is not None else 1
        min_valid_bytes = 1000
        if self.verbose:
            print(f"Saving {label} animation at {actual_fps} FPS")

        def check_success(path: str) -> bool:
            """Brief: Verify an animation file exists and exceeds a minimum byte size.

            Inputs:
                path: Candidate animation file path.

            Outputs:
                bool: ``True`` when the file looks successfully written.
            """
            if os.path.exists(path) and os.path.getsize(path) > min_valid_bytes:
                return True
            if self.verbose:
                print(f"  File not found or too small: {path}", flush=True)
            return False

        if filename.endswith(".mp4"):
            avi_name = filename[:-4] + ".avi"
            if sys.platform == "win32":
                candidates = [filename, avi_name]
            else:
                candidates = [avi_name, filename]
                print(
                    "[INFO] Writing animation as AVI "
                )
        else:
            candidates = [filename]

        vr = [1560, 788]
        for candidate in candidates:
            out_path = os.path.join(self.out_dir, candidate)
            try:
                SaveAnimation(
                    filename=out_path,
                    viewOrLayout=view_or_layout,
                    ImageResolution=list(vr),
                    FrameWindow=frame_window,
                    FrameRate=actual_fps,
                )
                if check_success(out_path):
                    suffix = " (AVI)" if candidate.endswith(".avi") and filename.endswith(".mp4") else ""
                    print(f"[OK] Wrote {label} animation{suffix}")
                    return
            except Exception as e:
                print(f"Error saving {candidate}: {e}")
        print(f"Warning: Failed to create animation for {filename}")
