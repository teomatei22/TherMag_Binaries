def _object(data: Mapping[str, Any], key: str, path: str) -> Mapping[str, Any]:
    """Brief: Require a nested JSON object field during defaults validation.

def _string_list(data: Mapping[str, Any], key: str, path: str) -> tuple[str, ...]:
    """Brief: Require a non-empty-string list JSON field.

    Inputs:
        data: Parent mapping being validated.
        key: Child key that must be a list of non-empty strings.
        path: Dot-path prefix used in error messages.

    Outputs:
        tuple[str, ...]: Immutable list of strings.

        raise ValueError(f"{path}.{key} must be a list of non-empty strings")
    return tuple(value)

def _number_list(data: Mapping[str, Any], key: str, path: str) -> tuple[float, ...]:
    """Brief: Require a non-empty numeric list JSON field.

    Inputs:
        data: Parent mapping being validated.
        key: Child key that must be a non-empty list of numbers.
        path: Dot-path prefix used in error messages.

    Outputs:
        tuple[float, ...]: Immutable list of floats.
    """
    value = data.get(key)
    if not isinstance(value, list) or not value:
        raise ValueError(f"{path}.{key} must be a non-empty list of numbers")
    out: list[float] = []
    for item in value:
        if isinstance(item, bool) or not isinstance(item, (int, float)):
            raise ValueError(f"{path}.{key} must be a non-empty list of numbers")
        out.append(float(item))
    return tuple(out)

def _parse_geometry_detection(data: Mapping[str, Any]) -> GeometryDetectionDefaults:
    """Brief: Validate and parse the ``geometry_detection`` defaults section.

    Inputs:
        data: ``geometry_detection`` object from defaults JSON.

    Outputs:
        GeometryDetectionDefaults: Parsed detection heuristics.
    """
    path = "geometry_detection"
    default_type = data.get("default_geometry_type")
    if default_type not in {"cylinder", "prism", "sphere"}:
        raise ValueError(f"{path}.default_geometry_type must be cylinder, prism, or sphere")
    return GeometryDetectionDefaults(
        sphere_path_substrings=_string_list(data, "sphere_path_substrings", path),
        sphere_content_markers=_string_list(data, "sphere_content_markers", path),
        default_geometry_type=str(default_type),
    )

def _parse_glyph(data: Mapping[str, Any]) -> GlyphDefaults:
    """Brief: Validate per-family J and B glyph arrow fractions of extent.

    Inputs:
        data: ``glyph`` object from defaults JSON.

    Outputs:
        GlyphDefaults: Parsed current-density and magnetic glyph fractions.
    """
    fractions: dict[str, float] = {}
    mag_fractions: dict[str, float] = {}
    for family in ("cylinder", "prism", "sphere"):
        cd_key = f"current_density_arrow_fraction_of_extent_{family}"
        mag_key = f"magnetic_arrow_fraction_of_extent_{family}"
        fractions[family] = _validate_positive(_number(data, cd_key, "glyph"), f"glyph.{cd_key}")
        mag_fractions[family] = _validate_positive(_number(data, mag_key, "glyph"), f"glyph.{mag_key}")
    return GlyphDefaults(
        current_density_arrow_fraction_of_extent=MappingProxyType(fractions),
        magnetic_arrow_fraction_of_extent=MappingProxyType(mag_fractions),
    )

def _parse_jint(data: Mapping[str, Any]) -> JintDefaults:
    """Brief: Validate interior J-probe dimensionless point multipliers.

    Inputs:
        data: ``extraction.jint_defaults`` object from defaults JSON.

    Outputs:
        JintDefaults: Parsed interior probe multipliers.
    """
    path = "extraction.jint_defaults"
    raw_points = data.get("point_multipliers")
    if not isinstance(raw_points, list) or not raw_points:
        raise ValueError(f"{path}.point_multipliers must be a non-empty list")
    points: list[Mapping[str, float]] = []
    for index, item in enumerate(raw_points):
        if not isinstance(item, dict):
            raise ValueError(f"{path}.point_multipliers[{index}] must be an object")
        points.append(
            MappingProxyType(
                {
                    "x_l": _number(item, "x_l", f"{path}.point_multipliers[{index}]"),
                    "y": _number(item, "y", f"{path}.point_multipliers[{index}]"),
                    "z_bext_scale": _number(
                        item, "z_bext_scale", f"{path}.point_multipliers[{index}]"
                    ),
                }
            )
        )
    return JintDefaults(point_multipliers=tuple(points))

def _parse_bext(data: Mapping[str, Any]) -> BextDefaults:
    """Brief: Validate exterior B trusted-band scales and midplane probe directions.

    Inputs:
        data: ``extraction.bext_defaults`` object from defaults JSON.

    Outputs:
        BextDefaults: Parsed magnetic probe-band configuration.
    """
    path = "extraction.bext_defaults"
    raw_dirs = data.get("point_directions")
    if not isinstance(raw_dirs, list) or not raw_dirs:
        raise ValueError(f"{path}.point_directions must be a non-empty list")
    directions: list[tuple[float, float, float]] = []
    for index, item in enumerate(raw_dirs):
        if not isinstance(item, Sequence) or isinstance(item, (str, bytes)) or len(item) < 3:
            raise ValueError(f"{path}.point_directions[{index}] must have 3 numbers")
        directions.append((float(item[0]), float(item[1]), float(item[2])))
    fractions = _number_list(data, "probe_band_fractions", path)
    band_fraction = _number(data, "probe_band_fraction", path)
    if not 0.0 <= band_fraction <= 1.0:
        raise ValueError(f"{path}.probe_band_fraction must be between 0 and 1")
    for frac in fractions:
        if not 0.0 <= frac <= 1.0:
            raise ValueError(f"{path}.probe_band_fractions values must be between 0 and 1")
    r_inf = _validate_positive(_number(data, "R_inf_trusted", path), f"{path}.R_inf_trusted")
    r_sup = _validate_positive(_number(data, "R_sup_trusted", path), f"{path}.R_sup_trusted")
    if r_sup <= r_inf:
        raise ValueError(f"{path}.R_sup_trusted must be greater than R_inf_trusted")
    return BextDefaults(
        probe_band_fraction=band_fraction,
        probe_band_fractions=fractions,
        r_inf_trusted=r_inf,
        r_sup_trusted=r_sup,
        point_directions=tuple(directions),
    )

def _parse_model_layouts(data: Mapping[str, Any]) -> Mapping[str, Mapping[str, Any]]:
    """Brief: Validate per-``model_type`` conductive/air body-id layouts.

    Inputs:
        data: ``model_layouts`` object from defaults JSON.

    Outputs:
        Mapping[str, Mapping[str, Any]]: Immutable layout map keyed by model type.
    """
    path = "model_layouts"
    layouts: dict[str, Mapping[str, Any]] = {}
    for name, raw in data.items():
        if not isinstance(name, str) or not name:
            raise ValueError(f"{path} keys must be non-empty strings")
        if not isinstance(raw, dict):
            raise ValueError(f"{path}.{name} must be an object")
        item_path = f"{path}.{name}"
        layouts[name] = MappingProxyType(
            {
                "outer_material_id": _integer(raw, "outer_material_id", item_path),
                "outer_air_id": _integer(raw, "outer_air_id", item_path),
                "inner_air_id": _optional_integer(raw, "inner_air_id", item_path),
                "inner_material_id": _optional_integer(raw, "inner_material_id", item_path),
            }
        )
    if not layouts:
        raise ValueError(f"{path} must contain at least one layout")
    return MappingProxyType(layouts)

def _parse_extraction(data: Mapping[str, Any]) -> ExtractionDefaults:
    """Brief: Validate IDW tolerances and nested Jint/Bext extraction defaults.

    Inputs:
        data: ``extraction`` object from defaults JSON.

    Outputs:
        ExtractionDefaults: Parsed IDW and probe extraction settings.
    """
    path = "extraction"
    neighbors = _integer(data, "idw_neighbors", path)
    if neighbors <= 0:
        raise ValueError(f"{path}.idw_neighbors must be greater than zero")
    return ExtractionDefaults(
        idw_neighbors=neighbors,
        idw_epsilon=_validate_positive(_number(data, "idw_epsilon", path), f"{path}.idw_epsilon"),
        on_node_epsilon=_validate_positive(
            _number(data, "on_node_epsilon", path), f"{path}.on_node_epsilon"
        ),
        jint=_parse_jint(_object(data, "jint_defaults", path)),
        bext=_parse_bext(_object(data, "bext_defaults", path)),
    )

def _parse_defaults(data: Mapping[str, Any]) -> ProcessingDefaults:
    """Brief: Validate the full post-processing defaults JSON object.

    Inputs:
        data: Root object from ``post_processing_defaults.json``.

    Outputs:
        ProcessingDefaults: Fully typed and validated defaults tree.
    """
    if "geometry_params" in data:
        raise ValueError(
            "geometry_params is not a processing default; geometry dimensions come from geometry.geo"
        )
    detection = _parse_geometry_detection(_object(data, "geometry_detection", "defaults"))
    glyph = _parse_glyph(_object(data, "glyph", "defaults"))
    reader_data = _object(data, "reader", "defaults")
    reader = ReaderDefaults(
        wanted_point_arrays=_string_list(reader_data, "wanted_point_arrays", "reader"),
        cell_arrays=_string_list(reader_data, "cell_arrays", "reader"),
    )
    extraction = _parse_extraction(_object(data, "extraction", "defaults"))
    model = _object(data, "model_config_defaults", "defaults")
    return ProcessingDefaults(
        geometry_detection=detection,
        glyph=glyph,
        reader=reader,
        extraction=extraction,
        model_config=ModelConfigDefaults(
            outer_material_id=_integer(model, "outer_material_id", "model_config_defaults"),
            outer_air_id=_integer(model, "outer_air_id", "model_config_defaults"),
            inner_air_id=_optional_integer(model, "inner_air_id", "model_config_defaults"),
            inner_material_id=_optional_integer(model, "inner_material_id", "model_config_defaults"),
        ),
        model_layouts=_parse_model_layouts(_object(data, "model_layouts", "defaults")),
        raw=MappingProxyType(copy.deepcopy(dict(data))),
    )

@cache
def _load_defaults_file(path: Path) -> ProcessingDefaults:
    """Brief: Load and cache validated defaults from a JSON file path.

    Inputs:
        path: Absolute path to ``post_processing_defaults.json``.

    Outputs:
        ProcessingDefaults: Cached parsed defaults for ``path``.
    """
    if not path.is_file():
        raise FileNotFoundError(f"Missing defaults JSON: {path}")
    with path.open("r", encoding="utf-8") as handle:
        data = json.load(handle)
    if not isinstance(data, dict):
        raise ValueError(f"Defaults JSON must contain an object: {path}")
    return _parse_defaults(data)

def deep_merge_dict(base: dict[str, Any], overlay: dict[str, Any]) -> dict[str, Any]:
    """Brief: Recursively merge overlay settings onto a deep-copied base dict.

    Inputs:
        base: Base settings mapping (not mutated).
        overlay: Override mapping whose nested dicts merge recursively.

    Outputs:
        dict[str, Any]: New merged mapping.
    """
    result = copy.deepcopy(base)
    for key, val in overlay.items():
        if key in result and isinstance(result[key], dict) and isinstance(val, dict):
            result[key] = deep_merge_dict(result[key], val)
        else:
            result[key] = copy.deepcopy(val)
    return result

def load_post_processing_settings(
    cwd: str | None = None,
    path: Path | None = None,
) -> dict[str, Any]:
    """Brief: Load defaults and optionally merge run-local ``processing_config.json`` overlays.

    Inputs:
        cwd: Optional FEM run directory; when set, merges ``results/processing_config.json``.
        path: Optional explicit defaults JSON path.

    Outputs:
        dict[str, Any]: Mutable post-processing settings for ParaView scripts.
    """
    data = load_processing_defaults(path).to_dict()
    if not cwd:
        return data
    config_path = Path(cwd) / "results" / "processing_config.json"
    if not config_path.is_file():
        return data
    with config_path.open("r", encoding="utf-8") as handle:
        run_cfg = json.load(handle)
    extra = run_cfg.get("post_processing_settings")
    if not isinstance(extra, dict):
        return data
    return deep_merge_dict(data, extra)

def truncate_scientific(value: float, decimals: int = MANTISSA_DECIMALS) -> str:
    """Brief: Format a float in truncated scientific notation for CSV/plot ticks.

    Inputs:
        value: Numeric value to format [any].
        decimals: Mantissa digits kept after truncation.

    Outputs:
        str: Scientific string with truncated (not rounded) mantissa.
    """
    number = float(value)
    if not math.isfinite(number):
        return str(number).lower()
    if number == 0.0:
        return f"0.{'0' * decimals}e+00"

    mantissa_text, exponent_text = f"{number:.17e}".split("e")
    exponent = int(exponent_text)
    mantissa = float(mantissa_text)
    factor = 10**decimals
    mantissa = math.trunc(mantissa * factor) / factor
    if mantissa == 0.0:
        return f"0.{'0' * decimals}e+00"
    return f"{mantissa:.{decimals}f}e{exponent:+03d}"

def scientific_tick_values(
    lower: float, upper: float, count: int = 5
) -> list[float]:
    """Brief: Build evenly spaced, truncated tick values for scientific color bars.

    Inputs:
        lower: Color-bar / axis lower bound [field units].
        upper: Color-bar / axis upper bound [field units].
        count: Desired number of ticks (minimum 2 for a span).

    Outputs:
        list[float]: Deduplicated truncated tick positions.
    """
    if count < 2 or not math.isfinite(lower) or not math.isfinite(upper):
        return []
    if upper <= lower:
        return [truncate_float(lower)]
    step = (upper - lower) / (count - 1)
    values = [truncate_float(lower + index * step) for index in range(count)]
    return list(dict.fromkeys(values))
