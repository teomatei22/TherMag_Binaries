        configs_dir: Directory containing ``*.json`` material definition files.

    Outputs:
        MaterialRepository: Repository for loading and saving material JSON files.
    """

    def __init__(self, configs_dir: Path):
        """Brief: Bind the repository to a materials configuration directory.

        Inputs:
            configs_dir: Directory containing ``*.json`` material definition files.

        Outputs:
            None.
        """
        self.configs_dir = configs_dir

    def load_all(self) -> list["Material"]:
        """Brief: Load every material definition found under ``configs_dir``.

        Inputs:
            N/A: Reads ``*.json`` files from ``self.configs_dir``.

        Outputs:
            list[Material]: Materials from all JSON files (empty if directory missing).
        """
        materials: list[Material] = []
        if not self.configs_dir.exists():
            return materials
        for json_file in sorted(self.configs_dir.glob("*.json")):
            materials.extend(self._load_file(json_file))
        return materials

    def _load_file(self, json_file: Path) -> list["Material"]:
        """Brief: Parse one JSON file as a single material or a list of materials.

        Inputs:
            json_file: Path to a material JSON file (object or array root).

        Outputs:
            list[Material]: One or more materials deserialized from the file.
        """
        from .database import Material

        with open(json_file, encoding="utf-8") as handle:
            data = json.load(handle)
        if isinstance(data, list):
            return [Material.from_dict(item, json_file) for item in data]
        return [Material.from_dict(data, json_file)]

    def save(self, material: "Material", file_path: Path) -> None:
        """Brief: Persist a material to JSON, creating parent directories as needed.

        Inputs:
            material: Material instance to write.
            file_path: Destination JSON path.

        Outputs:
            None: Writes the file and updates the material's ``source_file`` via ``save``.
        """
        file_path.parent.mkdir(parents=True, exist_ok=True)
        material.save(file_path)
