class MaterialProperties:
    """Brief: Scalar, tensor, and tabulated thermo-electromagnetic material properties.

    Inputs:
        density: Mass density [kg/m^3].
        heat_capacity: Specific heat capacity [J/(kg K)].
        heat_conductivity: Isotropic thermal conductivity [W/(m K)].
        electric_conductivity: Isotropic electric conductivity [S/m].
        relative_permittivity: Relative permittivity (dimensionless).
        relative_permeability: Relative permeability (dimensionless).
        seebeck_coefficient: Isotropic Seebeck coefficient [V/K].
        heat_conductivity_tensor: Optional anisotropic thermal conductivity tensor.
        electric_conductivity_tensor: Optional anisotropic electric conductivity tensor.
        seebeck_coefficient_tensor: Optional anisotropic Seebeck coefficient tensor.
        heat_conductivity_table: Optional T-dependent thermal conductivity table.
        electric_conductivity_table: Optional T-dependent electric conductivity table.
        hb_curve: Optional magnetic H-B hysteresis curve table.

    Outputs:
        MaterialProperties: Scalar, tensor, and tabulated constitutive property values.
    """

    density: float = 1.0
    heat_capacity: float = 1.0
    heat_conductivity: float = 1.0
    electric_conductivity: float = 0.0
    relative_permittivity: float = 1.0
    relative_permeability: float = 1.0
    seebeck_coefficient: float = 0.0

    heat_conductivity_tensor: list[float] | None = None
    electric_conductivity_tensor: list[float] | None = None
    seebeck_coefficient_tensor: list[float] | None = None

    heat_conductivity_table: list[list[float]] | None = None
    electric_conductivity_table: list[list[float]] | None = None

    hb_curve: list[list[float]] | None = None

    def to_dict(self) -> dict[str, Any]:
        """Brief: Serialize material properties, omitting unset tensor/table fields.

        Inputs:
            N/A: Uses fields on ``self``.

        Outputs:
            dict[str, Any]: JSON-serializable properties mapping.
        """
        result = {
            "density": self.density,
            "heat_capacity": self.heat_capacity,
            "heat_conductivity": self.heat_conductivity,
            "electric_conductivity": self.electric_conductivity,
            "relative_permittivity": self.relative_permittivity,
            "relative_permeability": self.relative_permeability,
            "seebeck_coefficient": self.seebeck_coefficient,
        }

        if self.heat_conductivity_tensor is not None:
            result["heat_conductivity_tensor"] = self.heat_conductivity_tensor
        if self.electric_conductivity_tensor is not None:
            result["electric_conductivity_tensor"] = self.electric_conductivity_tensor
        if self.seebeck_coefficient_tensor is not None:
            result["seebeck_coefficient_tensor"] = self.seebeck_coefficient_tensor
        if self.heat_conductivity_table is not None:
            result["heat_conductivity_table"] = self.heat_conductivity_table
        if self.electric_conductivity_table is not None:
            result["electric_conductivity_table"] = self.electric_conductivity_table
        if self.hb_curve is not None:
            result["hb_curve"] = self.hb_curve

        return result

class Material:
    """Brief: Named FEM material with classification flags and constitutive properties.

    Inputs:
        name: Material display name used as the registry key.
        classification: Model classification flags (anisotropy, T-dependence, etc.).
        properties: Scalar, tensor, and tabulated constitutive properties.
        description: Optional human-readable material description.
        source_file: Optional path to the JSON file this material was loaded from.

    Outputs:
        Material: Named FEM material definition with classification and properties.
    """

    name: str
    classification: MaterialClassification = field(default_factory=MaterialClassification)
    properties: MaterialProperties = field(default_factory=MaterialProperties)
    description: str = ""
    source_file: Path | None = None

    def save(self, file_path: Path | None = None) -> None:
        """Brief: Write the material JSON to disk and record ``source_file``.

        Inputs:
            file_path: Destination path; falls back to ``self.source_file`` when omitted.

        Outputs:
            None: Writes JSON and sets ``self.source_file`` to the written path.
        """
        path = file_path or self.source_file
        if not path:
            raise ValueError("No file path specified")

        with open(path, "w", encoding="utf-8") as f:
            json.dump(self.to_dict(), f, indent=2)

        self.source_file = path

class MaterialDatabase:
    """Brief: In-memory registry of builtin and user-configured FEM materials.

    Inputs:
        configs_dir: Optional extra directory of user JSON materials overlaid on builtins.

    Outputs:
        MaterialDatabase: Name-keyed in-memory registry of loaded FEM materials.
    """

    def __init__(self, configs_dir: Path | None = None):
        """Brief: Load builtin materials, then optionally overlay a user configs directory.

        Inputs:
            configs_dir: Optional extra directory of JSON materials; skipped if same as
                the package builtin directory.

        Outputs:
            None: Populates ``self._materials`` and stores ``self.configs_dir``.
        """
        self._materials: dict[str, Material] = {}
        self.configs_dir = configs_dir
        self._load_builtin_materials()
        self._load_config_files()



[docs]
    def _load_builtin_materials(self) -> None:
        """Brief: Load packaged default materials into the registry.

        Inputs:
            N/A: Reads JSON from the materials package directory.

        Outputs:
            None: Inserts builtin materials into ``self._materials``.
        """
        default_dir = Path(__file__).parent
        repo = MaterialRepository(default_dir)
        for material in repo.load_all():
            self._materials[material.name] = material





[docs]
    def _load_config_files(self) -> None:
        """Brief: Overlay user material JSON files from ``configs_dir`` when distinct.

        Inputs:
            N/A: Uses ``self.configs_dir`` if set and different from the package dir.

        Outputs:
            None: Merges user materials into ``self._materials`` by name.
        """
        if not self.configs_dir or not self.configs_dir.exists():
            return

        default_dir = Path(__file__).parent
        if self.configs_dir.resolve() == default_dir.resolve():
            return

        repo = MaterialRepository(self.configs_dir)
        for material in repo.load_all():
            self._materials[material.name] = material

    def get_materials_by_classification(
        self,
        anisotropic: bool | None = None,
        temperature_dependent: bool | None = None,
        hysteresis: bool | None = None,
    ) -> list[Material]:
        """Brief: Filter materials by optional classification flag criteria.

        Inputs:
            anisotropic: Required anisotropic flag, or ``None`` to ignore.
            temperature_dependent: Required T-dependent flag, or ``None`` to ignore.
            hysteresis: Required hysteresis flag, or ``None`` to ignore.

        Outputs:
            list[Material]: Materials matching all provided criteria.
        """
        result = []
        for material in self._materials.values():
            if anisotropic is not None and material.classification.anisotropic != anisotropic:
                continue
            if (
                temperature_dependent is not None
                and material.classification.temperature_dependent != temperature_dependent
            ):
                continue
            if hysteresis is not None and material.classification.hysteresis != hysteresis:
                continue
            result.append(material)
        return result

    def save_material(self, name: str, file_path: Path | None = None) -> None:
        """Brief: Persist a registered material to a JSON file under the configs directory.

        Inputs:
            name: Registered material name to save.
            file_path: Optional destination; defaults to ``configs_dir/<name>.json``.

        Outputs:
            None: Writes the material JSON via ``MaterialRepository.save``.
        """
        material = self.get_material(name)
        if not material:
            raise ValueError(f"Material not found: {name}")

        if file_path is None:
            if self.configs_dir is None:
                raise ValueError("No material configuration directory is configured")
            file_path = self.configs_dir / f"{name.lower().replace(' ', '_')}.json"

        MaterialRepository(file_path.parent).save(material, file_path)

def get_material_database(configs_dir: Path | None = None) -> MaterialDatabase:
    """Brief: Return the process-wide material database, recreating when ``configs_dir`` is set.

    Inputs:
        configs_dir: Optional user materials directory; forces rebuild when not ``None``.

    Outputs:
        MaterialDatabase: Shared database instance.
    """
    global _database
    if _database is None or configs_dir is not None:
        _database = MaterialDatabase(configs_dir)
    return _database
