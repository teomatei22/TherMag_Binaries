from __future__ import annotations

from dataclasses import dataclass
from typing import Any

from config.defaults import SweepDefaults, load_application_defaults

@dataclass(frozen=True)
class SweepConfig:
    """Brief: Coefficient range and observables for sweeping a geometric parameter in FEM runs.

    Inputs:
        swept_parameter: Name of the geometry parameter being swept (e.g. ``L``).
        observables: Ordered observable names to extract per run.
        lower_coefficient: Lower multiplicative coefficient for the sweep.
        upper_coefficient: Upper multiplicative coefficient for the sweep.
        coefficient_step: Step between consecutive coefficients.

    Outputs:
        SweepConfig: Immutable geometric parameter-sweep configuration.
    """

    swept_parameter: str
    observables: tuple[str, ...]
    lower_coefficient: float
    upper_coefficient: float
    coefficient_step: float

    @classmethod
    def from_dict(
        cls,
        data: Mapping[str, Any],
        *,
        defaults: SweepDefaults | None = None,
    ) -> SweepConfig:
        """Brief: Build sweep config from a mapping, filling gaps from application defaults.

        Inputs:
            data: Mapping with optional ``swept_parameter``, ``observables``, and coefficients.
            defaults: Optional ``SweepDefaults``; loads application defaults when omitted.

        Outputs:
            SweepConfig: Immutable sweep configuration.
        """
        defaults = defaults or load_application_defaults().sweep
        return cls(
            swept_parameter=str(data.get("swept_parameter", "L")),
            observables=tuple(str(value) for value in data.get("observables", ())),
            lower_coefficient=float(data.get("lower_coefficient", defaults.lower_coefficient)),
            upper_coefficient=float(data.get("upper_coefficient", defaults.upper_coefficient)),
            coefficient_step=float(data.get("coefficient_step", defaults.coefficient_step)),
        )
