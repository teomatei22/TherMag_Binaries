def _integration_steps(total_time: int, defaults: ApplicationDefaults) -> int:
    """Brief: Compute transient integration steps from total time and timestep size.

    Inputs:
        total_time: Simulated physical duration [s].
        defaults: Application defaults providing ``driver_runtime.timestep_size`` [s].

    Outputs:
        int: Number of integration steps (at least 1).
    """
    return max(1, int(math.ceil(total_time / defaults.driver_runtime.timestep_size)))

def _default_integration_steps() -> int:
    """Brief: Default integration-step count from packaged time-dependent defaults.

    Inputs:
        N/A: Loads application defaults.

    Outputs:
        int: Integration steps for the default total time.
    """
    defaults = _defaults()
    return _integration_steps(defaults.time_dependent.total_time, defaults)

class TemperatureConfig:
    """Brief: Hot/cold thermal BC temperatures and optional driving frequency [K, Hz].

    Inputs:
        t_hot: Hot-side boundary temperature [K].
        t_cold: Cold-side boundary temperature [K].
        frequency: Optional thermal-driver frequency [Hz]; ``None`` for steady.

    Outputs:
        TemperatureConfig: Immutable thermal BC temperatures and optional frequency.
    """

    t_hot: float = field(default_factory=lambda: _defaults().temperature.t_hot)
    t_cold: float = field(default_factory=lambda: _defaults().temperature.t_cold)
    frequency: float | None = None



[docs]
    @classmethod
    def from_dict(
        cls,
        data: dict[str, Any],
        *,
        defaults: ApplicationDefaults | None = None,
        driver_period: float | None = None,
    ) -> "TemperatureConfig":
        """Brief: Build temperature config; frequency is inverse of driver period when set.

        Inputs:
            data: Mapping with optional ``t_hot`` and ``t_cold`` [K].
            defaults: Optional application defaults for missing temperatures.
            driver_period: Optional thermal-driver period [s]; sets ``frequency = 1/period``.

        Outputs:
            TemperatureConfig: Immutable temperature configuration.
        """
        defaults = defaults or _defaults()
        if driver_period is not None and driver_period > 0:
            frequency: float | None = 1.0 / float(driver_period)
        else:
            frequency = None
        return cls(
            t_hot=float(data.get("t_hot", defaults.temperature.t_hot)),
            t_cold=float(data.get("t_cold", defaults.temperature.t_cold)),
            frequency=frequency,
        )

    @staticmethod
    def _parse_potential(raw: Any, *, enabled: bool) -> float | None:
        """Brief: Parse a potential value, treating free/empty markers as unset.

        Inputs:
            raw: Raw potential (number, ``None``, or free/null string).
            enabled: Whether the corresponding electric BC is active.

        Outputs:
            float | None: Potential in volts, or ``None`` when disabled or free.
        """
        if not enabled:
            return None
        if raw is None:
            return None
        if isinstance(raw, str) and raw.strip().lower() in ("", "free", "null"):
            return None
        return float(raw)

    @staticmethod
    def _serialize_potential(value: float | None) -> float | str:
        """Brief: Encode an optional potential for JSON (``free`` when unset).

        Inputs:
            value: Potential in volts, or ``None`` for a free boundary.

        Outputs:
            float | str: Numeric potential or the string ``free``.
        """
        return value if value is not None else "free"

    @classmethod
    def from_dict(
        cls,
        data: dict[str, Any],
        *,
        defaults: ApplicationDefaults | None = None,
    ) -> "ElectricBCConfig":
        """Brief: Deserialize electric BC enable flags and potentials.

        Inputs:
            data: Mapping with ``bc1_/bc2_enabled`` and ``bc1_/bc2_potential``.
            defaults: Optional application defaults for missing fields.

        Outputs:
            ElectricBCConfig: Immutable electric BC configuration.
        """
        defaults = defaults or _defaults()
        bc1_enabled = bool(data.get("bc1_enabled", defaults.electric_bc.bc1_enabled))
        bc2_enabled = bool(data.get("bc2_enabled", defaults.electric_bc.bc2_enabled))
        bc1_raw = data.get("bc1_potential", defaults.electric_bc.bc1_potential)
        bc2_raw = data.get("bc2_potential", defaults.electric_bc.bc2_potential)
        return cls(
            bc1_enabled=bc1_enabled,
            bc1_potential=cls._parse_potential(bc1_raw, enabled=bc1_enabled),
            bc2_enabled=bc2_enabled,
            bc2_potential=cls._parse_potential(bc2_raw, enabled=bc2_enabled),
        )

    def from_dict(
        cls,
        data: dict[str, Any],
        requested_processes: int | None = None,
        *,
        defaults: ApplicationDefaults | None = None,
    ) -> "SolverOptions":
        """Brief: Deserialize solver options with optional process-count override.

        Inputs:

            requested_processes: Optional override for default process count.
            defaults: Optional application defaults for missing fields.

        Outputs:
            SolverOptions: Immutable solver options.

            max_partitions=int(data.get("max_partitions", defaults.solver.max_partitions)),
            mesh_timeout=int(data.get("mesh_timeout", defaults.solver.mesh_timeout)),
            solve_timeout=int(data.get("solve_timeout", defaults.solver.solve_timeout)),
            post_timeout=int(data.get("post_timeout", defaults.solver.post_timeout)),
        )

@dataclass(frozen=True)
class SimulationConfig:
    """Brief: Full FEM case config: materials, thermal/electric BCs, driver, and solver.

    Inputs:
        base_type: Geometry/base case type label (e.g. ``full``).
        driver_type: Thermal driver kind (``steady``, ``harmonic``, ``sawtooth``).
        simulation_type: ``steady`` or ``time_dependent`` simulation mode.
        outer_material_model: Constitutive model name for the outer material.
        outer_material_name: Display name of the outer material.
        is_twolayer: Whether an inner material layer is present.
        inner_material_model: Optional constitutive model for the inner material.
        inner_material_name: Optional display name of the inner material.
        temperature: Thermal BC temperatures and optional frequency.
        electric_bc: Electric Dirichlet BC configuration.
        time_dependent: Transient settings when ``simulation_type`` is time-dependent.
        solver: MPI process counts and stage timeouts.

    Outputs:
        SimulationConfig: Immutable full FEM case configuration.
    """

    base_type: str = "full"
    driver_type: str = "steady"
    simulation_type: str = STEADY_SIMULATION_TYPE
    outer_material_model: str = "simple"
    outer_material_name: str = ""
    is_twolayer: bool = False
    inner_material_model: str | None = None
    inner_material_name: str | None = None
    temperature: TemperatureConfig = field(default_factory=TemperatureConfig)
    electric_bc: ElectricBCConfig = field(default_factory=ElectricBCConfig)
    time_dependent: TimeDependentConfig | None = None
    solver: SolverOptions = field(default_factory=SolverOptions)

    @classmethod
    def from_dict(
        cls,
        data: dict[str, Any],
        requested_processes: int | None = None,
        *,
        defaults: ApplicationDefaults | None = None,
    ) -> "SimulationConfig":
        """Brief: Deserialize a complete simulation configuration from a mapping.

        Inputs:
            data: Nested mapping of materials, temperature, electric BC, time-dependent, and solver sections.
            requested_processes: Optional override forwarded to ``SolverOptions``.
            defaults: Optional application defaults for nested sections.

        Outputs:
            SimulationConfig: Immutable simulation configuration.
        """
        defaults = defaults or _defaults()
        time_dependent_data = data.get("time_dependent_params")
        time_dependent = (
            TimeDependentConfig.from_dict(time_dependent_data, defaults=defaults)
            if time_dependent_data
            else None
        )
        return cls(
            base_type=str(data.get("base_type", "full")),
            driver_type=str(data.get("driver_type", "steady")),
            simulation_type=str(data.get("simulation_type", STEADY_SIMULATION_TYPE)),
            outer_material_model=str(data.get("outer_material_model", "simple")),
            outer_material_name=str(data.get("outer_material_name", "")),
            is_twolayer=bool(data.get("is_twolayer", False)),
            inner_material_model=data.get("inner_material_model"),
            inner_material_name=data.get("inner_material_name"),
            temperature=TemperatureConfig.from_dict(
                data.get("temperature", {}),
                defaults=defaults,
                driver_period=(
                    time_dependent.driver_period if time_dependent is not None else None
                ),
            ),
            electric_bc=ElectricBCConfig.from_dict(
                data.get("electric_bc", {}),
                defaults=defaults,
            ),
            time_dependent=time_dependent,
            solver=SolverOptions.from_dict(
                data.get("solver", {}),
                requested_processes=requested_processes,
                defaults=defaults,
            ),
        )

    def to_dict(self) -> dict[str, Any]:
        """Brief: Serialize the simulation configuration for manifest persistence.

        Inputs:
            N/A: Uses fields on ``self``.

        Outputs:
            dict[str, Any]: JSON-serializable simulation mapping.
        """
        result: dict[str, Any] = {
            "base_type": self.base_type,
            "driver_type": self.driver_type,
            "simulation_type": self.simulation_type,
            "outer_material_model": self.outer_material_model,
            "outer_material_name": self.outer_material_name,
            "is_twolayer": self.is_twolayer,
            "temperature": self.temperature.to_dict(),
            "electric_bc": self.electric_bc.to_dict(),
            "solver": self.solver.to_dict(),
        }
        if self.inner_material_model is not None:
            result["inner_material_model"] = self.inner_material_model
        if self.inner_material_name is not None:
            result["inner_material_name"] = self.inner_material_name
        if self.time_dependent is not None:
            result["time_dependent_params"] = self.time_dependent.to_dict()
        return result
