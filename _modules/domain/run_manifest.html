class RunStatus(StrEnum):
    """Brief: Lifecycle status of a FEM run from creation through post-processing.

    Inputs:
        None: Enum members ``CREATED``, ``GENERATED``, ``MESHED``, ``SOLVED``,
            ``POST_PROCESSED``, ``FAILED``, ``CANCELLED``.

    Outputs:
        RunStatus: String enum value describing the current run lifecycle stage.
    """

    CREATED = "created"
    GENERATED = "generated"
    MESHED = "meshed"
    SOLVED = "solved"
    POST_PROCESSED = "post_processed"
    FAILED = "failed"
    CANCELLED = "cancelled"

class ResultsConfig:
    """Brief: Post-processing scripts and Bext/Jint probe locations for a run.

    Inputs:
        scripts: Ordered post-processing script selections.
        bext_probes: Probe points for exterior magnetic field (Bext) [m].
        jint_probes: Probe points for interior current density (Jint) [m].

    Outputs:
        ResultsConfig: Immutable post-processing scripts and probe configuration.
    """

    scripts: list[ScriptSelection] = field(default_factory=list)
    bext_probes: list[ProbePoint] = field(default_factory=list)
    jint_probes: list[ProbePoint] = field(default_factory=list)

    def processing_config(self) -> dict[str, Any]:
        """Brief: Shape results settings into the post-processing config schema.

        Inputs:
            N/A: Uses scripts and probe lists on ``self``.

        Outputs:
            dict[str, Any]: Mapping with ``scripts`` and nested ``probe_points``.
        """
        scripts: list[dict[str, Any]] = []
        for script in self.scripts:
            entry: dict[str, Any] = {"name": script.name}
            if script.parameters:
                entry["parameters"] = script.parameters
            scripts.append(entry)
        return {
            "scripts": scripts,
            "probe_points": {
                "bext": [point.to_dict() for point in self.bext_probes],
                "jint": [point.to_dict() for point in self.jint_probes],
            },
        }

@dataclass(frozen=True)
class RunWorkspace:
    """Brief: Canonical on-disk paths for a FEM run or sweep project root.

    Inputs:
        root: Absolute or relative path to the run/sweep project directory.

    Outputs:
        RunWorkspace: Derived paths for geometry, SIF, results, and sweep artifacts.
    """

    root: Path

    @property
    def geometry_dir(self) -> Path:
        """Brief: Directory holding generated geometry artifacts.

        Inputs:
            N/A: Derived from ``self.root``.

        Outputs:
            Path: ``root/geometry``.
        """
        return self.root / "geometry"

    @property
    def results_dir(self) -> Path:
        """Brief: Directory holding solver and post-processing results.

        Inputs:
            N/A: Derived from ``self.root``.

        Outputs:
            Path: ``root/results``.
        """
        return self.root / "results"

    @property
    def geometry_file(self) -> Path:
        """Brief: Path to the Gmsh ``.geo`` case geometry file.

        Inputs:
            N/A: Derived from ``self.root``.

        Outputs:
            Path: ``root/geometry.geo``.
        """
        return self.root / "geometry.geo"

    @property
    def case_file(self) -> Path:
        """Brief: Path to the Elmer Solver Input File (``case.sif``).

        Inputs:
            N/A: Derived from ``self.root``.

        Outputs:
            Path: ``root/case.sif``.
        """
        return self.root / "case.sif"

    @property
    def manifest_file(self) -> Path:
        """Brief: Path to the project manifest JSON (``project.json``).

        Inputs:
            N/A: Derived from ``self.root``.

        Outputs:
            Path: ``root/project.json``.
        """
        return self.root / "project.json"

    @property
    def startinfo_file(self) -> Path:
        """Brief: Path to Elmer's ``ELMERSOLVER_STARTINFO`` launcher file.

        Inputs:
            N/A: Derived from ``self.root``.

        Outputs:
            Path: ``root/ELMERSOLVER_STARTINFO``.
        """
        return self.root / "ELMERSOLVER_STARTINFO"

    @property
    def processing_config_file(self) -> Path:
        """Brief: Path to the post-processing configuration JSON under results.

        Inputs:
            N/A: Derived from ``self.results_dir``.

        Outputs:
            Path: ``results/processing_config.json``.

    @property

        Inputs:
            N/A: Derived from ``self.root``.

        Outputs:

@dataclass(frozen=True)
class RunManifest:
    """Brief: Persisted project.json contents describing a FEM run's full configuration.

    Inputs:
        name: Human-readable run/project name.
        run_path: Filesystem path string for the run directory.
        geometry: Geometry specification for the case.
        simulation: Full simulation configuration (materials, BCs, solver).
        results: Post-processing scripts and probe configuration.
        materials: Material definition mapping keyed by role/name.
        material_roles: Optional mapping of geometry roles to material names.
        created: Timestamp when the manifest was first created.
        last_modified: Timestamp of the latest manifest update.
        status: Current ``RunStatus`` lifecycle value.

    Outputs:
        RunManifest: Immutable ``project.json`` representation of a FEM run.
    """

    name: str
    run_path: str
    geometry: GeometrySpec
    simulation: SimulationConfig
    results: ResultsConfig
    materials: dict[str, Any] = field(default_factory=dict)
    material_roles: dict[str, Any] = field(default_factory=dict)
    created: str = field(default_factory=lambda: datetime.now().isoformat())
    last_modified: str = field(default_factory=lambda: datetime.now().isoformat())
    status: RunStatus = RunStatus.CREATED

    @classmethod
    def from_dict(cls, data: dict[str, Any]) -> "RunManifest":
        """Brief: Deserialize a run manifest from a ``project.json`` mapping.

        Inputs:
            data: Nested mapping of name, paths, geometry, simulation, materials, results,
                and status.

        Outputs:
            RunManifest: Immutable run manifest.
        """
        status = data.get("status", RunStatus.CREATED.value)
        return cls(
            name=data.get("name", ""),
            created=data.get("created", datetime.now().isoformat()),
            last_modified=data.get("last_modified", datetime.now().isoformat()),
            run_path=data.get("run_path", ""),
            geometry=GeometrySpec.from_dict(data.get("geometry", {})),
            simulation=SimulationConfig.from_dict(data.get("simulation", {})),
            materials=dict(data.get("materials", {})),
            material_roles=dict(data.get("material_roles", {})),
            results=ResultsConfig.from_dict(data.get("results", {})),
            status=RunStatus(status),
        )

    def to_dict(self) -> dict[str, Any]:
        """Brief: Serialize the run manifest for ``project.json`` persistence.

        Inputs:
            N/A: Uses fields on ``self``.

        Outputs:
            dict[str, Any]: JSON-serializable project manifest mapping.
        """
        data = {
            "name": self.name,
            "created": self.created,
            "last_modified": self.last_modified,
            "run_path": self.run_path,
            "status": self.status.value,
            "geometry": self.geometry.to_dict(),
            "simulation": self.simulation.to_dict(),
            "materials": self.materials,
            "results": self.results.to_dict(),
        }
        if self.material_roles:
            data["material_roles"] = self.material_roles
        return data

    def updated(
        self, *, status: RunStatus | None = None, results: ResultsConfig | None = None
    ) -> "RunManifest":
        """Brief: Return a copy with optional status/results updates and fresh timestamp.

        Inputs:
            status: Optional new ``RunStatus``; keeps current status when omitted.
            results: Optional new ``ResultsConfig``; keeps current results when omitted.

        Outputs:
            RunManifest: New immutable manifest with ``last_modified`` set to now.
        """
        return RunManifest(
            name=self.name,
            created=self.created,
            last_modified=datetime.now().isoformat(),
            run_path=self.run_path,
            geometry=self.geometry,
            simulation=self.simulation,
            materials=self.materials,
            material_roles=self.material_roles,
            results=results if results is not None else self.results,
            status=status or self.status,
        )
