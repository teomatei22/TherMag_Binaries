    def resolve_run_dir(self, manifest_dir: Path) -> Path:
        """Brief: Resolve the on-disk run directory relative to the sweep manifest.

        Inputs:
            manifest_dir: Directory containing the sweep manifest file.

        Outputs:
            Path: Existing run directory when found; otherwise the stored ``run_dir`` path.
        """
        run_dir = Path(self.run_dir)
        if run_dir.is_dir():
            return run_dir

        candidates: list[Path] = []
        if run_dir.name:
            candidates.append(manifest_dir / run_dir.name)
        candidates.append(manifest_dir / f"{self.swept_param}_{self.coeff:.3f}")

        for candidate in candidates:
            if candidate.is_dir():
                return candidate

        return run_dir

@dataclass(frozen=True)
class SweepManifest:
    """Brief: Sweep metadata (parameter, base geometry, observables) and per-run entries.

    Inputs:
        swept_param: Name of the swept geometry parameter (e.g. ``L``).
        observables: Observable names collected across sweep runs.
        base_val: Base absolute value of the swept parameter before scaling [m].
        L0: Reference cylinder/sphere length used for scaling [m].
        R0: Reference radius used for scaling [m].
        runs: Ordered list of per-coefficient ``SweepRunEntry`` records.

    Outputs:
        SweepManifest: Immutable sweep project metadata and run index.
    """

    swept_param: str
    observables: list[str]
    base_val: float
    L0: float
    R0: float
    runs: list[SweepRunEntry] = field(default_factory=list)

    @classmethod
    def from_dict(cls, data: dict[str, Any]) -> SweepManifest:
        """Brief: Deserialize a sweep manifest from a mapping.

        Inputs:
            data: Mapping with ``swept_param``, ``base_val``, ``L0``, ``R0``, and ``runs``.

        Outputs:
            SweepManifest: Immutable sweep manifest.
        """
        return cls(
            swept_param=str(data["swept_param"]),
            observables=[str(item) for item in data.get("observables", [])],
            base_val=float(data["base_val"]),
            L0=float(data["L0"]),
            R0=float(data["R0"]),
            runs=[SweepRunEntry.from_dict(item) for item in data.get("runs", [])],
        )
