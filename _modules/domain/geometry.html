    @classmethod
    def from_value(cls, value: Any) -> "ProbePoint":
        """Brief: Build a probe from a ProbePoint, dict, or length-3 sequence.

        Inputs:
            value: Existing ``ProbePoint``, ``{x,y,z}`` mapping, or ``(x,y,z)`` sequence [m].

        Outputs:
            ProbePoint: Normalized immutable probe point.
        """
        if isinstance(value, cls):
            return value
        if isinstance(value, dict):
            return cls(
                x=float(value.get("x", 0.0)),
                y=float(value.get("y", 0.0)),
                z=float(value.get("z", 0.0)),
            )
        if isinstance(value, (list, tuple)) and len(value) == 3:
            return cls(float(value[0]), float(value[1]), float(value[2]))
        raise ValueError(f"Cannot build ProbePoint from {value!r}")

@dataclass(frozen=True)
class GeometrySpec:
    """Brief: Named geometry type, parametric dimensions, and optional Gmsh template.

    Inputs:
        name: Human-readable geometry case name.
        geometry_type: Geometry family identifier (e.g. cylinder, sphere).
        parameters: Parametric dimensions keyed by name [m or dimensionless].
        variant: Geometry variant label (default ``full``).
        template_file: Optional path to a Gmsh ``.geo`` template.

    Outputs:
        GeometrySpec: Immutable named geometry specification for FEM cases.
    """

    name: str
    geometry_type: str
    parameters: dict[str, float] = field(default_factory=dict)
    variant: str = "full"
    template_file: Path | None = None

    @classmethod
    def from_dict(cls, data: dict[str, Any]) -> "GeometrySpec":
        """Brief: Deserialize a geometry specification from a mapping.

        Inputs:
            data: Mapping with ``name``, ``geometry_type``, ``parameters``, optional
                ``variant`` and ``template_file``.

        Outputs:
            GeometrySpec: Immutable geometry specification.
        """
        parameters = {key: float(value) for key, value in data.get("parameters", {}).items()}
        template_file = data.get("template_file")
        return cls(
            name=data.get("name", ""),
            geometry_type=data.get("geometry_type", "unknown"),
            parameters=parameters,
            variant=data.get("variant", "full"),
            template_file=Path(template_file) if template_file else None,
        )

    def to_dict(self) -> dict[str, Any]:
        """Brief: Serialize the geometry specification for manifest persistence.

        Inputs:
            N/A: Uses geometry fields on ``self``.

        Outputs:
            dict[str, Any]: JSON-serializable geometry mapping.
        """
        result: dict[str, Any] = {
            "name": self.name,
            "geometry_type": self.geometry_type,
            "variant": self.variant,
            "parameters": dict(self.parameters),
        }
        if self.template_file:
            result["template_file"] = str(self.template_file)
        return result
