class CylinderVariant(Enum):
    """Brief: Topology variants for cylindrical geometries.

    Inputs:
        None: enumeration of solid, hollow, and layered cylinder layouts.

    Outputs:
        CylinderVariant: Enum member whose ``value`` is the registry variant key.
    """

    FULL = "full"
    HOLLOW = "hollow"
    LAYER_OVER_HOLLOW = "layer_over_hollow"
    LAYER_OVER_FULL = "layer_over_full"

class CylinderGeometry(BaseGeometry):
    """Brief: Gmsh cylinder geometry with ends as hot/cold and surrounding air.

    Inputs:
        variant: Topology selecting solid/hollow/layered templates.
        template_dir: Optional directory containing ``.geo`` templates [path].

    Outputs:
        CylinderGeometry: Configured geometry with physical volume/surface IDs.
    """

    geometry_type = "cylinder"

    VARIANT_CONFIGS = {
        CylinderVariant.FULL: {
            "name": "Full Cylinder",
            "description": "Full cylinder with surrounding spherical air",
            "template_name": "full_cylinder.geo",
            "volumes": {"outer_material": 1, "outer_air": 2},
            "surfaces": {"hot": 3, "cold": 4, "infinity": 5},
        },
        CylinderVariant.HOLLOW: {
            "name": "Hollow Cylinder",
            "description": "Hollow cylinder with inner air and surrounding spherical air",
            "template_name": "hollow_cylinder.geo",
            "volumes": {"outer_material": 1, "inner_air": 2, "outer_air": 3},
            "surfaces": {"hot": 4, "cold": 5, "infinity": 6},
        },
        CylinderVariant.LAYER_OVER_HOLLOW: {
            "name": "Layer Over Hollow Shape",
            "description": "Outer foil layer wrapping a hollow tube",
            "template_name": "layer_over_hollow_cylinder.geo",
            "volumes": {"outer_material": 1, "inner_material": 2, "inner_air": 3, "outer_air": 4},
            "surfaces": {"hot": 5, "cold": 6, "infinity": 7},
        },
        CylinderVariant.LAYER_OVER_FULL: {
            "name": "Layer Over Full Shape",
            "description": "Outer foil layer wrapping a full solid cylinder interior",
            "template_name": "layer_over_full_cylinder.geo",
            "volumes": {"outer_material": 1, "inner_material": 2, "outer_air": 3},
            "surfaces": {"hot": 4, "cold": 5, "infinity": 6},
        },
    }

    def __init__(self, variant: CylinderVariant, template_dir: Path | None = None):
        """Brief: Bind variant config and resolve the matching ``.geo`` template.

        Inputs:
            variant: Cylinder topology enum member.
            template_dir: Directory of cylinder templates; package default if omitted [path].

        Outputs:
            None: Constructs ``CylinderGeometry`` with name, description, and specs.
        """
        self.variant = variant
        self.config = self.VARIANT_CONFIGS[variant]
        self.name = self.config["name"]
        self.description = self.config["description"]

        if template_dir:
            template_file = template_dir / self.config["template_name"]
        else:
            template_file = Path(__file__).parent / "templates" / self.config["template_name"]

        super().__init__(template_file)

    def validate_parameters(self, params: dict[str, float]) -> tuple:
        """Brief: Enforce that wall/foil thickness leaves a positive inner radius.

        Inputs:
            params: Candidate cylinder parameters including ``R`` and thicknesses [m].

        Outputs:
            tuple: ``(valid, errors)`` where errors lists geometric violations.
        """
        valid, errors = super().validate_parameters(params)

        radius = params.get("R", self._parameters["R"].default_value)
        thickness = 0.0
        if self.variant == CylinderVariant.HOLLOW:
            thickness = params.get("T_outer", self._parameters["T_outer"].default_value)
        elif self.variant == CylinderVariant.LAYER_OVER_FULL:
            thickness = params.get("T_outer", self._parameters["T_outer"].default_value)
        elif self.variant == CylinderVariant.LAYER_OVER_HOLLOW:
            thickness = (
                params.get("T_inner", self._parameters["T_inner"].default_value)
                + params.get("T_outer", self._parameters["T_outer"].default_value)
            )

        if thickness >= radius:
            errors.append(
                f"Combined radial thickness ({thickness:.4f}m) must be < R ({radius:.4f}m)"
            )
            valid = False

        return valid, errors
