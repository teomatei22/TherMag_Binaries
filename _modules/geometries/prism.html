class PrismVariant(Enum):
    """Brief: Topology variants for rectangular prism geometries.

    Inputs:
        None: enumeration of solid, hollow, and layered prism layouts.

    Outputs:
        PrismVariant: Enum member whose ``value`` is the registry variant key.
    """

    FULL = "full"
    HOLLOW = "hollow"
    LAYER_OVER_HOLLOW = "layer_over_hollow"
    LAYER_OVER_FULL = "layer_over_full"

class PrismGeometry(BaseGeometry):
    """Brief: Gmsh rectangular prism with end faces as hot/cold and air sphere.

    Inputs:
        variant: Topology selecting solid/hollow/layered templates.
        template_dir: Optional directory containing ``.geo`` templates [path].

    Outputs:
        PrismGeometry: Configured geometry with physical volume/surface IDs.
    """

    geometry_type = "prism"

    VARIANT_CONFIGS = {
        PrismVariant.FULL: {
            "name": "Full Prism",
            "description": "Solid rectangular prism with surrounding spherical air",
            "template_name": "full_prism.geo",
            "volumes": {"outer_material": 1, "outer_air": 2},
            "surfaces": {"hot": 3, "cold": 4, "infinity": 5},
        },
        PrismVariant.HOLLOW: {
            "name": "Hollow Prism",
            "description": "Hollow rectangular prism with inner air cavity and surrounding spherical air",
            "template_name": "hollow_prism.geo",
            "volumes": {"outer_material": 1, "inner_air": 2, "outer_air": 3},
            "surfaces": {"hot": 4, "cold": 5, "infinity": 6},
        },
        PrismVariant.LAYER_OVER_HOLLOW: {
            "name": "Layer Over Hollow Shape",
            "description": "Outer foil layer wrapping a hollow tube",
            "template_name": "layer_over_hollow_prism.geo",
            "volumes": {"outer_material": 1, "inner_material": 2, "inner_air": 3, "outer_air": 4},
            "surfaces": {"hot": 5, "cold": 6, "infinity": 7},
        },
        PrismVariant.LAYER_OVER_FULL: {
            "name": "Layer Over Full Shape",
            "description": "Outer foil layer wrapping a full solid prism interior",
            "template_name": "layer_over_full_prism.geo",
            "volumes": {"outer_material": 1, "inner_material": 2, "outer_air": 3},
            "surfaces": {"hot": 4, "cold": 5, "infinity": 6},
        },
    }

    def __init__(self, variant: PrismVariant, template_dir: Path | None = None):
        """Brief: Bind variant config and resolve the matching ``.geo`` template.

        Inputs:
            variant: Prism topology enum member.
            template_dir: Directory of prism templates; package default if omitted [path].

        Outputs:
            None: Constructs ``PrismGeometry`` with name, description, and specs.
        """
        self.variant = variant
        self.config = self.VARIANT_CONFIGS[variant]
        self.name = self.config["name"]
        self.description = self.config["description"]

        if template_dir:
            template_file = template_dir / self.config["template_name"]
        else:
            template_file = Path(__file__).parent / "templates" / self.config["template_name"]

        super().__init__(template_file)

    def validate_parameters(self, params: dict[str, float]) -> tuple:
        """Brief: Enforce wall thickness and air-sphere containment of the prism.

        Inputs:
            params: Candidate prism parameters ``L``, ``W``, ``H``, thicknesses [m].

        Outputs:
            tuple: ``(valid, errors)`` where errors lists geometric violations.
        """
        valid, errors = super().validate_parameters(params)

        width = params.get("W", self._parameters["W"].default_value)
        height = params.get("H", self._parameters["H"].default_value)
        available = min(width, height)
        thickness = 0.0
        if self.variant == PrismVariant.HOLLOW:
            thickness = params.get("T_outer", self._parameters["T_outer"].default_value)
        elif self.variant == PrismVariant.LAYER_OVER_FULL:
            thickness = params.get("T_outer", self._parameters["T_outer"].default_value)
        elif self.variant == PrismVariant.LAYER_OVER_HOLLOW:
            thickness = (
                params.get("T_inner", self._parameters["T_inner"].default_value)
                + params.get("T_outer", self._parameters["T_outer"].default_value)
            )

        if thickness >= available:
            errors.append(
                f"Combined transverse thickness ({thickness:.4f}m) must be < "
                f"min(W, H) ({available:.4f}m)"
            )
            valid = False

        import math

        length = params.get("L", self._parameters["L"].default_value)
        r_sph = self.air_sphere_radius_m(params)
        min_sphere_radius = math.sqrt((length / 2) ** 2 + width**2 + height**2)
        if r_sph <= min_sphere_radius:
            errors.append(
                f"Computed air sphere radius ({r_sph:.4f}m) must be > {min_sphere_radius:.4f}m to contain the prism"
            )
            valid = False

        return valid, errors
