    def shape(self, geometry_key: str) -> ShapeDefaults:
        """Brief: Select shape defaults by geometry family key.

        Inputs:
            geometry_key: ``cylinder``, ``prism``, or ``sphere`` [dimensionless].

        Outputs:
            ShapeDefaults: Matching shape defaults record.
        """
        key = (geometry_key or "").lower()
        try:

                "cylinder": self.cylinder,
                "prism": self.prism,
                "sphere": self.sphere,

        except KeyError as exc:
            raise KeyError(f"Unknown geometry_key {geometry_key!r} in geometry defaults") from exc

def _parse_parameter(data: Mapping[str, Any], path: str) -> ParameterDefaults:
    """Brief: Parse and validate one parameter object from defaults JSON.

    Inputs:
        data: Raw parameter JSON object [dimensionless].
        path: Dot-path for nested validation errors [dimensionless].

    Outputs:
        ParameterDefaults: Validated immutable parameter defaults.
    """
    default_value = _number(data, "default_value", path)
    min_value = _optional_number(data, "min_value", path)
    max_value = _optional_number(data, "max_value", path)
    if min_value is not None and default_value < min_value:
        raise ValueError(f"{path}.default_value must be >= min_value")
    if max_value is not None and default_value > max_value:
        raise ValueError(f"{path}.default_value must be <= max_value")
    description = data.get("description")
    if not isinstance(description, str) or not description:
        raise ValueError(f"{path}.description must be a non-empty string")
    return ParameterDefaults(
        description=description,
        default_value=default_value,
        min_value=min_value,
        max_value=max_value,
        unit=str(data.get("unit", "")),
        step=_validate_positive(_number(data, "step", path), f"{path}.step")
        if "step" in data
        else 0.001,
        decimals=_integer(data, "decimals", path) if "decimals" in data else 6,
        read_only=_boolean(data, "read_only", path) if "read_only" in data else False,
    )

def _parse_parameter_map(data: Mapping[str, Any], path: str) -> Mapping[str, ParameterDefaults]:
    """Brief: Parse a mapping of parameter names to validated defaults.

    Inputs:
        data: JSON object of parameter entries [dimensionless].
        path: Dot-path prefix for nested validation errors [dimensionless].

    Outputs:
        Mapping[str, ParameterDefaults]: Read-only name-to-defaults map.
    """
    parsed: dict[str, ParameterDefaults] = {}
    for name, raw in data.items():
        if not isinstance(raw, dict):
            raise ValueError(f"{path}.{name} must be an object")
        parsed[name] = _parse_parameter(raw, f"{path}.{name}")
    return MappingProxyType(parsed)

def _parse_shape(data: Mapping[str, Any], path: str) -> ShapeDefaults:
    """Brief: Parse base and variant parameter maps for one shape family.

    Inputs:
        data: JSON object containing ``base_parameters`` and ``variant_parameters``.
        path: Dot-path prefix for nested validation errors [dimensionless].

    Outputs:
        ShapeDefaults: Validated shape defaults record.
    """
    variants_raw = _object(data, "variant_parameters", path)
    variants = {
        str(variant): _parse_parameter_map(parameters, f"{path}.variant_parameters.{variant}")
        for variant, parameters in variants_raw.items()
        if isinstance(parameters, dict)
    }
    if len(variants) != len(variants_raw):
        raise ValueError(f"{path}.variant_parameters values must be objects")
    return ShapeDefaults(
        base_parameters=_parse_parameter_map(
            _object(data, "base_parameters", path),
            f"{path}.base_parameters",
        ),
        variant_parameters=MappingProxyType(variants),
    )

def _parse_defaults(data: Mapping[str, Any]) -> GeometryDefaults:
    """Brief: Parse the top-level geometry defaults JSON object.

    Inputs:
        data: Root JSON object with ``cylinder``, ``prism``, and ``sphere`` keys.

    Outputs:

    return GeometryDefaults(

        prism=_parse_shape(_object(data, "prism", "defaults"), "prism"),
        sphere=_parse_shape(_object(data, "sphere", "defaults"), "sphere"),

@cache
def _load_defaults_file(path: Path) -> GeometryDefaults:
    """Brief: Load, parse, and cache geometry defaults from a JSON file path.

    Inputs:
        path: Absolute path to ``geometry_defaults.json`` [path].

    Outputs:
        GeometryDefaults: Cached validated defaults for that path.
    """
    if not path.is_file():
        raise FileNotFoundError(f"Missing defaults JSON: {path}")
    with path.open("r", encoding="utf-8") as handle:
        data = json.load(handle)
    if not isinstance(data, dict):
        raise ValueError(f"Defaults JSON must contain an object: {path}")
    return _parse_defaults(data)

def load_geometry_defaults(path: Path | None = None) -> GeometryDefaults:
    """Brief: Load geometry defaults from the package JSON or an override path.

    Inputs:
        path: Optional alternate defaults file; default is package sibling [path].

    Outputs:
        GeometryDefaults: Validated defaults container.
    """
    defaults_path = (
        Path(path) if path is not None else Path(__file__).with_name(DEFAULTS_FILENAME)
    ).resolve()
    return _load_defaults_file(defaults_path)

def build_parameter_specs(
    geometry_key: str, variant: str = "full"
) -> dict[str, ParameterSpec]:
    """Brief: Build ``ParameterSpec`` map for a shape family and variant.

    Inputs:
        geometry_key: ``cylinder``, ``prism``, or ``sphere`` [dimensionless].
        variant: Topology key such as ``full`` [dimensionless].

    Outputs:
        dict[str, ParameterSpec]: Name-to-spec mapping for that geometry variant.
    """
    shape = load_geometry_defaults().shape(geometry_key)
    return {
        name: entry.to_spec(name) for name, entry in shape.parameters(variant).items()
    }

def default_parameter_values(
    geometry_key: str, variant: str = "full"
) -> dict[str, float]:
    """Brief: Extract numeric default values for a shape family and variant.

        geometry_key: ``cylinder``, ``prism``, or ``sphere`` [dimensionless].
        variant: Topology key such as ``full`` [dimensionless].

    Outputs:
        dict[str, float]: Parameter name to default numeric value [mixed units].

        name: spec.default_value
        for name, spec in build_parameter_specs(geometry_key, variant).items()
    }

def parameter_value(
    params: Mapping[str, Any],
    geometry_key: str,
    name: str,
    *,
    variant: str = "full",
) -> float:
    """Brief: Read a parameter from ``params`` or fall back to defaults.

    Inputs:
        params: User-supplied parameter map that may omit keys [mixed units].
        geometry_key: Shape family used for default lookup [dimensionless].
        name: Parameter name to resolve [dimensionless].
        variant: Topology key for default lookup [dimensionless].

    Outputs:
        float: Resolved parameter value [matches parameter unit].
    """
    raw = params.get(name)
    if raw is not None:
        return float(raw)
    values = default_parameter_values(geometry_key, variant=variant)
    if name not in values:
        raise KeyError(f"Unknown parameter {name!r} for geometry {geometry_key!r}")
    return values[name]
