@dataclass
class ParameterSpec:
    """Brief: Declares a named geometry parameter with bounds and UI metadata.

    Inputs:
        name: Parameter identifier used in ``.geo`` templates [dimensionless].
        description: Human-readable meaning of the parameter [dimensionless].
        default_value: Numeric value when unset [matches ``unit``].
        min_value: Inclusive lower bound, if any [matches ``unit``].
        max_value: Inclusive upper bound, if any [matches ``unit``].
        unit: Physical unit label such as ``m`` or ``deg`` [dimensionless].
        step: Suggested UI increment [matches ``unit``].
        decimals: Digits retained when writing template values [dimensionless].
        read_only: Whether the GUI should forbid edits [dimensionless].

    Outputs:
        ParameterSpec: Frozen-style dataclass instance describing one parameter.
    """

    name: str
    description: str
    default_value: float
    min_value: float | None = None
    max_value: float | None = None
    unit: str = ""
    step: float = 0.001
    decimals: int = 6
    read_only: bool = False

    def validate(self, value: float) -> tuple[bool, str]:
        """Brief: Check that a candidate value lies within configured bounds.

        Inputs:
            value: Proposed parameter value [matches ``unit``].

        Outputs:
            tuple[bool, str]: ``(True, "")`` if valid, else ``(False, message)``.
        """
        if self.min_value is not None and value < self.min_value:
            return False, f"{self.name} must be >= {self.min_value}"
        if self.max_value is not None and value > self.max_value:
            return False, f"{self.name} must be <= {self.max_value}"
        return True, ""

def l_metal_max_m(
    geometry_type: str, params: Mapping[str, float], *, variant: str = "full"
) -> float:
    """Brief: Characteristic metal extent used for mesh/domain scaling.

    Inputs:
        geometry_type: Shape family ``cylinder``, ``prism``, or ``sphere`` [dimensionless].
        params: Geometry parameter map (lengths and radii) [m].
        variant: Topology variant such as ``full`` or ``hollow`` [dimensionless].

    Outputs:
        float: Maximum metal length scale for the shape [m].
    """
    from .defaults import parameter_value

    gt = (geometry_type or "cylinder").lower()
    variant_l = (variant or "full").lower()

    if gt == "prism":
        l_cyl = parameter_value(params, "prism", "L")
        w_cyl = parameter_value(params, "prism", "W")
        h_cyl = parameter_value(params, "prism", "H")
        return max(l_cyl, 2.0 * w_cyl, 2.0 * h_cyl)

    if gt == "sphere":
        r_sphere = parameter_value(params, "sphere", "R_sphere")
        if "hollow" in variant_l:
            return r_sphere
        return 2.0 * r_sphere

    l_cyl = parameter_value(params, "cylinder", "L")
    r_cyl = parameter_value(params, "cylinder", "R")
    return max(l_cyl, 2.0 * r_cyl)

def air_sphere_radius_m(
    geometry_type: str,
    params: Mapping[str, float],
    *,
    variant: str | None = None,
) -> float:
    """Brief: Resolve surrounding air-sphere radius via the geometry registry.

    Inputs:
        geometry_type: Shape family key such as ``cylinder`` [dimensionless].
        params: Geometry parameter map used by the ``.geo`` template [m].
        variant: Topology variant; defaults to ``full`` when omitted [dimensionless].

    Outputs:
        float: Outer air domain radius taken from generated geometry [m].
    """
    from .registry import get_geometry_registry

    variant_key = (variant or "full").lower()
    geometry_key = f"{variant_key}_{(geometry_type or 'cylinder').lower()}"
    registry = get_geometry_registry()
    resolved = registry.resolve_key(
        geometry_key,
        geometry_type=geometry_type,
        variant=variant_key,
    )
    return registry.air_sphere_radius_m(resolved, params)

class BaseGeometry(ABC):
    """Brief: Abstract Gmsh geometry: parameters, template fill, and physical IDs.

    Inputs:
        None: subclass sets ``geometry_type``, template path, and parameter specs.

    Outputs:
        BaseGeometry: Instance ready to validate params and emit ``.geo`` text.
    """

    geometry_type: str = "unknown"
    name: str = "Unknown Geometry"
    description: str = ""

    def __init__(self, template_file: Path | None = None):
        """Brief: Store template path and initialise parameter specifications.

        Inputs:
            template_file: Path to the Gmsh ``.geo`` template [path].

        Outputs:
            None: Constructs ``BaseGeometry`` with populated ``_parameters``.
        """
        self.template_file = template_file
        self._parameters: dict[str, ParameterSpec] = {}
        self._init_parameters()

    @property
    def parameters(self) -> dict[str, ParameterSpec]:
        """Brief: Expose the live parameter-spec dictionary for this geometry.

        Inputs:
            None: reads ``self._parameters``.

        Outputs:
            dict[str, ParameterSpec]: Name-to-spec mapping for GUI and validation.
        """
        return self._parameters

    @property
    def variant_key(self) -> str:
        """Brief: Normalize the topology variant to a lowercase registry key.

        Inputs:
            None: reads optional ``self.variant`` enum or string.

        Outputs:
            str: Variant key such as ``full`` or ``hollow`` [dimensionless].
        """
        variant = getattr(self, "variant", None)
        if variant is not None and hasattr(variant, "value"):
            return str(variant.value)
        return "full"

    def air_sphere_radius_m(self, params: Mapping[str, float]) -> float:
        """Brief: Parse outer air radius from generated ``.geo`` domain text.

        Inputs:
            params: Partial or full parameter overrides [m / deg as applicable].

        Outputs:
            float: Spherical air domain radius implied by the template [m].
        """
        from processing.scripts.geometry_contract import domain_radius_from_geo_text

        merged = {
            name: float(spec.default_value) for name, spec in self._parameters.items()
        }
        for key, value in params.items():
            if key not in self._parameters:
                continue
            try:
                merged[key] = float(value)
            except (TypeError, ValueError):
                continue
        return domain_radius_from_geo_text(self.generate_geo_content(merged))

    def validate_parameters(self, params: dict[str, float]) -> tuple[bool, list[str]]:
        """Brief: Validate all declared parameters against their specs.

        Inputs:
            params: Candidate parameter values keyed by name [matches units].

        Outputs:
            tuple[bool, list[str]]: Validity flag and list of error messages.
        """
        errors = []
        for name, spec in self._parameters.items():
            value = params.get(name, spec.default_value)
            valid, error = spec.validate(value)
            if not valid:
                errors.append(error)
        return len(errors) == 0, errors

    def load_template(self) -> str:
        """Brief: Read the Gmsh ``.geo`` template file from disk.

        Inputs:
            None: uses ``self.template_file``.

        Outputs:
            str: Raw template source text.
        """
        if not self.template_file or not self.template_file.exists():
            raise FileNotFoundError(f"Template file not found: {self.template_file}")

        with open(self.template_file, encoding="utf-8") as f:
            return f.read()

    def generate_geo_content(self, params: dict[str, float]) -> str:
        """Brief: Substitute numeric parameters into the Gmsh template.

        Inputs:
            params: Parameter name/value pairs to rewrite in assignments [m/deg].

        Outputs:
            str: Filled ``.geo`` content ready for meshing.
        """
        content = self.load_template()

        for param_name, param_value in params.items():
            spec = self._parameters.get(param_name)
            if isinstance(param_value, float):
                decimals = spec.decimals if spec is not None else 6
                formatted_value = f"{param_value:.{int(decimals)}f}"
            else:
                formatted_value = str(param_value)
            pattern = rf"\b({re.escape(param_name)}\s*=\s*)([0-9.eE+-]+)(\s*;)"
            replacement = rf"\g<1>{formatted_value}\3"
            content = re.sub(pattern, replacement, content)

        return content
