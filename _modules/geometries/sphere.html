def swept_geometry_key(geometry_type: str, swept_param: str) -> str:
    """Brief: Map a sweep axis choice onto the primary geometry parameter name.

    Inputs:
        geometry_type: Shape family ``sphere``, ``prism``, or ``cylinder`` [dimensionless].
        swept_param: Requested sweep parameter hint such as ``L`` or ``R`` [dimensionless].

    Outputs:
        str: Canonical parameter key to sweep (``L``, ``W``, or ``R``).
    """
    gtype = (geometry_type or "cylinder").lower()
    if gtype == "sphere":
        return "R_sphere"
    if gtype == "prism":
        return "L" if swept_param == "L" else "W"
    return "L" if swept_param == "L" else "R"

class SphereGeometry(BaseGeometry):
    """Brief: Gmsh sphere geometry with caps, layers, and surrounding air.

    Inputs:
        variant: Topology selecting solid/hollow/layered templates.
        template_dir: Optional directory containing ``.geo`` templates [path].

    Outputs:
        SphereGeometry: Configured geometry with physical volume/surface IDs.
    """

    geometry_type = "sphere"

    VARIANT_CONFIGS = {
        SphereVariant.FULL: {
            "name": "Full Sphere",
            "description": "Solid sphere with hot/cold cap patches and surrounding spherical air",
            "template_name": "full_sphere.geo",
            "volumes": {"outer_material": 1, "outer_air": 2},
            "surfaces": {"hot": 3, "cold": 4, "infinity": 5},
        },
        SphereVariant.HOLLOW: {
            "name": "Hollow Sphere",
            "description": "Hollow sphere shell with inner air cavity and surrounding spherical air",
            "template_name": "hollow_sphere.geo",
            "volumes": {"outer_material": 1, "inner_air": 2, "outer_air": 3},
            "surfaces": {"hot": 4, "cold": 5, "infinity": 6},
        },
        SphereVariant.LAYER_OVER_HOLLOW: {
            "name": "Layer Over Hollow Shape",
            "description": "Outer shell wrapping a hollow inner sphere",
            "template_name": "layer_over_hollow_sphere.geo",
            "volumes": {"outer_material": 1, "inner_material": 2, "inner_air": 3, "outer_air": 4},
            "surfaces": {"hot": 5, "cold": 6, "infinity": 7},
        },
        SphereVariant.LAYER_OVER_FULL: {
            "name": "Layer Over Full Shape",
            "description": "Outer shell wrapping a solid inner sphere",
            "template_name": "layer_over_full_sphere.geo",
            "volumes": {"outer_material": 1, "inner_material": 2, "outer_air": 3},
            "surfaces": {"hot": 4, "cold": 5, "infinity": 6},
        },
    }

    def __init__(self, variant: SphereVariant, template_dir: Path | None = None):
        """Brief: Bind variant config and resolve the matching ``.geo`` template.

        Inputs:
            variant: Sphere topology enum member.
            template_dir: Directory of sphere templates; package default if omitted [path].

        Outputs:
            None: Constructs ``SphereGeometry`` with name, description, and specs.
        """
        self.variant = variant
        self.config = self.VARIANT_CONFIGS[variant]
        self.name = self.config["name"]
        self.description = self.config["description"]

        if template_dir:
            template_file = template_dir / self.config["template_name"]
        else:
            template_file = Path(__file__).parent / "templates" / self.config["template_name"]

        super().__init__(template_file)

    def validate_parameters(self, params: dict[str, float]) -> tuple:
        """Brief: Enforce sphere cap, air, thickness, and non-overlap constraints.

        Inputs:
            params: Candidate sphere parameters including ``R_sphere``, ``R_cap``, angles [m/deg].

        Outputs:
            tuple: ``(valid, errors)`` where errors lists geometric violations.
        """
        valid, errors = super().validate_parameters(params)

        outer_radius = params.get("R_sphere", self._parameters["R_sphere"].default_value)
        cap_radius = params.get("R_cap", self._parameters["R_cap"].default_value)
        patch_angle_deg = params.get(
            "patch_angle_deg", self._parameters["patch_angle_deg"].default_value
        )
        air_radius = self.air_sphere_radius_m(params)

        if cap_radius >= outer_radius:
            errors.append(
                f"Cap radius ({cap_radius:.4f}m) must be smaller than sphere radius ({outer_radius:.4f}m)"
            )
            valid = False

        if air_radius <= outer_radius:
            errors.append(
                f"Computed air sphere radius ({air_radius:.4f}m) must be larger than sphere radius ({outer_radius:.4f}m)"
            )
            valid = False

        if outer_radius > 0 and cap_radius > 0:
            patch_half_angle = math.asin(min(1.0, cap_radius / outer_radius))
            patch_angle_rad = math.radians(patch_angle_deg)
            if 2.0 * patch_half_angle >= patch_angle_rad:
                errors.append(
                    "Cap patches overlap or touch; increase patch_angle_deg or reduce R_cap"
                )
                valid = False

            patch_center_chord = 2.0 * outer_radius * math.sin(0.5 * patch_angle_rad)
            if patch_center_chord <= 2.2 * cap_radius:
                errors.append(
                    "Hot/cold contact centres are too close; increase patch_angle_deg or reduce R_cap"
                )
                valid = False

        interface_radius: float | None = None
        cavity_radius: float | None = None
        if self.variant == SphereVariant.HOLLOW:
            thickness = params.get("T_outer", self._parameters["T_outer"].default_value)
            cavity_radius = outer_radius - thickness
        elif self.variant == SphereVariant.LAYER_OVER_FULL:
            thickness = params.get("T_outer", self._parameters["T_outer"].default_value)
            interface_radius = outer_radius - thickness
        elif self.variant == SphereVariant.LAYER_OVER_HOLLOW:
            outer_thickness = params.get(
                "T_outer", self._parameters["T_outer"].default_value
            )
            inner_thickness = params.get(
                "T_inner", self._parameters["T_inner"].default_value
            )
            interface_radius = outer_radius - outer_thickness
            cavity_radius = interface_radius - inner_thickness

        for label, radius in (
            ("layer-interface", interface_radius),
            ("cavity", cavity_radius),
        ):
            if radius is not None and radius <= 0.0:
                errors.append(
                    f"Derived {label} radius ({radius:.4f}m) must be positive"
                )
                valid = False

        contact_limit = cavity_radius if self.variant == SphereVariant.HOLLOW else interface_radius
        contact_limit_label = (
            "cavity" if self.variant == SphereVariant.HOLLOW else "layer-interface"
        )
        if contact_limit is not None and cap_radius >= contact_limit:
            errors.append(
                f"Cap radius ({cap_radius:.4f}m) must be smaller than the "
                f"derived {contact_limit_label} radius ({contact_limit:.4f}m)"
            )
            valid = False

        return valid, errors
