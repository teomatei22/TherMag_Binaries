class GeometryRegistry:
    """Brief: Catalog of available geometry variants keyed by template stem.

    Inputs:
        templates_dir: Optional directory of ``.geo`` templates [path].

    Outputs:
        GeometryRegistry: Instance with discovered geometries and lookup APIs.
    """

    def __init__(self, templates_dir: Path | None = None):
        """Brief: Set templates directory and discover present geometry files.

        Inputs:
            templates_dir: Override path to ``.geo`` templates [path].

        Outputs:
            None: Constructs registry with populated ``_geometries``.
        """
        self.templates_dir = templates_dir or (Path(__file__).parent / "templates")
        self._geometries: dict[str, BaseGeometry] = {}
        self._discover_geometries()

    def _discover_geometries(self) -> None:
        """Brief: Register all cylinder, prism, and sphere variants if files exist.

        Inputs:
            None: reads ``self.templates_dir``.

        Outputs:
            None: Mutates ``self._geometries``.
        """
        if not self.templates_dir.exists():
            return
        self._register_cylinder_variants()
        self._register_prism_variants()
        self._register_sphere_variants()

    def _register_cylinder_variants(self) -> None:
        """Brief: Add cylinder variants whose template files are present on disk.

        Inputs:
            None: iterates ``CylinderVariant`` against ``templates_dir``.

        Outputs:
            None: Inserts matching entries into ``self._geometries``.
        """
        for variant in CylinderVariant:
            geometry = CylinderGeometry(variant, self.templates_dir)
            if geometry.template_file.exists():
                self._geometries[geometry.template_file.stem] = geometry

    def get_geometry(self, name: str) -> BaseGeometry | None:
        """Brief: Look up a registered geometry by template-stem key.

        Inputs:
            name: Registry key such as ``full_cylinder`` [dimensionless].

        Outputs:
            BaseGeometry | None: Matching geometry or ``None`` if unknown.

    def resolve_key(
        self,
        name: str,
        *,
        template_file: str | Path | None = None,
        geometry_type: str = "",
        variant: str = "",
    ) -> str:
        """Brief: Normalize user-facing names to a concrete registry key.

        Inputs:
            name: Requested geometry name or stem [dimensionless].
            template_file: Optional ``.geo`` path whose stem may match [path].
            geometry_type: Shape family hint such as ``prism`` [dimensionless].
            variant: Topology hint such as ``hollow`` [dimensionless].

        Outputs:
            str: Resolved registry key, or original ``name`` if unresolved.
        """
        return resolve_geometry_registry_key(
            name,
            geometries=self._geometries,
            template_file=template_file,
            geometry_type=geometry_type,
            variant=variant,
        )

    def air_sphere_radius_m(self, geometry_name: str, params: Mapping[str, float]) -> float:
        """Brief: Compute air-sphere radius for a named registered geometry.

        Inputs:
            geometry_name: Registry key for the geometry [dimensionless].
            params: Geometry parameter values [m / deg as applicable].

        Outputs:
            float: Surrounding air domain radius [m].
        """
        geometry = self.get_geometry(geometry_name)
        if geometry is None:
            raise KeyError(f"Unknown geometry: {geometry_name!r}")
        return geometry.air_sphere_radius_m(params)

def resolve_geometry_registry_key(
    name: str,
    *,
    geometries: Mapping[str, BaseGeometry],
    template_file: str | Path | None = None,
    geometry_type: str = "",
    variant: str = "",
) -> str:
    """Brief: Resolve ambiguous geometry identifiers onto a registry key.

    Inputs:
        name: Direct key, display name, or slug [dimensionless].
        geometries: Mapping of known registry keys to geometry objects.
        template_file: Optional template path for stem matching [path].
        geometry_type: Optional shape family filter [dimensionless].
        variant: Optional topology filter [dimensionless].

    Outputs:
        str: Matching registry key, or ``name`` when no better match exists.
    """
    if name and name in geometries:
        return name

    if template_file:
        stem = Path(str(template_file)).stem
        if stem in geometries:
            return stem

    gt = (geometry_type or "").lower()
    var = (variant or "").lower()
    if gt and var:
        candidate = f"{var}_{gt}"
        if candidate in geometries:
            return candidate

    if name:
        name_l = name.strip().lower()
        for key, geo in geometries.items():
            if geo.name.strip().lower() == name_l and (not gt or geo.geometry_type.lower() == gt):
                if not var or getattr(getattr(geo, "variant", None), "value", "") == var:
                    return key
        slug = name_l.replace(" ", "_")
        if slug in geometries:
            return slug

    return name

def get_geometry_registry(templates_dir: Path | None = None) -> GeometryRegistry:
    """Brief: Return the process-wide geometry registry singleton.

    Inputs:
        templates_dir: If set, rebuilds the registry from that directory [path].

    Outputs:
        GeometryRegistry: Shared or freshly constructed registry instance.
    """
    global _registry
    if _registry is None or templates_dir is not None:
        _registry = GeometryRegistry(templates_dir)
    return _registry
