def _require_tensor(material_name: str, attribute: str, tensor: Sequence[float] | None) -> Sequence[float]:
    """Brief: Require a 3x3-style tensor sequence with at least five entries.

    Inputs:
        material_name: Material label used in error messages.
        attribute: Property attribute name being validated.
        tensor: Flattened tensor values; index 0 is parallel, 4 perpendicular.

    Outputs:
        Sequence[float]: The same tensor when it has length >= 5.
    """
    if tensor is None or len(tensor) < 5:
        raise ValueError(
            f"Material '{material_name}' is anisotropic but has no valid {attribute}"
        )
    return tensor

def build_material_params(material: Any, model: Any) -> dict[str, Any]:
    """Brief: Build validated SIF generator parameters for a material/model pair.

    Inputs:
        material: Material record exposing ``name`` and ``properties`` [mixed SI].
        model: Material model enum or string (``simple``, ``anisotropy``, …) [dimensionless].

    Outputs:
        dict[str, Any]: Parameter map consumed by ``SifGenerator`` material blocks.
    """
    props = material.properties
    model_name = str(getattr(model, "value", model)).lower()
    params: dict[str, Any] = {
        "name": material.name,
        "density": props.density,
        "heat_capacity": props.heat_capacity,
        "heat_conductivity": props.heat_conductivity,
        "electric_conductivity": props.electric_conductivity,
        "seebeck_coefficient": props.seebeck_coefficient,
        "relative_permittivity": props.relative_permittivity,
        "relative_permeability": props.relative_permeability,
        "heat_conductivity_table": props.heat_conductivity_table,
        "electric_conductivity_table": props.electric_conductivity_table,
    }

    if model_name == "anisotropy":
        tensor_fields = {
            "heat_conductivity_tensor": ("k_par", "k_perp"),
            "electric_conductivity_tensor": ("sigma_par", "sigma_perp"),
            "seebeck_coefficient_tensor": ("s_par", "s_perp"),
        }
        for attribute, (parallel_key, perpendicular_key) in tensor_fields.items():
            tensor = _require_tensor(
                material.name,
                attribute,
                getattr(props, attribute, None),
            )
            params[parallel_key] = float(tensor[0])
            params[perpendicular_key] = float(tensor[4])
    elif model_name == "hysteresis":
        hb_curve = getattr(props, "hb_curve", None)
        if hb_curve:
            lines = ["  H-B Curve = Variable", "\tReal"]
            lines.extend(f"    {h_value}       {b_value}" for h_value, b_value in hb_curve)
            lines.append("\tEnd")
            params["hb_curve_block"] = "\n".join(lines)
        else:
            params["hb_curve_block"] = "  ! No H-B curve data available"

    return params
