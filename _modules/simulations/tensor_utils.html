def _fmt(v):
    """Brief: Format a scalar for compact Elmer MATC expressions.

    Inputs:
        v: Numeric value to stringify [matches property SI unit].

    Outputs:
        str: Compact decimal or scientific representation without excess digits.
    """
    if v == 0:
        return "0"
    if isinstance(v, float) and v == int(v) and abs(v) < 1e15:
        iv = int(v)
        s_int = str(iv)

        if abs(iv) >= 1000:
            exp = 0
            t = abs(iv)
            while t % 10 == 0 and t > 0:
                t //= 10
                exp += 1
            s_sci = f"{'-' if iv < 0 else ''}{t}e{exp}"
            return s_sci if len(s_sci) < len(s_int) else s_int
        return s_int

    candidates = []
    for prec in range(4, 10):
        s = f"{v:.{prec}g}"

        s = s.replace("e+0", "e").replace("e-0", "e-")
        s = s.replace("e+", "e")
        candidates.append(s)
    return min(candidates, key=len)

def _add(v):
    """Brief: Format a signed addend for inline MATC arithmetic (leading ``+``/``-``).

    Inputs:
        v: Numeric addend [matches property SI unit].

    Outputs:
        str: Signed token such as ``+1.2`` or ``-3``.
    """
    s = _fmt(v)
    if s.startswith("-"):
        return s
    return "+" + s

def _sub(a, b):
    """Brief: Format the difference ``a - b`` as a compact MATC literal expression.

    Inputs:
        a: Minuend [matches property SI unit].
        b: Subtrahend [matches property SI unit].

    Outputs:
        str: Literal difference string embedded in MATC source.
    """
    return f"{_fmt(a)}{_add(-b)}"

def _matc_ident(*parts: str) -> str:
    """Brief: Build a safe Elmer MATC function identifier from name fragments.

    Inputs:
        parts: Fragments such as property name and numeric tokens [dimensionless].

    Outputs:
        str: Alphanumeric MATC identifier that does not start with a digit.
    """
    raw = "_".join(parts)
    out = []
    for ch in raw:
        if ch.isalnum():
            out.append(ch)
        elif ch == "-":
            out.append("m")
        else:
            out.append("p" if ch == "." else "_")
    ident = "".join(out)
    if not ident or ident[0].isdigit():
        ident = "f" + ident
    return ident

def _matc_cylinder(par, perp):
    """Brief: MATC expression for axial/radial anisotropy about the cylinder axis.

    Inputs:
        par: Parallel (axial) principal value [SI property unit].
        perp: Perpendicular (radial/tangential) principal value [SI property unit].

    Outputs:
        str: Single-line MATC body returning a flattened 3x3 tensor.
    """
    p = _fmt(par)
    ap = _add(par)

    sub = _sub(perp, par)
    expr = (
        f"y=tx(1);z=tx(2);"
        f"s=({sub})/(y*y+z*z+1e-9);"
        f"r=zeros(1,9);r(0)={p};"
        f"r(4)=s*y*y{ap};r(5)=s*y*z;r(7)=r(5);"
        f"r(8)=s*z*z{ap};r"
    )
    return expr

def _matc_prism(par, perp):
    """Brief: MATC expression for orthotropic prism tensors aligned to axes.

    Inputs:
        par: Parallel (length-axis) principal value [SI property unit].
        perp: Perpendicular (transverse) principal value [SI property unit].

    Outputs:
        str: Single-line MATC body returning a diagonal 3x3 tensor.
    """
    p = _fmt(par)
    q = _fmt(perp)
    return f"r=zeros(1,9);r(0)={p};r(4)={q};r(8)={q};r"

def _sphere_radial_function_body(par, perp, fname: str) -> str:
    """Brief: MATC function body for radially anisotropic spherical tensors.

    Inputs:
        par: Parallel (tangential) principal value [SI property unit].
        perp: Perpendicular (radial) principal value [SI property unit].
        fname: MATC function name assigned to the return variable [dimensionless].

    Outputs:
        str: MATC function body computing the local radial 3x3 tensor.
    """
    aq = _add(par)
    delta = _sub(perp, par)
    return (
        f'format(4,"rowform");'
        f"a=x(0);b=x(1);c=x(2);"
        f"s=({delta})/(a*a+b*b+c*c+1e-9);"
        f"r=zeros(1,9);"
        f"r(0)=s*a*a{aq};r(1)=s*a*b;r(2)=s*a*c;"
        f"r(3)=r(1);r(4)=s*b*b{aq};r(5)=s*b*c;"
        f"r(6)=r(2);r(7)=r(5);r(8)=s*c*c{aq};"
        f"_{fname}=r"
    )

def generate_cylinder_tensor(par, perp, prop_name):
    """Brief: Emit an Elmer SIF anisotropic tensor block for cylinder geometry.

    Inputs:
        par: Parallel principal conductivity/Seebeck value [SI].
        perp: Perpendicular principal value [SI].
        prop_name: Elmer property name such as ``Heat Conductivity`` [dimensionless].

    Outputs:
        str: SIF lines defining ``prop_name(3,3)`` via MATC.
    """
    matc = _matc_cylinder(par, perp)
    return f'  {prop_name}(3,3) = Variable Coordinate\n    Real MATC "{matc}"'

def generate_sphere_tensor(par, perp, prop_name):
    """Brief: Emit an Elmer SIF anisotropic tensor block for sphere geometry.

    Inputs:
        par: Parallel (tangential) principal conductivity/Seebeck value [SI].
        perp: Perpendicular (radial) principal value [SI].
        prop_name: Elmer property name such as ``Seebeck Coefficient`` [dimensionless].

    Outputs:
        str: SIF ``$ function`` plus ``prop_name(3,3)`` MATC call (≤127 chars).
    """
    fname = _matc_ident("sph", prop_name, _fmt(par), _fmt(perp))
    body = _sphere_radial_function_body(par, perp, fname)
    call = f"{fname}(tx)"
    if len(call) > 127:
        raise ValueError(f"Sphere MATC call exceeds Elmer limit 127: {call}")
    return (
        f"$ function {fname}(x) {{{body}}}\n"
        f"  {prop_name}(3,3) = Variable Coordinate\n"
        f'    Real MATC "{call}"'
    )

def generate_tensor_block(geometry_shape, prop_name, par, perp):
    """Brief: Dispatch tensor-block generation by conductor geometry shape.

    Inputs:
        geometry_shape: ``cylinder``, ``sphere``, or other (prism) [dimensionless].
        prop_name: Elmer property name for the tensor [dimensionless].
        par: Parallel principal value [SI].
        perp: Perpendicular principal value [SI].

    Outputs:
        str: Shape-appropriate anisotropic SIF property block.
    """
    if geometry_shape == "cylinder":
        return generate_cylinder_tensor(par, perp, prop_name)
    elif geometry_shape == "sphere":
        return generate_sphere_tensor(par, perp, prop_name)
    else:
        return generate_prism_tensor(par, perp, prop_name)

def generate_tensor_vector_product(geometry_shape, par, perp):
    """Brief: Emit MATC expression with shape-aware Seebeck anisotropy.

    Inputs:
        geometry_shape: ``cylinder``, ``prism``, or ``sphere`` [dimensionless].
        par: Parallel Seebeck principal value [V/K].
        perp: Perpendicular Seebeck principal value [V/K].

    Outputs:
        tuple[str, str, str]: MATC expressions for the three Cartesian product
            components ``(Jx_seebeck, Jy_seebeck, Jz_seebeck)``.
    """
    if geometry_shape == "cylinder":
        p = _fmt(par)
        sub = _sub(perp, par)

        denom = "(tx(1)*tx(1)+tx(2)*tx(2)+1e-9)"

        return (
            f"{p}*tx(3)",
            (
                f"({p}+({sub})*tx(1)*tx(1)/{denom})*tx(4)"
                f"+(({sub})*tx(1)*tx(2)/{denom})*tx(5)"
            ),
            (
                f"(({sub})*tx(1)*tx(2)/{denom})*tx(4)"
                f"+({p}+({sub})*tx(2)*tx(2)/{denom})*tx(5)"
            ),
        )

    if geometry_shape == "prism":
        p = _fmt(par)
        q = _fmt(perp)

        return (
            f"{p}*tx(3)",
            f"{q}*tx(4)",
            f"{q}*tx(5)",
        )

    if geometry_shape == "sphere":
        q = _fmt(par)
        delta = _sub(perp, par)

        denom = "(tx(0)*tx(0)+tx(1)*tx(1)+tx(2)*tx(2)+1e-9)"

        return (
            (
                f"({q}+({delta})*tx(0)*tx(0)/{denom})*tx(3)"
                f"+(({delta})*tx(0)*tx(1)/{denom})*tx(4)"
                f"+(({delta})*tx(0)*tx(2)/{denom})*tx(5)"
            ),
            (
                f"(({delta})*tx(0)*tx(1)/{denom})*tx(3)"
                f"+({q}+({delta})*tx(1)*tx(1)/{denom})*tx(4)"
                f"+(({delta})*tx(1)*tx(2)/{denom})*tx(5)"
            ),
            (
                f"(({delta})*tx(0)*tx(2)/{denom})*tx(3)"
                f"+(({delta})*tx(1)*tx(2)/{denom})*tx(4)"
                f"+({q}+({delta})*tx(2)*tx(2)/{denom})*tx(5)"
            ),
        )

    raise NotImplementedError(
        f"Tensor-vector product not yet implemented for {geometry_shape}"
    )
