    def __init__(self, base_dir: Path | None = None):

        self.drivers_dir = base_dir / "drivers"
        self.materials_dir = base_dir / "materials"
        self.solver_blocks_dir = base_dir / "solver_blocks"

    def get_driver_params_defaults(self, driver: DriverType) -> DriverParams:
        """Brief: Build default driver temperatures and runtime from app defaults.

        Inputs:
            driver: Driver kind selecting defaults subsection [dimensionless].

        Outputs:
            DriverParams: Defaults including derived mean/amplitude and timesteps.
        """
        defaults = load_application_defaults()
        try:
            drv_defaults = defaults.drivers[driver.value]
        except KeyError as exc:
            raise KeyError(f"Missing defaults for driver: {driver.value}") from exc

        t_cold = drv_defaults.t_cold
        t_hot = drv_defaults.t_hot
        runtime = defaults.driver_runtime
        return DriverParams(
            t_cold=t_cold,
            t_hot=t_hot,
            t_mean=(t_hot + t_cold) / 2.0,
            t_amplitude=(t_hot - t_cold) / 2.0,
            frequency=None,
            t_min=t_cold,
            t_max=t_hot,
            period=drv_defaults.period,
            timestep_size=runtime.timestep_size,
            timestep_intervals=runtime.timestep_intervals,
            output_intervals=runtime.output_intervals,
        )

    def with_backend_runtime_defaults(self, params: DriverParams) -> DriverParams:
        """Brief: Overlay backend runtime policy onto caller-supplied driver params.

        Inputs:
            params: Caller driver params; may set ``physical_total_time`` [s].

        Outputs:
            DriverParams: Copy with enforced timestep size/intervals/output intervals.
        """
        runtime_defaults = self.runtime_defaults()
        timestep_size = float(runtime_defaults["timestep_size"])
        output_intervals = int(runtime_defaults["output_intervals"])
        if params.physical_total_time is not None:
            physical_total_time = float(params.physical_total_time)
            if physical_total_time <= 0.0:
                raise ValueError("physical_total_time must be > 0")
            timestep_intervals = max(1, int(math.ceil(physical_total_time / timestep_size)))
        else:
            timestep_intervals = int(runtime_defaults["timestep_intervals"])
        return DriverParams(
            t_cold=params.t_cold,
            t_hot=params.t_hot,
            t_mean=params.t_mean,
            t_amplitude=params.t_amplitude,
            frequency=params.frequency,
            t_min=params.t_min,
            t_max=params.t_max,
            period=params.period,
            timestep_size=timestep_size,
            timestep_intervals=timestep_intervals,
            output_intervals=output_intervals,
            physical_total_time=params.physical_total_time,
        )

    @staticmethod
    def _format_sif_number(value: Any) -> str:
        """Brief: Format a numeric value for Elmer Simulation-block assignment lines.

        Inputs:
            value: Int/float-like value, or fallback string [mixed].

        Outputs:
            str: Compact SIF-friendly numeric literal.
        """
        if isinstance(value, int):
            return str(value)
        try:
            value_f = float(value)
        except Exception:
            return str(value)
        if value_f.is_integer():
            return f"{value_f:.1f}" if abs(value_f) < 10.0 else str(int(value_f))
        return f"{value_f:g}"

    @staticmethod
    def _replace_or_insert_simulation_line(
        text: str,
        key: str,
        value: Any,
        insert_after_key: str,
    ) -> str:
        """Brief: Replace a Simulation key or insert it after an anchor line.

        Inputs:
            text: Full or partial SIF text containing a Simulation block.
            key: Simulation keyword such as ``Timestep Sizes`` [dimensionless].
            value: Numeric value to assign [matches key units].
            insert_after_key: Existing keyword after which to insert if missing.

        Outputs:
            str: Updated SIF text with the key present and set.
        """
        value_text = SifGenerator._format_sif_number(value)
        pattern = rf"(?mi)^(\s*{re.escape(key)}\s*=\s*).*$"
        replacement = rf"\g<1>{value_text}"
        new_text, count = re.subn(pattern, replacement, text)
        if count:
            return new_text

        after_pattern = rf"(?mi)^(\s*{re.escape(insert_after_key)}\s*=.*)$"
        new_text, count = re.subn(
            after_pattern,
            rf"\1\n  {key} = {value_text}",
            text,
            count=1,
        )
        if count:
            return new_text

        raise RuntimeError(
            f"Generated SIF is missing '{key}' and insertion anchor "
            f"'{insert_after_key}' was not found."
        )

    def _enforce_driver_runtime_policy(
        self,
        sif_content: str,
        params: DriverParams,
    ) -> str:
        """Brief: Force timestep size/intervals/output intervals to backend policy.

        Inputs:
            sif_content: Generated time-dependent SIF text.
            params: Driver params providing expected timestep interval count.

        Outputs:
            str: SIF text with runtime keywords rewritten and asserted.
        """
        runtime_defaults = self.runtime_defaults()
        sif_content = self._replace_or_insert_simulation_line(
            sif_content,
            "Timestep Sizes",
            runtime_defaults["timestep_size"],
            "BDF Order",
        )
        sif_content = self._replace_or_insert_simulation_line(
            sif_content,
            "Timestep Intervals",
            params.timestep_intervals,
            "Timestep Sizes",
        )
        sif_content = self._replace_or_insert_simulation_line(
            sif_content,
            "Output Intervals",
            runtime_defaults["output_intervals"],
            "Timestep Intervals",
        )
        self._assert_driver_runtime_policy(sif_content, params.timestep_intervals)
        return sif_content

    def _assert_driver_runtime_policy(self, sif_content: str, expected_intervals: int) -> None:
        """Brief: Verify generated SIF runtime keywords match application defaults.

        Inputs:
            sif_content: Time-dependent SIF text to inspect.
            expected_intervals: Required ``Timestep Intervals`` value.

        Outputs:
            None: Raises ``RuntimeError`` when policy is violated.
        """
        runtime_defaults = self.runtime_defaults()
        expected = {
            "Timestep Sizes": float(runtime_defaults["timestep_size"]),
            "Timestep Intervals": int(expected_intervals),
            "Output Intervals": int(runtime_defaults["output_intervals"]),
        }

        for key, expected_value in expected.items():
            match = re.search(rf"(?mi)^\s*{re.escape(key)}\s*=\s*([^\s!]+)", sif_content)
            if not match:
                raise RuntimeError(f"Generated time_dependent SIF is missing '{key}'.")

            raw_value = match.group(1)
            try:
                if isinstance(expected_value, int):
                    actual = int(float(raw_value))
                else:
                    actual = float(raw_value)
            except Exception as exc:
                raise RuntimeError(
                    f"Generated time_dependent SIF has non-numeric {key}: {raw_value!r}"
                ) from exc

            if actual != expected_value:
                raise RuntimeError(
                    f"Generated time_dependent SIF has {key} = {actual}, "
                    f"expected {expected_value} from application_defaults.json."
                )

    def generate(
        self,
        base_type: SifBaseType,
        driver: DriverType,
        outer_material_model: MaterialModel,
        outer_material_params: dict[str, Any],
        inner_material_model: MaterialModel | None = None,
        inner_material_params: dict[str, Any] | None = None,
        driver_params: DriverParams | None = None,
        air_params: dict[str, Any] | None = None,
        output_path: Path | None = None,
        geometry_shape: str = "cylinder",
        bc1_potential: float | None = None,
        bc2_potential: float | None = None,
    ) -> str:
        """Brief: Generate a complete coupled thermoelectric-magnetic Elmer SIF.

        Inputs:
            base_type: Mesh topology selecting base template and BC/body IDs.
            driver: Steady/harmonic/sawtooth driving schedule.
            outer_material_model: Constitutive model for the outer conductor.
            outer_material_params: Outer material property map.
            inner_material_model: Inner conductor model for layered topologies.
            inner_material_params: Inner material property map.
            driver_params: Optional temperatures/runtime; defaults if omitted.
            air_params: Required Air material property map.
            output_path: Optional filesystem path to write the SIF.
            geometry_shape: Shape used for anisotropic tensor orientation.
            bc1_potential: Optional electrode electric potential if set [V].
            bc2_potential: Cold electrode electric potential if set [V].

        Outputs:
            str: Fully substituted SIF source text.
        """
        if driver_params is None:
            driver_params = self.get_driver_params_defaults(driver)
        else:
            driver_params = self.with_backend_runtime_defaults(driver_params)

        is_twolayer = base_type in (
            SifBaseType.LAYER_OVER_HOLLOW,
            SifBaseType.LAYER_OVER_FULL,
        )

        if is_twolayer and (inner_material_model is None or inner_material_params is None):
            raise ValueError("Two-layer geometries require inner_material_model and inner_material_params")

        inner_seebeck = None
        if is_twolayer:
            inner_seebeck = float(inner_material_params["seebeck_coefficient"])

        base_content = self._load_base_template(base_type)
        topology = TOPOLOGY_CONFIGS[base_type]
        self._validate_base_template(base_content, base_type)
        bc_config = topology.boundary_config
        outer_seebeck = float(outer_material_params["seebeck_coefficient"])

        if outer_material_model == MaterialModel.ANISOTROPY:
            outer_body_force_block = self._generate_tensor_current_body_force(
                1,
                geometry_shape,
                float(outer_material_params["s_par"]),
                float(outer_material_params["s_perp"]),
            )
        else:
            outer_body_force_block = self._generate_scalar_current_body_force(
                1,
                outer_seebeck,
            )

        inner_body_force_block = ""
        if is_twolayer:
            if inner_material_model == MaterialModel.ANISOTROPY:
                inner_body_force_block = self._generate_tensor_current_body_force(
                    2,
                    geometry_shape,
                    float(inner_material_params["s_par"]),
                    float(inner_material_params["s_perp"]),
                )
            else:
                inner_body_force_block = self._generate_scalar_current_body_force(
                    2,
                    inner_seebeck,
                )

        if air_params is None:
            raise ValueError(
                "air_params is required (Air material block is generated from properties)"
            )

        outer_material_block = self._generate_material_block(outer_material_model, 1, outer_material_params, geometry_shape)

        if is_twolayer:
            inner_material_block = self._generate_material_block(inner_material_model, 2, inner_material_params, geometry_shape)
            air_block = self._generate_material_block(
                MaterialModel.SIMPLE, 3, air_params, geometry_shape
            )
        else:
            inner_material_block = ""
            air_block = self._generate_material_block(
                MaterialModel.SIMPLE, 2, air_params, geometry_shape
            )

        sim_block, bc_hot_block, bc_cold_block = self._generate_driver_blocks(
            driver,
            driver_params,
            bc_config,
            bc1_potential=bc1_potential,
            bc2_potential=bc2_potential,
        )

        output_name = self._generate_output_name()

        sif_content = base_content

        solver_blocks = self._build_solver_blocks(output_name)
        sif_content = sif_content.replace("{{SOLVER1_BLOCK}}", solver_blocks["thermoelectric"])
        sif_content = sif_content.replace("{{SOLVER2_BLOCK}}", solver_blocks["seebeck_current"])
        sif_content = sif_content.replace("{{SOLVER3_BLOCK}}", solver_blocks["current_density"])
        sif_content = sif_content.replace("{{SOLVER4_BLOCK}}", solver_blocks["whitney_av"])
        sif_content = sif_content.replace("{{SOLVER5_BLOCK}}", solver_blocks["magnetic_field"])
        sif_content = sif_content.replace("{{SOLVER6_BLOCK}}", solver_blocks["result_output"])

        sif_content = sif_content.replace("{{SIMULATION_BLOCK}}", sim_block)
        sif_content = sif_content.replace("{{OUTER_BODY_FORCE_BLOCK}}", outer_body_force_block)
        sif_content = sif_content.replace("{{INNER_BODY_FORCE_BLOCK}}", inner_body_force_block)
        sif_content = sif_content.replace("{{BC_HOT_BLOCK}}", bc_hot_block)
        sif_content = sif_content.replace("{{BC_COLD_BLOCK}}", bc_cold_block)

        t_ic = load_application_defaults().temperature.t_ic
        sif_content = sif_content.replace("{{T_IC}}", f"{t_ic:g}")

        sif_content = sif_content.replace("{{OUTER_MATERIAL_BLOCK}}", outer_material_block)
        sif_content = sif_content.replace("{{INNER_MATERIAL_BLOCK}}", inner_material_block)
        sif_content = sif_content.replace("{{MATERIAL_AIR_BLOCK}}", air_block)

        sif_content = sif_content.replace(
            "{{OUTER_MATERIAL_NAME}}", outer_material_params.get("name", "OuterMaterial")
        )
        if inner_material_params:
            sif_content = sif_content.replace(
                "{{INNER_MATERIAL_NAME}}", inner_material_params.get("name", "InnerMaterial")
            )

        sif_content = sif_content.replace("{{BC_INFINITY_NUM}}", str(bc_config.bc_infinity_num))
        sif_content = sif_content.replace(
            "{{BC_INFINITY_BOUNDARY}}", str(bc_config.bc_infinity_boundary)
        )
        sif_content = sif_content.replace("{{BC_INFINITY_TEMPERATURE}}", "")

        sif_content = sif_content.replace("{{OUTPUT_FILE_NAME}}", output_name)
        permittivity = load_application_defaults().physics_constants["permittivity_of_vacuum"]
        sif_content = re.sub(
            r"(Permittivity of Vacuum\s*=\s*Real\s+)[0-9.eE+-]+",
            rf"\g<1>{permittivity:g}",
            sif_content,
        )

        if driver != DriverType.STEADY:
            sif_content = self._enforce_driver_runtime_policy(sif_content, driver_params)

        if output_path:
            output_path.parent.mkdir(parents=True, exist_ok=True)

            with open(output_path, "w", encoding="utf-8", newline="\n") as f:
                f.write(sif_content)

        return sif_content

    def build_magnetic_warm_start_sif(self, base_sif: str) -> str:
        """Brief: Insert a linear WhitneyAV precursor before the nonlinear H-B solve.

        Inputs:
            base_sif: SIF that already defines at least one ``H-B Curve`` material.

        Outputs:
            str: Modified SIF with precursor Solver 4 and renumbered trailing solvers.
        """
        if not re.search(r"(?m)^\s*H-B Curve\s*=", base_sif):
            raise RuntimeError("In-memory warm-start requires at least one H-B Curve material")

        sif = re.sub(
            r"(?ms)^Equation\s+1\s*$\n.*?^End\s*$",
            (
                "Equation 1\n"
                '  Name = "CoupledPhysicsWarmStart"\n'
                "  Active Solvers(7) = 1 2 3 4 5 6 7\n"
                "End"
            ),
            base_sif,
            count=1,
        )

        for solver_num in (4, 5, 6):
            sif = self._remove_solver_comment_header(sif, solver_num)

        solver4 = self._extract_solver_block(sif, 4)
        solver5 = self._extract_solver_block(sif, 5)
        solver6 = self._extract_solver_block(sif, 6)

        nonlinear_whitney = self._renumber_solver_header(solver4, 5)
        nonlinear_whitney = re.sub(
            r'(?m)^(\s*Equation\s*=\s*")[^"]*(")',
            r"\1MGDynamics\2",
            nonlinear_whitney,
            count=1,
        )
        nonlinear_whitney = self._tune_post_warm_whitney(nonlinear_whitney)
        calc_fields = self._renumber_solver_header(solver5, 6)
        result_output = self._renumber_solver_header(solver6, 7)

        replacement = (
            self._solver_comment_header(4, "MagnetoDynamics AV Precursor")
            + "\n"
            + _MAGNETIC_PRECURSOR_BLOCK.rstrip()
            + "\n\n"
            + self._solver_comment_header(5, "MagnetoDynamics AV")
            + "\n"
            + nonlinear_whitney
            + "\n\n"
            + self._solver_comment_header(6, "Magnetic Field Calculation")
            + "\n"
            + calc_fields
            + "\n\n"
            + self._solver_comment_header(7, "ResultOutput")
            + "\n"
            + result_output
        )
        sif = sif.replace(solver4, replacement, 1)
        sif = sif.replace(solver5, "", 1)
        return sif.replace(solver6, "", 1)

    @staticmethod
    def _tune_post_warm_whitney(nonlinear_whitney: str) -> str:
        """Brief: Retune nonlinear WhitneyAV settings after a linear warm-start.

        Inputs:
            nonlinear_whitney: Renumbered WhitneyAV solver block text.

        Outputs:
            str: Solver block with After-Timestep exec and relaxed Newton limits.
        """
        if not re.search(r"(?m)^\s*Exec Solver\s*=", nonlinear_whitney):
            nonlinear_whitney = re.sub(
                r"(?m)^(Solver\s+\d+\s*)$",
                r"\1\n  Exec Solver = After Timestep",
                nonlinear_whitney,
                count=1,
            )
        else:
            nonlinear_whitney = re.sub(
                r"(?m)^(\s*Exec Solver\s*=\s*).*$",
                r"\1After Timestep",
                nonlinear_whitney,
                count=1,
            )

        if not re.search(
            r"(?m)^\s*Linear System Refactorize\s*=",
            nonlinear_whitney,
        ):
            nonlinear_whitney = re.sub(
                r"(?m)^(\s*Linear System Solver\s*=\s*.*)$",
                r"  Linear System Refactorize = Logical True\n\1",
                nonlinear_whitney,
                count=1,
            )
        else:
            nonlinear_whitney = re.sub(
                r"(?m)^(\s*Linear System Refactorize\s*=\s*).*$",
                r"\1Logical True",
                nonlinear_whitney,
                count=1,
            )

        nonlinear_whitney = re.sub(
            r"(?m)^(\s*Nonlinear System Max Iterations\s*=\s*).*$",
            rf"\g<1>{_POST_WARM_NL_MAX}",
            nonlinear_whitney,
            count=1,
        )
        if re.search(
            r"(?m)^\s*Nonlinear System Newton After Iterations\s*=",
            nonlinear_whitney,
        ):
            return re.sub(
                r"(?m)^(\s*Nonlinear System Newton After Iterations\s*=\s*).*$",
                rf"\g<1>{_POST_WARM_NEWTON_AFTER}",
                nonlinear_whitney,
                count=1,
            )
        return re.sub(
            r"(?m)^(\s*Newton-Raphson Iteration\s*=\s*.*)$",
            rf"\1\n  Nonlinear System Newton After Iterations = {_POST_WARM_NEWTON_AFTER}",
            nonlinear_whitney,
            count=1,
        )

    def _validate_base_template(self, content: str, base_type: SifBaseType) -> None:
        """Brief: Assert base template body IDs and infinity/IC placeholders.

        Inputs:
            content: Loaded base SIF template text.
            base_type: Topology whose expected outer-air body ID is checked.

        Outputs:
            None: Raises ``ValueError`` when the template contract is broken.
        """
        topology = TOPOLOGY_CONFIGS[base_type]

        outer_air_pattern = (
            r"Body\s+\d+\s*\n"
            r"(?:(?:  .*)\n)*?"
            r'  Name = "Outer Air"\n'
            r"(?:(?:  .*)\n)*?"
            rf"  Target Bodies\(1\) = {topology.outer_air_body}\b"
        )
        if not re.search(outer_air_pattern, content, flags=re.MULTILINE):
            raise ValueError(
                f"Base template {base_type.value!r} does not map the 'Outer Air' body "
                f"to Target Bodies(1) = {topology.outer_air_body}."
            )

        if "{{BC_INFINITY_BOUNDARY}}" not in content:
            raise ValueError(
                f"Base template {base_type.value!r} is missing the infinity-boundary placeholder."
            )
        if "{{BC_INFINITY_TEMPERATURE}}" not in content:
            raise ValueError(
                f"Base template {base_type.value!r} is missing {{BC_INFINITY_TEMPERATURE}}."
            )
        if "{{T_IC}}" not in content:
            raise ValueError(
                f"Base template {base_type.value!r} is missing {{T_IC}}."
            )

        infinity_count = content.count("Infinity BC = Logical True")
        if infinity_count != 1:
            raise ValueError(
                f"Base template {base_type.value!r} must contain exactly one infinity boundary, "
                f"found {infinity_count}."
            )

    def _generate_material_block(
        self,
        model: MaterialModel,
        mat_num: int,
        params: dict[str, Any],
        geometry_shape: str = "cylinder",
    ) -> str:
        """Brief: Fill a material snippet with properties and optional anisotropy tensors.

        Inputs:
            model: Constitutive model selecting the material snippet.
            mat_num: Elmer Material number to assign.
            params: Property map including scalars, tables, and tensor principals.
            geometry_shape: Shape controlling anisotropic tensor orientation.

        Outputs:
            str: Complete Material block ready for SIF insertion.
        """
        snippet = self._load_material_snippet(model)

        result = snippet.replace("{{MATERIAL_NUM}}", str(mat_num))
        mat_name = params.get("name", f"Material{mat_num}")
        result = result.replace("{{MATERIAL_NAME}}", mat_name)

        hc_def = self._generate_property_def(params, "heat_conductivity")
        result = result.replace("{{HEAT_CONDUCTIVITY_DEF}}", hc_def)

        ec_def = self._generate_property_def(params, "electric_conductivity")
        result = result.replace("{{ELECTRIC_CONDUCTIVITY_DEF}}", ec_def)

        sc_def = self._generate_property_def(params, "seebeck_coefficient")
        result = result.replace("{{SEEBECK_COEFFICIENT_DEF}}", sc_def)

        for key, value in params.items():
            placeholder = "{{" + key.upper() + "}}"
            if placeholder in result:
                result = result.replace(placeholder, str(value))

        if model == MaterialModel.ANISOTROPY:
            try:
                k_par = params["k_par"]
                k_perp = params["k_perp"]
                sigma_par = params["sigma_par"]
                sigma_perp = params["sigma_perp"]
                s_par = params["s_par"]
                s_perp = params["s_perp"]
            except KeyError as exc:
                raise ValueError(
                    f"Missing anisotropy parameter {exc} for {params.get('name', '<material>')}"
                ) from exc
            heat_block = tensor_utils.generate_tensor_block(
                geometry_shape, "Heat Conductivity", k_par, k_perp
            )
            electric_block = tensor_utils.generate_tensor_block(
                geometry_shape, "Electric Conductivity", sigma_par, sigma_perp
            )
            seebeck_block = tensor_utils.generate_tensor_block(
                geometry_shape, "Seebeck Coefficient", s_par, s_perp
            )
            result = result.replace("{{HEAT_CONDUCTIVITY_BLOCK}}", heat_block)
            result = result.replace("{{ELECTRIC_CONDUCTIVITY_BLOCK}}", electric_block)
            result = result.replace("{{SEEBECK_COEFFICIENT_BLOCK}}", seebeck_block)

        return result

    def _generate_property_def(self, params: dict[str, Any], key_prefix: str) -> str:
        """Brief: Emit a Real scalar or Temperature-table property definition.

        Inputs:
            params: Material parameter map that may include ``*_table`` data.
            key_prefix: Base property key such as ``heat_conductivity``.

        Outputs:
            str: Elmer property RHS (``Real ...`` or ``Variable Temperature`` table).
        """

        table_key = f"{key_prefix}_table"
        if table_key in params and params[table_key]:
            table_data = params[table_key]

            lines = ["Variable Temperature", "    Real"]
            for t, v in table_data:
                lines.append(f"      {t} {v}")
            lines.append("    End")
            return "\n".join(lines)

        if key_prefix not in params:
            raise ValueError(
                f"Missing required property '{key_prefix}' (and no '{table_key}' table) for {params.get('name', '<material>')}"
            )
        return f"Real {params[key_prefix]}"

    def _generate_driver_blocks(
        self,
        driver: DriverType,
        params: DriverParams,
        bc_config: BoundaryConfig,
        bc1_potential: float | None = None,
        bc2_potential: float | None = None,
    ) -> tuple:
        """Brief: Split a driver snippet into Simulation and hot/cold BC blocks.

        Inputs:
            driver: Driver selecting the snippet file.
            params: Temperatures and timestep values for placeholder substitution.
            bc_config: Boundary numbering for hot/cold target surfaces.
            bc1_potential: Optional hot-end Potential line value [V].
            bc2_potential: Optional cold-end Potential line value [V].

        Outputs:
            tuple: ``(sim_block, bc_hot_block, bc_cold_block)`` filled SIF sections.
        """
        snippet = self._load_driver_snippet(driver)
        sections = snippet.split("!---------------------------------------------------------")

        sim_block = ""
        bc_hot_block = ""
        bc_cold_block = ""

        for section in sections:
            if "Simulation" in section and "Simulation Type" in section:
                sim_block = "!---------------------------------------------------------" + section
            elif "BC Hot End" in section or "hot_end" in section.lower():
                bc_hot_block = (
                    "!---------------------------------------------------------" + section
                )
            elif "BC Cold End" in section or "cold_end" in section.lower():
                bc_cold_block = (
                    "!---------------------------------------------------------" + section
                )

        bc_hot_block = bc_hot_block.replace("{{BC_HOT_NUM}}", str(bc_config.bc_hot_num))
        bc_hot_block = bc_hot_block.replace("{{BC_HOT_BOUNDARY}}", str(bc_config.bc_hot_boundary))
        bc_cold_block = bc_cold_block.replace("{{BC_COLD_NUM}}", str(bc_config.bc_cold_num))
        bc_cold_block = bc_cold_block.replace(
            "{{BC_COLD_BOUNDARY}}", str(bc_config.bc_cold_boundary)
        )

        bc_hot_potential_line = (
            f"Potential = {bc1_potential:.3f}" if bc1_potential is not None else ""
        )
        bc_cold_potential_line = (
            f"Potential = {bc2_potential:.3f}" if bc2_potential is not None else ""
        )

        param_map = {
            "T_HOT": params.t_hot,
            "T_COLD": params.t_cold,
            "T_MEAN": params.t_mean,
            "T_AMPLITUDE": params.t_amplitude,
            "FREQUENCY": params.frequency,
            "T_MIN": params.t_min,
            "T_MAX": params.t_max,
            "PERIOD": params.period,
            "TIMESTEP_SIZE": params.timestep_size,
            "TIMESTEP_INTERVALS": params.timestep_intervals,
            "OUTPUT_INTERVALS": params.output_intervals,
            "BC_HOT_POTENTIAL_LINE": bc_hot_potential_line,
            "BC_COLD_POTENTIAL_LINE": bc_cold_potential_line,
        }

        for key, value in param_map.items():
            placeholder = "{{" + key + "}}"
            sim_block = sim_block.replace(placeholder, str(value))
            bc_hot_block = bc_hot_block.replace(placeholder, str(value))
            bc_cold_block = bc_cold_block.replace(placeholder, str(value))

        return sim_block, bc_hot_block, bc_cold_block
