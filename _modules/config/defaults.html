def _object(data: Mapping[str, Any], key: str, path: str) -> Mapping[str, Any]:
    """Brief: Require a nested mapping value at ``key``.

    Inputs:
        data: Parent mapping to read.
        key: Field name expected to hold an object.
        path: Dot-path used in validation error messages.

    Outputs:
        Mapping[str, Any]: The nested object at ``key``.
    """
    value = data.get(key)
    if not isinstance(value, dict):
        raise ValueError(f"{path}.{key} must be an object")
    return value

def _number(data: Mapping[str, Any], key: str, path: str) -> float:
    """Brief: Require a numeric (non-bool) value at ``key``.

    Inputs:
        data: Parent mapping to read.
        key: Field name expected to hold a number.
        path: Dot-path used in validation error messages.

    Outputs:
        float: Parsed numeric value.
    """
    value = data.get(key)
    if isinstance(value, bool) or not isinstance(value, (int, float)):
        raise ValueError(f"{path}.{key} must be a number")
    return float(value)

def _optional_number(data: Mapping[str, Any], key: str, path: str) -> float | None:
    """Brief: Read an optional numeric field, allowing ``None``.

    Inputs:
        data: Parent mapping to read.
        key: Field name that may be absent, null, or numeric.
        path: Dot-path used in validation error messages.

    Outputs:
        float | None: Parsed number, or ``None`` when the field is null.
    """
    value = data.get(key)
    if value is None:
        return None
    return _number(data, key, path)

def _parse_defaults(data: Mapping[str, Any]) -> ApplicationDefaults:
    """Brief: Parse and validate raw JSON defaults into typed dataclasses.

    Inputs:
        data: Root object from ``application_defaults.json``.

    Outputs:
        ApplicationDefaults: Immutable, validated defaults tree.
    """
    simulation = _object(data, "simulation", "defaults")
    limits = _object(simulation, "temperature_limits", "simulation")
    drivers_data = _object(simulation, "driver_defaults", "simulation")
    drivers: dict[str, DriverDefaults] = {}
    for name in ("steady", "harmonic", "sawtooth"):
        raw = _object(drivers_data, name, "simulation.driver_defaults")
        period = _optional_number(raw, "period", f"simulation.driver_defaults.{name}")
        drivers[name] = DriverDefaults(
            t_hot=_number(raw, "t_hot", f"simulation.driver_defaults.{name}"),
            t_cold=_number(raw, "t_cold", f"simulation.driver_defaults.{name}"),
            period=(
                _validate_positive(period, f"simulation.driver_defaults.{name}.period")
                if period is not None
                else None
            ),
        )

    runtime = _object(simulation, "driver_runtime", "simulation")
    time_dependent = _object(simulation, "time_dependent", "simulation")
    electric_bc = _object(simulation, "electric_bc", "simulation")
    solver = _object(simulation, "solver", "simulation")
    sweep = _object(data, "sweep", "defaults")
    bounds = _object(sweep, "bounds", "sweep")
    default_bounds = _object(bounds, "default", "sweep.bounds")
    length_bounds = _object(bounds, "cylinder_length", "sweep.bounds")
    radius_bounds = _object(bounds, "cylinder_radius", "sweep.bounds")
    prism_length_bounds = _object(bounds, "prism_length", "sweep.bounds")
    prism_width_bounds = _object(bounds, "prism_width", "sweep.bounds")
    sphere_bounds = _object(bounds, "sphere", "sweep.bounds")
    sif = _object(data, "sif", "defaults")
    application_runtime = _object(data, "runtime", "defaults")
    physics = _object(sif, "physics_constants", "sif")

    temperature = TemperatureDefaults(
        min_k=_number(limits, "min_k", "simulation.temperature_limits"),
        max_k=_number(limits, "max_k", "simulation.temperature_limits"),
        t_hot=drivers["steady"].t_hot,
        t_cold=drivers["steady"].t_cold,
        t_ic=_number(simulation, "temperature_ic", "simulation"),
    )
    if temperature.min_k >= temperature.max_k:
        raise ValueError("simulation.temperature_limits.min_k must be below max_k")
    if not (
        temperature.min_k <= temperature.t_cold <= temperature.max_k
        and temperature.min_k <= temperature.t_hot <= temperature.max_k
        and temperature.min_k <= temperature.t_ic <= temperature.max_k
    ):
        raise ValueError("default temperatures must be within configured limits")

    parsed_sweep = SweepDefaults(
        lower_coefficient=_number(sweep, "lower_coefficient", "sweep"),
        upper_coefficient=_number(sweep, "upper_coefficient", "sweep"),
        coefficient_step=_validate_positive(
            _number(sweep, "coefficient_step", "sweep"),
            "sweep.coefficient_step",
        ),
        length_period_coefficient=_validate_positive(
            _number(sweep, "length_period_coefficient", "sweep"),
            "sweep.length_period_coefficient",
        ),
        bounds=SweepBounds(
            default_lower_min=_number(default_bounds, "lower_min", "sweep.bounds.default"),
            default_upper_max=_number(default_bounds, "upper_max", "sweep.bounds.default"),
            cylinder_length_lower_min=_number(
                length_bounds,
                "lower_min",
                "sweep.bounds.cylinder_length",
            ),
            cylinder_length_upper_max=_number(
                length_bounds,
                "upper_max",
                "sweep.bounds.cylinder_length",
            ),
            cylinder_radius_lower_min=_number(
                radius_bounds,
                "lower_min",
                "sweep.bounds.cylinder_radius",
            ),
            cylinder_radius_upper_max=_number(
                radius_bounds,
                "upper_max",
                "sweep.bounds.cylinder_radius",
            ),
            prism_length_lower_min=_number(
                prism_length_bounds,
                "lower_min",
                "sweep.bounds.prism_length",
            ),
            prism_length_upper_max=_number(
                prism_length_bounds,
                "upper_max",
                "sweep.bounds.prism_length",
            ),
            prism_width_lower_min=_number(
                prism_width_bounds,
                "lower_min",
                "sweep.bounds.prism_width",
            ),
            prism_width_upper_max=_number(
                prism_width_bounds,
                "upper_max",
                "sweep.bounds.prism_width",
            ),
            sphere_lower_min=_number(sphere_bounds, "lower_min", "sweep.bounds.sphere"),
            sphere_upper_max=_number(sphere_bounds, "upper_max", "sweep.bounds.sphere"),
            sphere_shell_margin_m=_number(
                bounds,
                "sphere_shell_margin_m",
                "sweep.bounds",
            ),
        ),
    )
    if parsed_sweep.lower_coefficient > parsed_sweep.upper_coefficient:
        raise ValueError("sweep.lower_coefficient must not exceed upper_coefficient")

    return ApplicationDefaults(
        default_threads=_integer(application_runtime, "default_threads", "runtime"),
        temperature=temperature,
        drivers=MappingProxyType(drivers),
        driver_runtime=DriverRuntimeDefaults(
            timestep_size=_validate_positive(
                _number(runtime, "timestep_size", "simulation.driver_runtime"),
                "simulation.driver_runtime.timestep_size",
            ),
            timestep_intervals=_integer(
                runtime,
                "timestep_intervals",
                "simulation.driver_runtime",
            ),
            output_intervals=_integer(
                runtime,
                "output_intervals",
                "simulation.driver_runtime",
            ),
        ),
        time_dependent=TimeDependentDefaults(
            total_time=_integer(time_dependent, "total_time", "simulation.time_dependent"),
            driver_period=_validate_positive(
                _number(time_dependent, "driver_period", "simulation.time_dependent"),
                "simulation.time_dependent.driver_period",
            ),
        ),
        electric_bc=ElectricBCDefaults(
            bc1_enabled=_boolean(electric_bc, "bc1_enabled", "simulation.electric_bc"),
            bc1_potential=_optional_number(
                electric_bc,
                "bc1_potential",
                "simulation.electric_bc",
            ),
            bc2_enabled=_boolean(electric_bc, "bc2_enabled", "simulation.electric_bc"),
            bc2_potential=_optional_number(
                electric_bc,
                "bc2_potential",
                "simulation.electric_bc",
            ),
        ),
        solver=SolverDefaults(
            requested_processes=_integer(solver, "requested_processes", "simulation.solver"),
            max_partitions=_integer(solver, "max_partitions", "simulation.solver"),
            mesh_timeout=_integer(solver, "mesh_timeout", "simulation.solver"),
            solve_timeout=_integer(solver, "solve_timeout", "simulation.solver"),
            post_timeout=_integer(solver, "post_timeout", "simulation.solver"),
        ),
        sweep=parsed_sweep,
        bc_defaults=MappingProxyType(dict(_object(sif, "bc_defaults", "sif"))),
        physics_constants=MappingProxyType(
            {
                "permittivity_of_vacuum": _number(
                    physics,
                    "permittivity_of_vacuum",
                    "sif.physics_constants",
                )
            }
        ),
    )

@cache
def _load_defaults_file(path: Path) -> ApplicationDefaults:
    """Brief: Load, parse, and cache defaults from a resolved JSON path.

    Inputs:
        path: Absolute path to the defaults JSON file.

    Outputs:
        ApplicationDefaults: Cached validated defaults for ``path``.
    """
    if not path.is_file():
        raise FileNotFoundError(f"Missing defaults JSON: {path}")
    with path.open("r", encoding="utf-8") as handle:
        data = json.load(handle)
    if not isinstance(data, dict):
        raise ValueError(f"Defaults JSON must contain an object: {path}")
    return _parse_defaults(data)

def load_application_defaults(path: Path | None = None) -> ApplicationDefaults:
    """Brief: Load and validate defaults, caching each resolved file path.

    Inputs:
        path: Optional overrides path; defaults to packaged ``application_defaults.json``.

    Outputs:
        ApplicationDefaults: Validated application defaults.
    """
    defaults_path = (
        Path(path) if path is not None else Path(__file__).with_name(DEFAULTS_FILENAME)
    ).resolve()
    return _load_defaults_file(defaults_path)
