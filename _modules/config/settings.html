class ToolPaths:
    """Brief: Resolved filesystem paths to external FEM and post-processing tools.

    Inputs:
        gmsh: Path to the Gmsh executable, or ``None`` if not found.
        elmergrid: Path to ElmerGrid, or ``None`` if not found.
        elmersolver: Path to serial ElmerSolver, or ``None`` if not found.
        elmersolver_mpi: Path to MPI ElmerSolver, or ``None`` if not found.
        mpiexec: Path to ``mpiexec``/``mpirun``, or ``None`` if not found.
        pvpython: Path to ParaView ``pvpython``, or ``None`` if not found.

    Outputs:
        ToolPaths: Container of discovered external tool executable paths.
    """

    elmergrid: Path | None = None
    elmersolver: Path | None = None
    elmersolver_mpi: Path | None = None
    mpiexec: Path | None = None
    pvpython: Path | None = None

@dataclass
class Settings:
    """Brief: Process-wide settings including tool paths, runs directory, and timeouts.

    Inputs:
        platform: OS platform string (e.g. ``linux``, ``win32``).
        tools: Resolved paths to Gmsh, Elmer, MPI, and ParaView tools.
        default_runs_dir: Default directory for simulation run outputs.
        default_threads: Default worker/thread count from application defaults.
        post_processing_timeout: Post-processing stage timeout [s].

    Outputs:
        Settings: Process-wide runtime settings with detected tool paths.
    """

    platform: str = field(default_factory=lambda: sys.platform)

    tools: ToolPaths = field(default_factory=ToolPaths)

    default_runs_dir: Path = field(default_factory=lambda: Settings._get_default_runs_dir())

    default_threads: int = field(
        default_factory=lambda: load_application_defaults().default_threads
    )
    post_processing_timeout: int = field(
        default_factory=lambda: load_application_defaults().solver.post_timeout
    )

    @staticmethod
    def _get_default_runs_dir() -> Path:
        """Brief: Choose the default directory for simulation run outputs.

        Inputs:
            N/A: Uses platform and optional Windows ``config.ini``.

        Outputs:
            Path: Default runs directory for the current platform.
        """
        if sys.platform == "win32":
            import configparser

            config_file = Path(__file__).parent.parent / "config.ini"

            if config_file.exists():
                try:
                    config = configparser.ConfigParser()
                    config.read(config_file)
                    if "Paths" in config and "runs_dir" in config["Paths"]:
                        return Path(config["Paths"]["runs_dir"])
                except (OSError, configparser.Error):
                    pass

            docs = Path(os.path.expandvars("%USERPROFILE%\\Documents"))
            if docs.exists():
                return docs
            else:
                return Path.home() / "Documents"
        else:
            return Path.cwd() / "runs"

    def _detect_tools(self):
        """Brief: Resolve paths for Gmsh, Elmer, MPI, and ParaView executables.

        Inputs:
            N/A: Operates on ``self`` and the environment.

        Outputs:
            None: Assigns discovered paths onto ``self.tools``.
        """
        self.tools.gmsh = self._find_tool("gmsh")
        self.tools.elmergrid = self._find_tool("ElmerGrid") or self._find_tool("elmergrid")
        self.tools.elmersolver = self._find_tool("ElmerSolver") or self._find_tool("elmersolver")
        self.tools.elmersolver_mpi = self._find_tool("ElmerSolver_mpi") or self._find_tool(
            "elmersolver_mpi"
        )
        self.tools.mpiexec = self._find_tool("mpiexec") or self._find_tool("mpirun")
        self.tools.pvpython = self._find_pvpython()

    def _find_tool(self, name: str) -> Path | None:
        """Brief: Locate a named executable via env override, bundle, PATH, or platform paths.

        Inputs:
            name: Tool base name (e.g. ``gmsh``, ``ElmerSolver``).

        Outputs:
            Path | None: Absolute path to the executable, or ``None`` if not found.
        """
        env_var = f"{name.upper()}_PATH"
        override = os.environ.get(env_var)
        if override and Path(override).is_file():
            return Path(override)

        bundled = self._first_file(self._bundled_candidates(name))
        if bundled:
            return bundled

        for executable in self._executable_names(name):
            found = shutil.which(executable)
            if found:
                return Path(found)

        return self._first_file(self._platform_candidates(name))

    @staticmethod
    def _executable_names(name: str) -> tuple[str, ...]:
        """Brief: Build candidate executable names including Windows ``.exe`` variants.

        Inputs:
            name: Tool base name or already-suffixed executable name.

        Outputs:
            tuple[str, ...]: Ordered candidate executable names.
        """
        if name.lower().endswith(".exe"):
            return (name,)
        return (name, f"{name}.exe")

    @staticmethod
    def _first_file(candidates) -> Path | None:
        """Brief: Return the first existing file among candidate paths.

        Inputs:
            candidates: Iterable of ``Path`` candidates to test with ``is_file()``.

        Outputs:
            Path | None: First existing file path, or ``None``.
        """
        return next((path for path in candidates if path.is_file()), None)

    def _bundled_candidates(self, name: str) -> list[Path]:
        """Brief: List bundled ``bin/`` locations for a tool under the package tree.

        Inputs:
            name: Tool base name used to form executable filenames.

        Outputs:
            list[Path]: Candidate paths under the application ``bin`` directory.
        """
        bin_dir = Path(__file__).resolve().parent.parent / "bin"
        names = self._executable_names(name)
        subdirs = (
            Path(),
            Path("gmsh"),
            Path("elmerfem") / "bin",
            Path("elmer") / "bin",
            Path("paraview") / "bin",
        )
        return [bin_dir / subdir / executable for subdir in subdirs for executable in names]

    def _platform_candidates(self, name: str) -> list[Path]:
        """Brief: List OS-specific fallback install locations for a tool.

        Inputs:
            name: Tool base name (uses Windows MPI and ``ELMER_HOME`` when applicable).

        Outputs:
            list[Path]: Platform-specific candidate executable paths.
        """
        names = self._executable_names(name)
        if self.platform != "win32":
            bases = (Path("/usr/bin"), Path("/usr/local/bin"), Path.home() / "bin")
            return [base / executable for base in bases for executable in names]

        candidates: list[Path] = []
        if name.lower() in {"mpiexec", "mpirun"}:
            candidates.extend(
                [
                    Path(r"C:\Program Files\Microsoft MPI\Bin\mpiexec.exe"),
                    Path(r"C:\Program Files (x86)\Microsoft MPI\Bin\mpiexec.exe"),
                ]
            )
        elmer_home = os.environ.get("ELMER_HOME")
        if elmer_home:
            candidates.extend(Path(elmer_home) / "bin" / executable for executable in names)
        return candidates

    def _find_pvpython(self) -> Path | None:
        """Brief: Locate ParaView's ``pvpython`` for post-processing scripts.

        Inputs:
            N/A: Uses env vars, bundled bin, PATH, and platform install paths.

        Outputs:
            Path | None: Absolute path to ``pvpython``, or ``None`` if not found.
        """
        pvpython_env = os.environ.get("PVPYTHON_PATH") or os.environ.get("PARAVIEW_PVPYTHON")
        if pvpython_env:
            p = Path(pvpython_env)
            if p.is_file():
                return p

        bundled = self._first_file(self._bundled_candidates("pvpython"))
        if bundled:
            return bundled

        for name in self._executable_names("pvpython"):
            result = shutil.which(name)
            if result:
                return Path(result)

        if self.platform == "win32":
            pf86 = Path(os.environ.get("ProgramFiles(x86)", r"C:\Program Files (x86)"))
            pf = Path(os.environ.get("ProgramFiles", r"C:\Program Files"))
            fixed_candidates = [
                pf86 / "emc-theatre" / "bin" / "paraview" / "bin" / "pvpython.exe",
                pf / "emc-theatre" / "bin" / "paraview" / "bin" / "pvpython.exe",
                pf86 / "EMC-Theatre" / "bin" / "paraview" / "bin" / "pvpython.exe",
                pf / "EMC-Theatre" / "bin" / "paraview" / "bin" / "pvpython.exe",
            ]
            for exe in fixed_candidates:
                if exe.is_file():
                    return exe
            return None
        return self._first_file(
            [
                Path("/usr/bin/pvpython"),
                Path("/usr/local/bin/pvpython"),
                Path("/opt/ParaView/bin/pvpython"),
            ]
        )

    def get_file_manager_command(self) -> list[str]:
        """Brief: Return the platform command used to open a folder in a file manager.

        Inputs:
            N/A: Uses ``self.platform``.

        Outputs:
            list[str]: Command argv (e.g. ``explorer`` or ``xdg-open``).

            return ["explorer"]
        else:
            return ["xdg-open"]

    def check_dependencies(self) -> dict[str, bool]:
        """Brief: Report whether required external FEM tools were discovered.

        Inputs:
            N/A: Reads ``self.tools``.

        Outputs:
            dict[str, bool]: Availability flags keyed by tool group name.
        """
        return {
            "gmsh": self.tools.gmsh is not None,
            "elmergrid": self.tools.elmergrid is not None,
            "elmersolver": self.tools.elmersolver is not None
            or self.tools.elmersolver_mpi is not None,
            "mpi": self.tools.mpiexec is not None,
            "pvpython": self.tools.pvpython is not None,
        }

    def get_missing_tools(self) -> list[str]:
        """Brief: List tool groups that are unavailable on this machine.

        Inputs:
            N/A: Uses ``check_dependencies``.

        Outputs:
            list[str]: Names of missing dependency groups.
        """
        deps = self.check_dependencies()
        return [name for name, available in deps.items() if not available]

def get_settings() -> Settings:
    """Brief: Return the process-wide singleton ``Settings`` instance.

    Inputs:
        N/A: Initializes global ``_settings`` on first call.

    Outputs:
        Settings: Shared settings object with detected tool paths.
    """
    global _settings
    if _settings is None:
        _settings = Settings()
    return _settings
